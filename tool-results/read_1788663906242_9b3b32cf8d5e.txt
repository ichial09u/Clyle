     1→/**
     2→ * Skip-time lookup: OP/ED intervals for every catalog anime.
     3→ *
     4→ * Four tiers, best data wins:
     5→ *   1. Zoko's own per-episode `skip` payload (frame-accurate, but rare —
     6→ *      only some titles carry it)
     7→ *   2. The vendored AniSkip community dataset (skip-times.data.json,
     8→ *      built from aniskip's public dump — covers 42/43 catalog titles,
     9→ *      3,290 OP entries, highest-vote row per episode)
    10→ *   3. Per-anime consensus: openings are the SAME video across episodes,
    11→ *      so episodes missing from the dump reuse the median interval of
    12→ *      the anime's other episodes
    13→ *   4. LIVE AniSkip query — for brand-new episodes (and freshly imported
    14→ *      anime) that predate the vendored dump. Fetched once on first open,
    15→ *      cached on disk (positive entries forever, misses 7 days) and warmed
    16→ *      proactively by the catalog sync, so a newly imported episode
    17→ *      effectively "imports" its own skip intro with it. The player still
    18→ *      has its own last-resort 8-90s heuristic window for TV-length
    19→ *      episodes with no data at all.
    20→ *
    21→ * Server-side only (the dataset ships inside the server bundle).
    22→ */

    24→import { createHash } from "node:crypto";
    25→import { mkdir, readFile, writeFile } from "node:fs/promises";

    27→import { join } from "node:path";

    29→import { db } from "@/lib/db";

    32→export interface SkipInterval {
    33→  start: number;
    34→  end: number;
    35→}
    36→
    37→export interface SkipData {
    38→  intro?: SkipInterval;
    39→  outro?: SkipInterval;
    40→  /** where the data came from — handy for debugging */
    41→  source?: "zoko" | "scout" | "aniskip" | "aniskip-live" | "crowd";
    42→}
    43→
    44→type EpisodeEntry = { op?: [number, number]; ed?: [number, number] };
    45→type Dataset = Record<string, Record<string, EpisodeEntry>>;
    46→
    47→const data = skipDataset as unknown as Dataset;
    48→
    49→/** median across an anime's known episodes, per type */
    50→const consensusCache = new Map<string, { op?: [number, number]; ed?: [number, number] }>();
    51→
    52→function median(values: number[]): number | null {
    53→  if (!values.length) return null;
    54→  const s = [...values].sort((a, b) => a - b);
    55→  return s[Math.floor(s.length / 2)];
    56→}
    57→
    58→function consensus(malId: number): { op?: [number, number]; ed?: [number, number] } {
    59→  const key = String(malId);
    60→  const hit = consensusCache.get(key);
    61→  if (hit) return hit;
    62→
    63→  const out: { op?: [number, number]; ed?: [number, number] } = {};
    64→  const byEp = data[key];
    65→  if (byEp) {
    66→    for (const type of ["op", "ed"] as const) {
    67→      const starts: number[] = [];
    68→      const ends: number[] = [];
    69→      for (const entry of Object.values(byEp)) {
    70→        const iv = entry[type];
    71→        if (iv) {
    72→          starts.push(iv[0]);
    73→          ends.push(iv[1]);
    74→        }
    75→      }
    76→      const s = median(starts);
    77→      const e = median(ends);
    78→      if (s != null && e != null) out[type] = [s, e];
    79→    }
    80→  }
    81→  consensusCache.set(key, out);
    82→  return out;
    83→}
    84→
    85→/** Look up OP/ED intervals for one episode: exact entry, else per-anime consensus. */
    86→export function lookupSkipTimes(malId: number, ep: number): { intro?: SkipInterval; outro?: SkipInterval } {
    87→  const byEp = data[String(malId)];
    88→  const entry = byEp?.[String(ep)] ?? consensus(malId);
    89→  if (!entry) return {};
    90→  const out: { intro?: SkipInterval; outro?: SkipInterval } = {};
    91→  if (entry.op) out.intro = { start: entry.op[0], end: entry.op[1] };
    92→  if (entry.ed) out.outro = { start: entry.ed[0], end: entry.ed[1] };
    93→  return out;
    94→}
    95→
    96→/**
    97→ * The vendored dataset's EXACT op row for this episode — no consensus
    98→ * fallback. Used to defend known openings against the scout tier's
    99→ * "couldn't measure" verdicts: a fleet nodetect row means the bots failed
   100→ * to cross-match THIS scan (audio quirks, odd encodes), not that the show
   101→ * has no opening. When the community has a vote-backed op for the exact
   102→ * episode (Kimetsu ep1: 41.3–131.3 at 10k votes), it must never be
   103→ * silenced by a failed measurement. Consensus has no such protection —
   104→ * finale episodes genuinely drop their OP, and the anime-wide median
   105→ * would wrongly button them.
   106→ */
   107→export function exactDatasetIntro(malId: number, ep: number): SkipInterval | null {
   108→  const entry = data[String(malId)]?.[String(ep)];
   109→  if (!entry?.op) return null;
   110→  return { start: entry.op[0], end: entry.op[1] };
   111→}
   112→
   113→function valid(iv: unknown): iv is SkipInterval {
   114→  if (!iv || typeof iv !== "object") return false;
   115→  const v = iv as SkipInterval;
   116→  return Number.isFinite(v.start) && Number.isFinite(v.end) && v.end > v.start;
   117→}
   118→
   119→/**
   120→ * Plausibility gate — a "skip intro" that STARTS 10 minutes into a
   121→ * 24-minute episode is not an opening (Zoko mis-tags recap blocks and
   122→ * random chapters as intro/outro on plenty of shows — Grand Blue ep9
   123→ * "intro 619-709"). Since the blind 8-90s guess was removed, a WRONG
   124→ * interval shows a wrong button — worse than no button. Anything that
   125→ * fails the sanity shape is dropped: the Skip Intro button must actually
   126→ * skip the opening, or not exist.
   127→ */
   128→export function plausibleIntro(iv: SkipInterval | undefined | null, lenMin?: number | null): iv is SkipInterval {
   129→  if (!iv || !valid(iv)) return false;
   130→  const cap = lenMin && lenMin > 30 ? 420 : 320; // hour-long formats may open late
   131→  return iv.start <= cap && iv.end - iv.start <= 180;
   132→}
   133→
   134→export function plausibleOutro(iv: SkipInterval | undefined | null): iv is SkipInterval {
   135→  return !!iv && valid(iv) && iv.end - iv.start <= 300;
   136→}
   137→
   138→/**
   139→ * Merge Zoko's native skip payload with the AniSkip dataset.
   140→ * Zoko wins per-field (it is frame-accurate for its own encodes);
   141→ * the dataset fills whatever Zoko left blank. BOTH sources pass the
   142→ * plausibility gate here — this is where early-returning callers
   143→ * (withSkipDataLive's "nothing left to ask") would otherwise leak a
   144→ * mis-tagged interval (Grand Blue ep9 "intro 619-709") into the player.
   145→ */
   146→export function withSkipData<T extends { skip?: SkipData }>(
   147→  malId: number,
   148→  ep: number,
   149→  stream: T,
   150→  opts: { episodeLengthMin?: number | null } = {},
   151→): T & { skip?: SkipData } {
   152→  const zoko = stream.skip ?? {};
   153→  const filled = lookupSkipTimes(malId, ep);
   154→
   155→  const merged: SkipData = {};
   156→  const zi = valid(zoko.intro) && plausibleIntro(zoko.intro, opts.episodeLengthMin) ? zoko.intro : undefined;
   157→  const zo = valid(zoko.outro) && plausibleOutro(zoko.outro) ? zoko.outro : undefined;
   158→  const fi = valid(filled.intro) && plausibleIntro(filled.intro, opts.episodeLengthMin) ? filled.intro : undefined;
   159→  const fo = valid(filled.outro) && plausibleOutro(filled.outro) ? filled.outro : undefined;
   160→  const intro = zi ?? fi;
   161→  const outro = zo ?? fo;
   162→  if (intro) merged.intro = intro;
   163→  if (outro) merged.outro = outro;
   164→  if (intro || outro) {
   165→    merged.source = zi || zo ? "zoko" : "aniskip";
   166→  }
   167→
   168→  return { ...stream, skip: Object.keys(merged).length ? merged : undefined };
   169→}
   170→
   171→/* ------------------------------------------------------------------ *
   172→ * Tier 2.5 — the intro-scout fleet. Bots watch the first minutes of
   173→ * every catalog episode (the SAME streams the player serves) and
   174→ * cross-match loudness envelopes across episodes: the opening is the
   175→ * one part of a show that repeats identically, so where episodes align
   176→ * IS the opening. Because these intervals are measured on OUR encodes,
   177→ * they beat AniSkip's community timings (taken on other encodes, often
   178→ * offset by recaps/stream bumps — the "skip intro is inaccurate" bug).
   179→ * Priority: Zoko native (frame-accurate) > scout > AniSkip > crowd.
   180→ * ---------------------------------------------------------------- */
   181→
   182→const SCOUT_CACHE_TTL_MS = 60_000;
   183→const scoutCache = new Map<number, { rows: Array<{ ep: number; start: number; end: number; nodetect: boolean }>; at: number }>();
   184→
   185→async function scoutRows(malId: number): Promise<Array<{ ep: number; start: number; end: number; nodetect: boolean }>> {
   186→  const hit = scoutCache.get(malId);
   187→  if (hit && Date.now() - hit.at < SCOUT_CACHE_TTL_MS) return hit.rows;
   188→  const rows = await db.introScout.findMany({
   189→    where: { malId },
   190→    orderBy: { updatedAt: "desc" },
   191→    take: 60, // room for nodetect rows too — real measurements are filtered below
   192→    select: { ep: true, start: true, end: true, nodetect: true },
   193→  });
   194→  scoutCache.set(malId, { rows, at: Date.now() });
   195→  return rows;
   196→}
   197→
   198→/**
   199→ * The fleet's verdict for one episode:
   200→ *   interval set    — the opening interval (exact per-episode row, or the
   201→ *                     anime-wide median when 2+ episodes agree)
   202→ *   confirmedNone   — bots WATCHED this exact episode and found NO opening:
   203→ *                     the Skip Intro button must not appear for it (no
   204→ *                     guessing, no AniSkip resurrection)
   205→ *   null            — the fleet has nothing on this anime
   206→ */
   207→export async function scoutIntro(
   208→  malId: number,
   209→  ep: number,
   210→): Promise<{ interval: SkipInterval | null; exact: boolean; confirmedNone?: boolean } | null> {
   211→  try {
   212→    const rows = await scoutRows(malId);
   213→    const exact = rows.find((r) => r.ep === ep);
   214→    if (exact) {
   215→      if (exact.nodetect) return { interval: null, exact: true, confirmedNone: true };
   216→      // degenerate row guard — merge upserts can land as {0,0} (old establish
   217→      // failures came back from prod without their nodetect flag); a zero or
   218→      // inverted interval is a broken row, not a measurement. Ignore it and
   219→      // fall through to the anime-wide median / other tiers.
   220→      if (exact.end > exact.start) return { interval: { start: exact.start, end: exact.end }, exact: true };
   221→    }
   222→    // degenerate {0,0} rows are excluded here too — they must not skew the median
   223→    const measured = rows.filter((r) => !r.nodetect && r.end > r.start);
   224→    if (measured.length >= 2) {
   225→      const med = (xs: number[]) => {
   226→        const s = [...xs].sort((a, b) => a - b);
   227→        return s[Math.floor(s.length / 2)];
   228→      };
   229→      const start = med(measured.map((r) => r.start));
   230→      const end = med(measured.map((r) => r.end));
   231→      if (end > start && end - start <= 150 && start <= 320) {
   232→        return { interval: { start, end }, exact: false };
   233→      }
   234→    }
   235→    return null;
   236→  } catch {
   237→    return null;
   238→  }
   239→}

   241→/* ------------------------------------------------------------------ *
   242→ * Tier 4 — LIVE AniSkip lookups (cached), for episodes the vendored
   243→ * dump + consensus don't cover: brand-new episodes of airing shows and
   244→ * freshly imported anime. Also used to prefetch skip data when the
   245→ * catalog sync imports new episodes (see prefetchSkipTimes below).
   246→ * ------------------------------------------------------------------ */
   247→
   248→const ANISKIP_API = "https://api.aniskip.com/v2/skip-times";
   249→// Submissions trickle in over the days after an episode airs (2026-season
   250→// shows were 404 on air day, populated within the week) — re-ask daily so a
   251→// fresh episode picks up community times as soon as anyone submits them.
   252→const ANISKIP_MISS_TTL_MS = 24 * 60 * 60 * 1000;
   253→const ANISKIP_DISK_DIR =
   254→  process.env.VERCEL === "1" ? join(tmpdir(), "aniskip-cache") : join(process.cwd(), "db", "aniskip-cache");
   255→
   256→type LiveEntry = { op?: [number, number]; ed?: [number, number]; at: number; miss?: boolean };
   257→type LiveTimes = { op?: [number, number]; ed?: [number, number] };
   258→type LiveFetchResult = { times: LiveTimes | null; transient: boolean };
   259→type LiveFile = Record<string, LiveEntry>;
   260→const liveMem = new Map<string, LiveFile>();
   261→const liveInFlight = new Map<string, Promise<LiveFetchResult>>();
   262→
   263→function liveDiskPath(malId: number): string {
   264→  const hash = createHash("sha1").update(String(malId)).digest("hex").slice(0, 16);
   265→  return join(ANISKIP_DISK_DIR, `${hash}.json`);
   266→}
   267→
   268→async function liveRead(malId: number): Promise<LiveFile> {
   269→  const key = String(malId);
   270→  const mem = liveMem.get(key);
   271→  if (mem) return mem;
   272→  try {
   273→    const file = JSON.parse(await readFile(liveDiskPath(malId), "utf8")) as LiveFile;
   274→    liveMem.set(key, file);
   275→    return file;
   276→  } catch {
   277→    return {};
   278→  }
   279→}
   280→
   281→async function liveWrite(malId: number, file: LiveFile): Promise<void> {
   282→  liveMem.set(String(malId), file);
   283→  try {
   284→    await mkdir(ANISKIP_DISK_DIR, { recursive: true });
   285→    await writeFile(liveDiskPath(malId), JSON.stringify(file), "utf8");
   286→  } catch {
   287→    /* read-only FS (serverless) — the in-memory copy still helps this instance */
   288→  }
   289→}
   290→
   291→/**
   292→ * Ask api.aniskip.com for one episode's OP/ED times. `lenMin` is the
   293→ * episode's length in MINUTES — the API requires an `episodeLength` param
   294→ * (400s without it); 24 is the TV standard when we don't know better.
   295→ * Returns the times plus whether the failure was TRANSIENT (5xx / rate
   296→ * limit / timeout) — transient failures must NOT be negative-cached.
   297→ */
   298→async function fetchAniSkipLive(malId: number, ep: number, lenMin?: number): Promise<LiveFetchResult> {
   299→  const len = lenMin && Number.isFinite(lenMin) && lenMin > 0 ? Math.round(lenMin) : 24;
   300→  try {
   301→    const res = await fetch(`${ANISKIP_API}/${malId}/${ep}?types=op&types=ed&episodeLength=${len}`, {
   302→      headers: { Accept: "application/json" },
   303→      cache: "no-store",
   304→      signal: AbortSignal.timeout(4_500),
   305→    });
   306→    if (res.status === 404) return { times: null, transient: false }; // genuine "not submitted yet"
   307→    if (!res.ok) return { times: null, transient: true }; // 429/5xx — retry another time
   308→    const body = (await res.json()) as {
   309→      found?: boolean;
   310→      results?: Array<{ interval?: { startTime?: number; endTime?: number }; skipType?: string }>;
   311→    };
   312→    if (!body?.found || !Array.isArray(body.results)) return { times: null, transient: false };
   313→    const out: LiveTimes = {};
   314→    for (const r of body.results) {
   315→      const s = r?.interval?.startTime;
   316→      const e = r?.interval?.endTime;
   317→      if (!Number.isFinite(s) || !Number.isFinite(e) || e! <= s!) continue;
   318→      if (r.skipType === "op") out.op = [s!, e!];
   319→      else if (r.skipType === "ed") out.ed = [s!, e!];
   320→    }
   321→    return { times: out.op || out.ed ? out : null, transient: false };
   322→  } catch {
   323→    return { times: null, transient: true }; // timeout / offline — retryable
   324→  }
   325→}
   326→
   327→/**
   328→ * Cached live lookup for one episode. Positive entries are kept forever
   329→ * (community timings don't rot); misses are retried after 7 days.
   330→ */
   331→async function liveLookup(malId: number, ep: number, lenMin?: number): Promise<LiveTimes | null> {
   332→  const epKey = String(ep);
   333→  const file = await liveRead(malId);
   334→  const hit = file[epKey];
   335→  if (hit && Date.now() - hit.at < (hit.miss ? ANISKIP_MISS_TTL_MS : Number.MAX_SAFE_INTEGER)) {
   336→    return hit.miss ? null : { op: hit.op, ed: hit.ed };
   337→  }
   338→  const inflightKey = `${malId}/${ep}`;
   339→  let p: Promise<LiveFetchResult> | undefined = liveInFlight.get(inflightKey);
   340→  if (!p) {
   341→    p = fetchAniSkipLive(malId, ep, lenMin)
   342→      .then((r) => {
   343→        liveInFlight.delete(inflightKey);
   344→        return r;
   345→      })
   346→      .catch(() => {
   347→        liveInFlight.delete(inflightKey);
   348→        return { times: null, transient: true } as LiveFetchResult;
   349→      });
   350→    liveInFlight.set(inflightKey, p);
   351→  }
   352→  const wrapped = await p;
   353→  const data = wrapped.times;
   354→  const transient = wrapped.transient;
   355→  if (!transient) {
   356→    const fresh = await liveRead(malId); // another request may have written meanwhile
   357→    fresh[epKey] = data ? { ...data, at: Date.now() } : { at: Date.now(), miss: true };
   358→    await liveWrite(malId, fresh);
   359→  }
   360→  return data;
   361→}

   363→/**
   364→ * Merge like `withSkipData`, plus the live AniSkip tier: anything the
   365→ * vendored data left blank gets one (cached) live query. This is what makes
   366→ * EVERY episode — including ones imported after the dump was built — carry
   367→ * its skip intro/ending.
   368→ */
   369→export async function withSkipDataLive<T extends { skip?: SkipData }>(
   370→  malId: number,
   371→  ep: number,
   372→  stream: T,
   373→  opts: { episodeLengthMin?: number | null } = {},
   374→): Promise<T & { skip?: SkipData }> {
   375→  const merged = withSkipData(malId, ep, stream, opts);
   376→  let skip: SkipData = { ...(merged.skip ?? {}) };
   377→
   378→  // scout tier — measured on OUR encodes by the fleet. An exact per-episode
   379→  // measurement outranks everything (including Zoko's payload, which marks
   380→  // the repeated cold-open as "intro start" on many shows); the anime-wide
   381→  // median only fills gaps (Zoko/frame data wins there).
   382→  // A CONFIRMED-NONE verdict (the fleet watched this episode against an
   383→  // established reference and found no opening) wins too — EXCEPT when this
   384→  // exact episode already has concrete intro data: Zoko's native payload or
   385→  // the dataset's exact op row. A nodetect is "our scan couldn't align",
   386→  // which must never silence a vote-backed community interval for the same
   387→  // episode (Kimetsu ep1: 10k votes vs one failed audio match). Consensus /
   388→  // live / crowd intros are NOT protected — finale episodes genuinely drop
   389→  // their OP and the anime-wide median must not button them.
   390→  const scout = await scoutIntro(malId, ep);
   391→  const confirmedNone = !!scout?.confirmedNone;
   392→  const exactKnown = valid(stream.skip?.intro) || !!exactDatasetIntro(malId, ep);
   393→  const suppressIntro = confirmedNone && !exactKnown;
   394→  if (suppressIntro) {
   395→    delete skip.intro;
   396→    if (!skip.outro) delete skip.source;
   397→  } else if (scout?.interval && (scout.exact || !valid(stream.skip?.intro))) {
   398→    skip.intro = scout.interval;
   399→    skip.source = "scout";
   400→  }
   401→  if (skip.intro && skip.outro) return { ...merged, skip }; // nothing left to ask
   402→
   403→  try {
   404→    const live = await liveLookup(malId, ep, opts.episodeLengthMin ?? undefined);
   405→    if (live) {
   406→      if (!skip.intro && live.op && !suppressIntro) {
   407→        skip.intro = { start: live.op[0], end: live.op[1] };
   408→        skip.source = "aniskip-live";
   409→      }
   410→      if (!skip.outro && live.ed) skip.outro = { start: live.ed[0], end: live.ed[1] };
   411→      if (skip.intro || skip.outro) {
   412→        if (!skip.source) skip.source = "aniskip-live";
   413→        return { ...merged, skip };
   414→      }
   415→    }
   416→  } catch {
   417→    /* live tier is best-effort — the crowd/heuristic tiers still cover episodes */
   418→  }
   419→  // crowd tier — for shows nobody has submitted anywhere yet, viewers who
   420→  // scrubbed past the opening have already taught us where it ends
   421→  // (never applied when the fleet confirmed there IS no opening)
   422→  if (!suppressIntro) {
   423→    try {
   424→      const crowd = await crowdIntroTarget(malId, ep);
   425→      if (crowd != null) {
   426→        if (!skip.intro) {
   427→          skip.intro = { start: 8, end: crowd };
   428→          skip.source = "crowd";
   429→        }
   430→        if (skip.intro || skip.outro) return { ...merged, skip };
   431→      }
   432→    } catch {
   433→      /* votes are best-effort too */
   434→    }
   435→  }
   436→  // final plausibility gate — the button must actually skip the opening:
   437→  // drop any interval that fails the sanity shape (wrong data < no button)
   438→  if (skip.intro && !plausibleIntro(skip.intro, opts.episodeLengthMin)) {
   439→    delete skip.intro;
   440→    if (!skip.outro) delete skip.source;
   441→  }
   442→  if (skip.outro && !plausibleOutro(skip.outro)) {
   443→    delete skip.outro;
   444→    if (!skip.intro) delete skip.source;
   445→  }
   446→  return { ...merged, skip };
   447→}

   449→/**
   450→ * Coverage check used by the skip audit (bot): where would this episode's
   451→ * Skip Intro data come from? Checks the vendored dataset (exact row, then
   452→ * per-anime consensus), then the live tier's cache — `allowLiveFetch` also
   453→ * permits one cached network query to AniSkip (used when probing gaps), and
   454→ * finally the crowd-learned scrub median — the audit counts anything the
   455→ * player would actually show. `confirmedNone` = the fleet verified there is
   456→ * NO opening on this episode (the button is suppressed by design — not a
   457→ * coverage gap).
   458→ */
   459→export interface SkipCoverage {
   460→  intro?: SkipInterval;
   461→  outro?: SkipInterval;
   462→  source: "scout" | "exact" | "consensus" | "live" | "crowd";
   463→  confirmedNone?: boolean;
   464→}
   465→
   466→export async function skipCoverage(
   467→  malId: number,
   468→  ep: number,
   469→  opts: { allowLiveFetch?: boolean; episodeLengthMin?: number | null } = {},
   470→): Promise<SkipCoverage | null> {
   471→  // scout tier first — what the player would actually use (see the priority
   472→  // in withSkipDataLive). A confirmed-none yields to an exact dataset op row
   473→  // for THIS episode (same precedence as the player merge).
   474→  const scout = await scoutIntro(malId, ep);
   475→  if (scout?.confirmedNone) {
   476→    const ex = exactDatasetIntro(malId, ep);
   477→    if (ex && plausibleIntro(ex, opts.episodeLengthMin)) return { source: "exact", intro: ex };
   478→    return { source: "scout", confirmedNone: true };
   479→  }
   480→  if (scout?.interval) return { source: "scout", intro: scout.interval };
   481→  const byEp = data[String(malId)];
   482→  const exact = byEp?.[String(ep)];
   483→  const exOp = exact?.op ? { start: exact.op[0], end: exact.op[1] } : null;
   484→  const exEd = exact?.ed ? { start: exact.ed[0], end: exact.ed[1] } : null;
   485→  if (exOp || exEd) {
   486→    const cov: SkipCoverage = { source: "exact" };
   487→    if (plausibleIntro(exOp, opts.episodeLengthMin)) cov.intro = exOp;
   488→    if (plausibleOutro(exEd)) cov.outro = exEd;
   489→    if (cov.intro || cov.outro) return cov;
   490→  }
   491→  const cons = consensus(malId);
   492→  const coOp = cons.op ? { start: cons.op[0], end: cons.op[1] } : null;
   493→  const coEd = cons.ed ? { start: cons.ed[0], end: cons.ed[1] } : null;
   494→  if (coOp || coEd) {
   495→    const cov: SkipCoverage = { source: "consensus" };
   496→    if (plausibleIntro(coOp, opts.episodeLengthMin)) cov.intro = coOp;
   497→    if (plausibleOutro(coEd)) cov.outro = coEd;
   498→    if (cov.intro || cov.outro) return cov;
   499→  }
   500→  const liveCov = (op?: [number, number], ed?: [number, number]): SkipCoverage | null => {
   501→    const i = op ? { start: op[0], end: op[1] } : null;
   502→    const o = ed ? { start: ed[0], end: ed[1] } : null;
   503→    if (!plausibleIntro(i, opts.episodeLengthMin) && !plausibleOutro(o)) return null;
   504→    const cov: SkipCoverage = { source: "live" };
   505→    if (plausibleIntro(i, opts.episodeLengthMin)) cov.intro = i;
   506→    if (plausibleOutro(o)) cov.outro = o;
   507→    return cov;
   508→  };
   509→  if (opts.allowLiveFetch) {
   510→    const live = await liveLookup(malId, ep, opts.episodeLengthMin ?? undefined);
   511→    const cov = live ? liveCov(live.op, live.ed) : null;
   512→    if (cov) return cov;
   513→  } else {
   514→    const cached = (await liveRead(malId))[String(ep)];
   515→    const cov = cached && !cached.miss ? liveCov(cached.op, cached.ed) : null;
   516→    if (cov) return cov;
   517→  }
   518→  // crowd tier — viewers' scrub medians count as coverage too (the player
   519→  // shows a working Skip Intro button for them)
   520→  const crowd = await crowdIntroTarget(malId, ep);
   521→  if (crowd != null) return { source: "crowd", intro: { start: 8, end: crowd } };
   522→  return null;
   523→}

   525→/**
   526→ * Warm the live cache ahead of time — called after the catalog sync imports
   527→ * new episodes so their skip data is already on disk before anyone presses
   528→ * play. Sequential with ~1.1s spacing (AniSkip's documented rate limit is
   529→ * 1 req/s); bounded so a huge import can't run away.
   530→ */
   531→export async function prefetchSkipTimes(
   532→  pairs: Array<{ malId: number; ep: number; lenMin?: number | null }>,
   533→  budget = 16,
   534→): Promise<number> {
   535→  let fetched = 0;
   536→  for (const { malId, ep, lenMin } of pairs.slice(0, budget)) {
   537→    try {
   538→      const file = await liveRead(malId);
   539→      const hit = file[String(ep)];
   540→      if (hit && (hit.op || hit.ed || (hit.miss && Date.now() - hit.at < ANISKIP_MISS_TTL_MS))) continue;
   541→      const { times, transient } = await fetchAniSkipLive(malId, ep, lenMin ?? undefined);
   542→      if (!transient) {
   543→        const fresh = await liveRead(malId);
   544→        fresh[String(ep)] = times ? { ...times, at: Date.now() } : { at: Date.now(), miss: true };
   545→        await liveWrite(malId, fresh);
   546→      }
   547→      if (times) fetched++;
   548→    } catch {
   549→      /* keep going */
   550→    }
   551→    await new Promise((r) => setTimeout(r, 1_100)); // respect 1 req/s
   552→  }
   553→  return fetched;
   554→}

   556→/* ------------------------------------------------------------------ *
   557→ * Tier 5 — crowd-learned openings. Brand-new seasons often have NO
   558→ * skip data anywhere (AniSkip 404s for months on niche shows), so the
   559→ * fixed 8-90s guess lands in the middle of their long cold-opens.
   560→ * Instead, viewers who manually scrub forward past the opening get
   561→ * recorded (see POST /api/skip-vote) and the median scrub target
   562→ * becomes this anime's Skip Intro end for everyone. Self-healing:
   563→ * the more people watch, the sharper the button gets.
   564→ * ---------------------------------------------------------------- */
   565→
   566→const CROWD_MIN_VOTES_EP = 2; // per-episode median needs at least this many
   567→const CROWD_MIN_VOTES_ANIME = 4; // anime-wide fallback needs more agreement
   568→const CROWD_VOTE_MAX_AGE_MS = 60 * 24 * 60 * 60 * 1000; // tastes/encodes drift
   569→const CROWD_KEEP_PER_ANIME = 200; // storage cap per anime
   570→
   571→export const SKIP_VOTE_TARGET_MIN = 30;
   572→export const SKIP_VOTE_TARGET_MAX = 420;
   573→
   574→/**
   575→ * Record one implied OP-end vote. Best-effort: validates the catalog entry
   576→ * and the target range, caps rows per anime, never throws.
   577→ */
   578→export async function recordSkipVote(malId: number, ep: number, target: number): Promise<boolean> {
   579→  try {
   580→    if (!Number.isFinite(malId) || malId <= 0 || !Number.isFinite(ep) || ep < 1 || ep > 2000) return false;
   581→    if (!Number.isFinite(target)) return false;
   582→    const t = Math.round(target);
   583→    if (t < SKIP_VOTE_TARGET_MIN || t > SKIP_VOTE_TARGET_MAX) return false;
   584→    const anime = await db.anime.findUnique({ where: { malId }, select: { malId: true } });
   585→    if (!anime) return false;
   586→    await db.skipVote.create({ data: { malId, ep, target: t } });
   587→    // prune: keep the newest KEEP votes for this anime (votes are cheap to drop)
   588→    const count = await db.skipVote.count({ where: { malId } });
   589→    if (count > CROWD_KEEP_PER_ANIME) {
   590→      const old = await db.skipVote.findMany({
   591→        where: { malId },
   592→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   593→        take: count - CROWD_KEEP_PER_ANIME,
   594→        select: { id: true },
   595→      });
   596→      if (old.length) await db.skipVote.deleteMany({ where: { id: { in: old.map((r) => r.id) } } });
   597→    }
   598→    return true;
   599→  } catch {
   600→    return false;
   601→  }
   602→}
   603→
   604→function medianNum(values: number[]): number | null {
   605→  if (!values.length) return null;
   606→  const s = [...values].sort((a, b) => a - b);
   607→  return s[Math.floor(s.length / 2)];
   608→}
   609→
   610→/**
   611→ * The crowd's OP-end for one episode: median of this episode's recent votes,
   612→ * else (with more agreement) the anime-wide median — openings are the same
   613→ * video across episodes. Returns whole seconds or null when thin/old.
   614→ */
   615→export async function crowdIntroTarget(malId: number, ep: number): Promise<number | null> {
   616→  try {
   617→    const from = new Date(Date.now() - CROWD_VOTE_MAX_AGE_MS);
   618→    const rows = await db.skipVote.findMany({
   619→      where: { malId, createdAt: { gte: from } },
   620→      select: { ep: true, target: true },
   621→      orderBy: { createdAt: "desc" },
   622→      take: 400,
   623→    });
   624→    const epTargets = rows.filter((r) => r.ep === ep).map((r) => r.target);
   625→    if (epTargets.length >= CROWD_MIN_VOTES_EP) return medianNum(epTargets);
   626→    const all = rows.map((r) => r.target);
   627→    if (all.length >= CROWD_MIN_VOTES_ANIME) return medianNum(all);
   628→    return null;
   629→  } catch {
   630→    return null;
   631→  }
   632→}
