    19→    19→function readJson<T>(key: string, fallback: T): T {
    20→    20→  if (typeof window === "undefined") return fallback;
    21→    21→  try {
    22→    22→    const raw = window.localStorage.getItem(key);
    23→    23→    return raw ? (JSON.parse(raw) as T) : fallback;
    24→    24→  } catch {
    25→    25→    return fallback;
    26→    26→  }
    27→    27→}
    28→    28→
    29→    29→function writeJson(key: string, value: unknown) {
    30→    30→  try {
    31→    31→    window.localStorage.setItem(key, JSON.stringify(value));
    32→    32→  } catch {
    33→    33→    /* storage full/blocked — ignore */
    34→    34→  }
    35→    35→}
    36→    36→
    37→    37→/* ---------------- offline outbox (sync when back online) ----------------
    38→    38→ * Every server write the app makes while offline (progress saves, watch
    39→    39→ * time, removal deletes, my-list hydrate) is parked in localStorage under a
    40→    40→ * dedupe key and replayed the moment the connection is back — either by the
    41→    41→ * browser's `online` event or the next time the app opens. Same-key items
    42→    42→ * replace each other (latest position wins), so a whole offline session of
    43→    43→ * heartbeats collapses into one fresh write per episode.
    44→    44→ */
    45→    45→
    46→    46→const OUTBOX_KEY = "animetest123.outbox";
    47→    47→const OUTBOX_MAX = 120;
    48→    48→
    49→    49→interface OutboxItem {
    50→    50→  k: string; // dedupe key
    51→    51→  url: string;
    52→    52→  method: "POST" | "DELETE";
    53→    53→  body?: string;
    54→    54→  at: number;
    55→    55→}
    56→    56→
    57→    57→function readOutbox(): OutboxItem[] {
    58→    58→  return readJson<OutboxItem[]>(OUTBOX_KEY, []);
    59→    59→}
    60→    60→
    61→    61→function writeOutbox(items: OutboxItem[]) {
    62→    62→  writeJson(OUTBOX_KEY, items.slice(-OUTBOX_MAX));
    63→    63→}
    64→    64→
    65→    65→function queueOutbox(key: string, url: string, method: "POST" | "DELETE", body?: string) {
    66→    66→  try {
    67→    67→    const items = readOutbox().filter((i) => i.k !== key);
    68→    68→    items.push({ k: key, url, method, body, at: Date.now() });
    69→    69→    writeOutbox(items);
    70→    70→  } catch {
    71→    71→    /* ignore */
    72→    72→  }
    73→    73→}
    74→    74→
    75→    75→function dequeueOutbox(key: string) {
    76→    76→  try {
    77→    77→    const items = readOutbox();
    78→    78→    if (!items.some((i) => i.k === key)) return;
    79→    79→    writeOutbox(items.filter((i) => i.k !== key));
    80→    80→  } catch {
    81→    81→    /* ignore */
    82→    82→  }
    83→    83→}
    84→    84→
    85→    85→/**
    86→    86→ * Send a write request, or park it in the outbox when the network is down.
    87→    87→ * Returns the Response, or null when the item was queued (or the request
    88→    88→ * failed outright and is now queued). `queueFirst` parks the item BEFORE
    89→    89→ * attempting — for must-eventually-land writes (progress deletes) that
    90→    90→ * survive the page dying mid-request.
    91→    91→ */
    92→    92→export function outboxSend(
    93→    93→  key: string,
    94→    94→  url: string,
    95→    95→  method: "POST" | "DELETE",
    96→    96→  body?: string,
    97→    97→  opts?: { queueFirst?: boolean },
    98→    98→): Promise<Response | null> {
    99→    99→  if (opts?.queueFirst) queueOutbox(key, url, method, body);
   100→   100→  if (typeof navigator !== "undefined" && navigator.onLine === false) {
   101→   101→    if (!opts?.queueFirst) queueOutbox(key, url, method, body);
   102→   102→    return Promise.resolve(null);
   103→   103→  }
   104→   104→  return fetch(url, {
   105→   105→    method,
   106→   106→    headers: body ? { "Content-Type": "application/json" } : undefined,
   107→   107→    body,
   108→   108→    ...(opts?.queueFirst ? { keepalive: true } : {}),
   109→   109→  })
   110→   110→    .then((res) => {
   111→   111→      if (res.ok && opts?.queueFirst) dequeueOutbox(key);
   112→   112→      return res;
   113→   113→    })
   114→   114→    .catch(() => {
   115→   115→      if (!opts?.queueFirst) queueOutbox(key, url, method, body);
   116→   116→      return null;
   117→   117→    });
   118→   118→}
   119→   119→
   120→   120→let flushingOutbox = false;
   121→   121→
   122→   122→/**
   123→   123→ * Replay the outbox in order. Stops at the first failure (still offline /
   124→   124→ * server refusing) so later items don't jump the queue. Dispatches
   125→   125→ * `clyle-synced` (detail = number of synced writes) when anything landed.
   126→   126→ */
   127→   127→export async function flushOutbox(): Promise<number> {
   128→   128→  if (flushingOutbox) return 0;
   129→   129→  if (typeof navigator !== "undefined" && navigator.onLine === false) return 0;
   130→   130→  flushingOutbox = true;
   131→   131→  let synced = 0;
   132→   132→  try {
   133→   133→    let items = readOutbox();
   134→   134→    while (items.length) {
   135→   135→      const item = items[0];
   136→   136→      try {
   137→   137→        const res = await fetch(item.url, {
   138→   138→          method: item.method,
   139→   139→          headers: item.body ? { "Content-Type": "application/json" } : undefined,
   140→   140→          body: item.body,
   141→   141→        });
   142→   142→        if (!res.ok) break; // server refused — retry on a later flush
   143→   143→        items = items.slice(1);
   144→   144→        writeOutbox(items);
   145→   145→        synced++;
   146→   146→      } catch {
   147→   147→        break; // still offline — stop
   148→   148→      }
   149→   149→    }
   150→   150→  } finally {
   151→   151→    flushingOutbox = false;
   152→   152→  }
   153→   153→  if (synced > 0) {
   154→   154→    try {
   155→   155→      window.dispatchEvent(new CustomEvent("clyle-synced", { detail: synced }));
   156→   156→      window.dispatchEvent(new CustomEvent("progress-changed"));
   157→   157→    } catch {
   158→   158→      /* ignore */
   159→   159→    }
   160→   160→  }
   161→   161→  return synced;
   162→   162→}
   163→   163→
   164→   164→/** How many writes are waiting to be synced (debug/settings display). */
   165→   165→export function outboxPendingCount(): number {
   166→   166→  return readOutbox().length;
   167→   167→}

   301→   301→export interface SaveProgressArgs {
   302→   302→  malId: number;
   303→   303→  episode: number;
   304→   304→  lang: "sub" | "dub";
   305→   305→  server: string;
   306→   306→  position: number;
   307→   307→  duration: number;
   308→   308→  completed?: boolean;
   309→   309→  title?: string;
   310→   310→  image?: string | null;
   311→   311→  total?: number;

   314→   314→/**
   315→   315→ * Persist progress locally immediately; mirror to the server. When
   316→   316→ * `keepalive` is set (page unload / tab hide) the request is sent with
   317→   317→ * keepalive so it survives navigation.
   318→   318→ */
   319→   319→export function saveProgressDetailed(args: SaveProgressArgs, opts?: { keepalive?: boolean }) {
   320→   320→  if (!Number.isFinite(args.position) || args.position < 5) return;
   321→   321→  // removed from Continue Watching — stale saves (heartbeats, beacons, the
   322→   322→  // unmount flush of a just-left player) must not resurrect it. The only
   323→   323→  // way back in is a deliberate watch (clearProgressTombstone on mount).
   324→   324→  if (isProgressRemoved(args.malId)) return;
   325→   325→  const completed = args.completed ?? (args.duration > 0 && args.position >= args.duration * 0.92);
   326→   326→  const entry: DetailedProgress = {
   327→   327→    pos: Math.floor(args.position),
   328→   328→    dur: Math.floor(args.duration || 0),
   329→   329→    completed,
   330→   330→    lang: args.lang,
   331→   331→    server: args.server,
   332→   332→    title: args.title,
   333→   333→    image: args.image,
   334→   334→    total: args.total,
   335→   335→    updatedAt: Date.now(),
   336→   336→  };
   337→   337→
   338→   338→  // local (instant — bars/resume render without a round trip)
   339→   339→  const map = readEpMap();
   340→   340→  const key = String(args.malId);
   341→   341→  map[key] = map[key] ?? {};
   342→   342→  map[key][String(args.episode)] = entry;
   343→   343→  writeJson(EP_PROGRESS_KEY, map);
   344→   344→
   345→   345→  // legacy "last watched episode" map (details badge, resume fallback)
   346→   346→  try {
   347→   347→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   348→   348→    legacy[key] = { ep: args.episode, lang: args.lang, pos: entry.pos, updatedAt: entry.updatedAt };
   349→   349→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   350→   350→  } catch {
   351→   351→    /* ignore */
   352→   352→  }
   353→   353→
   354→   354→  try {
   355→   355→    window.dispatchEvent(new CustomEvent("progress-changed"));
   356→   356→  } catch {
   357→   357→    /* ignore */
   358→   358→  }
   359→   359→
   360→   360→  // server mirror (best effort, never blocks playback)
   361→   361→  const userId = getUserId();
   362→   362→  const body = JSON.stringify({
   363→   363→    userId,
   364→   364→    malId: args.malId,
   365→   365→    episode: args.episode,
   366→   366→    lang: args.lang,
   367→   367→    server: args.server,
   368→   368→    position: entry.pos,
   369→   369→    duration: entry.dur,
   370→   370→    completed,
   371→   371→    title: args.title,
   372→   372→    image: args.image,
   373→   373→  });
   374→   374→  try {
   375→   375→    if (opts?.keepalive && "sendBeacon" in navigator) {
   376→   376→      navigator.sendBeacon("/api/progress", new Blob([body], { type: "application/json" }));
   377→   377→    } else {
   378→   378→      // offline-safe: a failed save parks in the outbox and lands when the
   379→   379→      // connection returns (same dedupe key — the latest position wins)
   380→   380→      void outboxSend(`prog:${args.malId}:${args.episode}:${args.lang}`, "/api/progress", "POST", body);
   381→   381→    }
   382→   382→  } catch {
   383→   383→    /* ignore */
   384→   384→  }
   385→   385→}

   387→   387→/** Forget everything watched for one anime (Continue Watching "X" button). */
   388→   388→export function removeProgress(malId: number) {
   389→   389→  // tombstone FIRST — an in-flight save must not re-add this anime between
   390→   390→  // here and the server DELETE ("deleted it but it came back" bug)
   391→   391→  markProgressRemoved(malId);
   392→   392→  const map = readEpMap();
   393→   393→  delete map[String(malId)];
   394→   394→  writeJson(EP_PROGRESS_KEY, map);
   395→   395→  try {
   396→   396→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   397→   397→    delete legacy[String(malId)];
   398→   398→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   399→   399→  } catch {
   400→   400→    /* ignore */
   401→   401→  }
   402→   402→  try {
   403→   403→    window.dispatchEvent(new CustomEvent("progress-changed"));
   404→   404→  } catch {
   405→   405→    /* ignore */
   406→   406→  }
   407→   407→  // queueFirst: the DELETE is parked BEFORE it is attempted, so a lost
   408→   408→  // request (subway dead zone, app killed mid-flight) is replayed by the
   409→   409→  // next flush instead of leaving the server row alive — the old retry
   410→   410→  // loop could never cover "app died before the request fired".
   411→   411→  const userId = getUserId();
   412→   412→  void outboxSend(
   413→   413→    `progd:${malId}`,
   414→   414→    `/api/progress?userId=${encodeURIComponent(userId)}&malId=${malId}`,
   415→   415→    "DELETE",
   416→   416→    undefined,
   417→   417→    { queueFirst: true },
   418→   418→  );
   419→   419→}
   420→   420→
   421→   421→/**
   422→   422→ * Reconciliation for the "deleted it but it came back" bug: re-fire the
   423→   423→ * server DELETE for every anime this device has tombstoned in the last 48h.
   424→   424→ * A removal whose DELETE request was lost (offline moment, app killed) gets
   425→   425→ * healed the next time the Home rail loads; the local tombstone keeps the
   426→   426→ * card hidden in the meantime. Idempotent — the server DELETE just deletes
   427→   427→ * (and re-stamps) again.
   428→   428→ */
   429→   429→export function sweepRemovedProgress(): void {
   430→   430→  const map = readRemovedMap();
   431→   431→  const cutoff = Date.now() - 48 * 3600_000;
   432→   432→  const fresh = Object.entries(map).filter(([, at]) => at >= cutoff);
   433→   433→  if (!fresh.length) return;
   434→   434→  const userId = getUserId();
   435→   435→  for (const [malId, at] of fresh) {
   436→   436→    const id = Number(malId);
   437→   437→    if (!Number.isFinite(id)) continue;
   438→   438→    // pace slightly: these are keepalive DELETEs, tiny and idempotent
   439→   439→    window.setTimeout(() => {
   440→   440→      try {
   441→   441→        fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${id}`, {
   442→   442→          method: "DELETE",
   443→   443→          keepalive: true,
   444→   444→        }).catch(() => undefined);
   445→   445→      } catch {
   446→   446→        /* best-effort */
   447→   447→      }
   448→   448→      void at;
   449→   449→    }, 150 + Math.random() * 400);
   450→   450→  }
   451→   451→}

   453→   453→/* ---------------- removal tombstones (local half) ----------------
   454→   454→ * The server rejects saves for removed anime (RemovedProgress table), but
   455→   455→ * the local map must obey too: a save that raced the delete would otherwise
   456→   456→ * re-add bars/Continue entries the moment the server list is merged. */
   457→   457→
   458→   458→const REMOVED_KEY = "animetest123.progress-removed";
   459→   459→
   460→   460→function readRemovedMap(): Record<string, number> {
   461→   461→  return readJson<Record<string, number>>(REMOVED_KEY, {});

   464→   464→function markProgressRemoved(malId: number) {
   465→   465→  const map = readRemovedMap();

   468→   468→}

   470→   470→function isProgressRemoved(malId: number): boolean {

   473→   473→
   474→   474→/**
   475→   475→ * Deliberate playback (WatchView mounted for this anime) — lift the removal
   476→   476→ * tombstone locally + on the server so progress tracking resumes.
   477→   477→ */
   478→   478→export function clearProgressTombstone(malId: number) {
   479→   479→  if (!isProgressRemoved(malId)) return;
   480→   480→  const map = readRemovedMap();
   481→   481→  delete map[String(malId)];

   483→   483→  try {
   484→   484→    void outboxSend(`tomb:${malId}`, "/api/progress", "POST", JSON.stringify({ userId: getUserId(), malId, resume: true }));
   485→   485→  } catch {
   486→   486→    /* ignore */

   489→   489→
   490→   490→/**
   491→   491→ * Server-side progress for one anime — used as a fallback so resume points
   492→   492→ * and red bars survive cleared storage / a fresh browser.
   493→   493→ */
   494→   494→export async function fetchServerProgress(malId: number): Promise<Record<string, DetailedProgress>> {
   495→   495→  try {
   496→   496→    const res = await fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`);
   497→   497→    if (!res.ok) return {};
   498→   498→    const data = (await res.json()) as {
   499→   499→      episodes: Array<{ episode: number; position: number; duration: number; completed: boolean; lang: string; server: string }>;
   500→   500→    };
   501→   501→    const out: Record<string, DetailedProgress> = {};
   502→   502→    for (const e of data.episodes ?? []) {
   503→   503→      out[String(e.episode)] = {
   504→   504→        pos: e.position,
   505→   505→        dur: e.duration,
   506→   506→        completed: e.completed,
   507→   507→        lang: e.lang === "dub" ? "dub" : "sub",
   508→   508→        server: e.server,
   509→   509→        updatedAt: Date.now(),
   510→   510→      };
   511→   511→    }
   512→   512→    // merge into the local map (server rows are authoritative for these keys)
   513→   513→    if (Object.keys(out).length) {
   514→   514→      const map = readEpMap();
   515→   515→      const key = String(malId);
   516→   516→      map[key] = { ...(map[key] ?? {}), ...out };
   517→   517→      writeJson(EP_PROGRESS_KEY, map);
   518→   518→    }
   519→   519→    return out;
   520→   520→  } catch {
   521→   521→    return {};
   522→   522→  }
   523→   523→}

   525→   525→/* ---------------- Continue Watching ---------------- */
   526→   526→
   527→   527→/**
   528→   528→ * Freshest in-progress episode per anime — server list merged with the
   529→   529→ * instant local map (so the rail renders with zero latency and still shows
   530→   530→ * entries when the server DB is cold).
   531→   531→ */
   532→   532→export function mergeContinueItems(serverItems: ContinueItem[]): ContinueItem[] {
   533→   533→  let localAll: Record<string, Record<string, DetailedProgress>> = {};
   534→   534→  try {
   535→   535→    localAll = JSON.parse(window.localStorage.getItem(EP_PROGRESS_KEY) ?? "{}");
   536→   536→  } catch {
   537→   537→    localAll = {};
   538→   538→  }
   539→   539→  // the local removal tombstone is the user's expressed intent: a server row
   540→   540→  // that slipped through the delete race (lost DELETE request, beacon that
   541→   541→  // landed a heartbeat ahead of it on another lambda) must not resurrect the
   542→   542→  // card in the UI. Deliberate playback lifts the tombstone (both halves).
   543→   543→  const removed = readRemovedMap();
   544→   544→
   545→   545→  type Row = { item: ContinueItem; at: number };
   546→   546→  const byAnime = new Map<number, Row>();
   547→   547→
   548→   548→  for (const it of serverItems) {
   549→   549→    if (it.completed) continue;
   550→   550→    if (removed[String(it.malId)]) continue; // deleted here — stays deleted
   551→   551→    byAnime.set(it.malId, { item: it, at: Date.parse(it.updatedAt) });
   552→   552→  }
   553→   553→
   554→   554→  for (const [malId, eps] of Object.entries(localAll)) {
   555→   555→    // removed from Continue Watching — a save that raced the delete must not
   556→   556→    // re-add the entry here either
   557→   557→    if (readRemovedMap()[malId]) continue;
   558→   558→    for (const [ep, e] of Object.entries(eps)) {
   559→   559→      if (e.completed || (e.pos || 0) < 5) continue;
   560→   560→      const malIdNum = Number(malId);
   561→   561→      const prev = byAnime.get(malIdNum);
   562→   562→      if (prev && prev.at >= e.updatedAt) continue;
   563→   563→      const percent = e.dur > 0 ? Math.min(100, Math.round((e.pos / e.dur) * 100)) : 0;
   564→   564→      byAnime.set(malIdNum, {
   565→   565→        at: e.updatedAt,
   566→   566→        item: {
   567→   567→          malId: malIdNum,
   568→   568→          title: e.title ?? prev?.item.title ?? "Unknown title",
   569→   569→          image: e.image ?? prev?.item.image ?? null,
   570→   570→          banner: prev?.item.banner ?? null,
   571→   571→          totalEpisodes: e.total ?? prev?.item.totalEpisodes ?? 0,
   572→   572→          type: prev?.item.type ?? null,
   573→   573→          episode: Number(ep),
   574→   574→          lang: e.lang,
   575→   575→          server: e.server,
   576→   576→          position: e.pos,
   577→   577→          duration: e.dur,
   578→   578→          percent,
   579→   579→          completed: false,
   580→   580→          updatedAt: new Date(e.updatedAt).toISOString(),
   581→   581→        },
   582→   582→      });
   583→   583→    }
   584→   584→  }
   585→   585→
   586→   586→  return [...byAnime.values()].sort((a, b) => b.at - a.at).map((r) => r.item);
   587→   587→}

   604→   604→/* ---------------- Watch time (Top 10 Today) ---------------- *
   605→   605→ *
   606→   606→ * "Top 10 Today" ranks anime by how long people ACTUALLY watch them — the
   607→   607→ * rail stays empty until real playback seconds exist. The player records
   608→   608→ * wall-clock deltas while the video is playing; they accumulate locally
   609→   609→ * (instant, offline-safe) and flush to the server so EVERY viewer's time
   610→   610→ * lands in one global per-day ranking, Crunchyroll-style.
   611→   611→ *
   612→   612→ * Local shape: day -> malId -> { seconds, title, image, sent }.
   613→   613→ * `sent` is the amount already ACKed by the server, so flushes POST only
   614→   614→ * the un-sent remainder. Display merges server + local with max() — never
   615→   615→ * a sum — so nothing is double-counted and offline time still shows.
   616→   616→ */
   617→   617→
   618→   618→const WATCHTIME_KEY = "animetest123.watchtime";
   619→   619→const WATCHTIME_EVENT = "watchtime-changed";
   620→   620→const WATCHTIME_KEEP_DAYS = 7; // local buckets older than a week are dropped
   621→   621→
   622→   622→export interface LocalWatchEntry {
   623→   623→  seconds: number;
   624→   624→  title?: string;
   625→   625→  image?: string | null;
   626→   626→  sent?: number; // seconds already accepted by the server
   627→   627→}
   628→   628→
   629→   629→type WatchTimeMap = Record<string, Record<string, LocalWatchEntry>>;
   630→   630→
   631→   631→/**
   632→   632→ * The SHARED calendar day ("YYYY-MM-DD", UTC) — one global bucket, so every
   633→   633→ * device in every timezone contributes to (and sees) the exact same
   634→   634→ * "Top 10 Today" ranking, like Netflix's worldwide lists. Rolls over at
   635→   635→ * UTC midnight for everyone.
   636→   636→ */
   637→   637→export function localDayKey(d = new Date()): string {
   638→   638→  return d.toISOString().slice(0, 10);
   639→   639→}
   640→   640→
   641→   641→function readWatchTime(): WatchTimeMap {
   642→   642→  return readJson<WatchTimeMap>(WATCHTIME_KEY, {});
   643→   643→}
   644→   644→
   645→   645→function writeWatchTime(map: WatchTimeMap) {
   646→   646→  // prune: keep only the newest KEEP_DAYS day buckets
   647→   647→  const days = Object.keys(map).sort();
   648→   648→  for (const stale of days.slice(0, Math.max(0, days.length - WATCHTIME_KEEP_DAYS))) delete map[stale];
   649→   649→  writeJson(WATCHTIME_KEY, map);
   650→   650→}
   651→   651→
   652→   652→/**
   653→   653→ * Add `delta` seconds of real playback to today's bucket for this anime and
   654→   654→ * flush the un-sent remainder to the server (best effort, never throws).
   655→   655→ * `keepalive` switches to sendBeacon for pagehide/unload flushes.
   656→   656→ */
   657→   657→export function recordWatchSeconds(
   658→   658→  args: { malId: number; title?: string; image?: string | null; delta: number; day?: string },
   659→   659→  opts?: { keepalive?: boolean },
   660→   660→) {
   661→   661→  const delta = Math.min(120, Math.max(0, Number(args.delta) || 0));
   662→   662→  if (delta < 1 || !Number.isFinite(args.malId)) return;
   663→   663→  const day = args.day ?? localDayKey(); // shared UTC day — one global ranking
   664→   664→
   665→   665→  let pending = 0;
   666→   666→  try {
   667→   667→    const map = readWatchTime();
   668→   668→    const dayBucket = (map[day] = map[day] ?? {});
   669→   669→    const entry = dayBucket[String(args.malId)] ?? { seconds: 0, sent: 0 };
   670→   670→    entry.seconds += delta;
   671→   671→    if (args.title) entry.title = args.title;
   672→   672→    if (args.image !== undefined) entry.image = args.image;
   673→   673→    dayBucket[String(args.malId)] = entry;
   674→   674→    pending = entry.seconds - (entry.sent ?? 0);
   675→   675→    if (pending >= 5) {
   676→   676→      entry.sent = entry.seconds; // optimistic — reverted below if the send fails outright
   677→   677→      writeWatchTime(map);
   678→   678→    } else {
   679→   679→      pending = 0;
   680→   680→      writeWatchTime(map);
   681→   681→    }
   682→   682→  } catch {
   683→   683→    return; // storage blocked — ranking is decorative, never fatal
   684→   684→  }
   685→   685→
   686→   686→  if (pending < 5) {
   687→   687→    // still notify listeners so Home can re-rank from the local bucket alone
   688→   688→    try {
   689→   689→      window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   690→   690→    } catch {
   691→   691→      /* ignore */
   692→   692→    }
   693→   693→    return;
   694→   694→  }
   695→   695→
   696→   696→  const body = JSON.stringify({
   697→   697→    userId: getUserId(),
   698→   698→    malId: args.malId,
   699→   699→    day,
   700→   700→    delta: Math.round(pending),
   701→   701→    title: args.title,
   702→   702→    image: args.image,
   703→   703→  });
   704→   704→  const revert = () => {
   705→   705→    try {
   706→   706→      const map = readWatchTime();
   707→   707→      const entry = map[day]?.[String(args.malId)];
   708→   708→      if (entry) {
   709→   709→        entry.sent = Math.max(0, (entry.sent ?? 0) - pending);
   710→   710→        writeWatchTime(map);
   711→   711→      }
   712→   712→    } catch {
   713→   713→      /* ignore */
   714→   714→    }
   715→   715→  };
   716→   716→  try {
   717→   717→    if (opts?.keepalive && "sendBeacon" in navigator) {
   718→   718→      navigator.sendBeacon("/api/watchtime", new Blob([body], { type: "application/json" }));
   719→   719→    } else {
   720→   720→      void outboxSend(`wt:${day}:${args.malId}`, "/api/watchtime", "POST", body).then((res) => {
   721→   721→        // null = the delta was parked in the offline outbox — treat as sent
   722→   722→        // (the flush replays it); only a live server rejection reverts
   723→   723→        if (res !== null && !res.ok) revert();
   724→   724→      });
   725→   725→    }
   726→   726→  } catch {
   727→   727→    revert();
   728→   728→  }
   729→   729→
   730→   730→  try {
   731→   731→    window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   732→   732→  } catch {
   733→   733→    /* ignore */
   734→   734→  }
   735→   735→}

   737→   737→/** Subscribe to watch-time changes (local accrual + cross-tab). */

   910→   910→  }
   911→   911→
   912→   912→  return [...byId.values()].sort((a, b) => b.seconds - a.seconds).slice(0, limit);

  1033→  1033→  const hr = Math.floor(min / 60);

  1035→  1035→  const day = Math.floor(hr / 24);

  1037→  1037→  return new Date(t).toLocaleDateString();

  1283→  1283→/**
  1284→  1284→ * Turn a user-chosen image into a home-screen icon data URL, DOWNSCALED on a
  1285→  1285→ * canvas so it can never blow the localStorage quota again (a full-res phone
  1286→  1286→ * photo as a data URL is multiple MB — that used to be stored raw INSIDE the
  1287→  1287→ * settings record, silently killing every settings write afterwards).
  1288→  1288→ * PNG keeps its transparency at 384px; everything else becomes a 512px JPEG.
  1289→  1289→ */
  1290→  1290→export async function fileToAppIconDataUrl(file: File): Promise<string> {
  1291→  1291→  if (!file.type.startsWith("image/")) throw new Error("Not an image");
  1292→  1292→  if (file.size > 25_000_000) throw new Error("Image too large (max ~25 MB)");
  1293→  1293→
  1294→  1294→  const url = URL.createObjectURL(file);
  1295→  1295→  let img: HTMLImageElement;
  1296→  1296→  try {
  1297→  1297→    img = await new Promise<HTMLImageElement>((res, rej) => {
  1298→  1298→      const el = new Image();
  1299→  1299→      el.onload = () => res(el);
  1300→  1300→      el.onerror = () => rej(new Error("Could not read image"));
  1301→  1301→      el.src = url;
  1302→  1302→    });
  1303→  1303→  } finally {
  1304→  1304→    setTimeout(() => URL.revokeObjectURL(url), 5_000);
  1305→  1305→  }
  1306→  1306→
  1307→  1307→  const png = file.type.includes("png");
  1308→  1308→  const S = png ? 384 : 512; // covers the 192/512 manifest slots
  1309→  1309→  const side = Math.min(img.naturalWidth || img.width, img.naturalHeight || img.height) || S;
  1310→  1310→  const canvas = document.createElement("canvas");
  1311→  1311→  canvas.width = S;
  1312→  1312→  canvas.height = S;
  1313→  1313→  const ctx = canvas.getContext("2d");
  1314→  1314→  if (!ctx) throw new Error("Canvas unavailable");
  1315→  1315→  const sw = side, sh = side;
  1316→  1316→  const sx = ((img.naturalWidth || img.width) - sw) / 2;
  1317→  1317→  const sy = ((img.naturalHeight || img.height) - sh) / 2;
  1318→  1318→  ctx.drawImage(img, sx, sy, sw, sh, 0, 0, S, S);
  1319→  1319→  return png ? canvas.toDataURL("image/png") : canvas.toDataURL("image/jpeg", 0.85);
  1320→  1320→}
