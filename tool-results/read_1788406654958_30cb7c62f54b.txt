    27→function fmtTime(s: number): string {
    28→  if (!Number.isFinite(s) || s < 0) return "0:00";
    29→  const h = Math.floor(s / 3600);
    30→  const m = Math.floor((s % 3600) / 60);
    31→  const sec = Math.floor(s % 60);
    32→  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${String(sec).padStart(2, "0")}` : `${m}:${String(sec).padStart(2, "0")}`;
    33→}
