    19→    19→function readJson<T>(key: string, fallback: T): T {
    20→    20→  if (typeof window === "undefined") return fallback;
    21→    21→  try {
    22→    22→    const raw = window.localStorage.getItem(key);
    23→    23→    return raw ? (JSON.parse(raw) as T) : fallback;
    24→    24→  } catch {
    25→    25→    return fallback;
    26→    26→  }
    27→    27→}
    28→    28→
    29→    29→function writeJson(key: string, value: unknown) {
    30→    30→  try {
    31→    31→    window.localStorage.setItem(key, JSON.stringify(value));
    32→    32→  } catch {
    33→    33→    /* storage full/blocked — ignore */
    34→    34→  }
    35→    35→}

    37→    37→const MYLIST_EVENT = "mylist-changed";

    40→    40→  window.addEventListener(MYLIST_EVENT, onChange);

    42→    42→  return () => {
    43→    43→    window.removeEventListener(MYLIST_EVENT, onChange);
    44→    44→    window.removeEventListener("storage", onChange);
    45→    45→  };
    46→    46→}

    56→    56→export function useMyList() {
    57→    57→  const raw = useSyncExternalStore(subscribeMyList, getMyListSnapshot, getMyListServerSnapshot);
    58→    58→
    59→    59→  const list: MyListEntry[] = (() => {
    60→    60→    try {
    61→    61→      return JSON.parse(raw) as MyListEntry[];
    62→    62→    } catch {
    63→    63→      return [];
    64→    64→    }
    65→    65→  })();
    66→    66→
    67→    67→  const has = useCallback((malId: number) => list.some((e) => e.malId === malId), [list]);
    68→    68→
    69→    69→  const toggle = useCallback((anime: Pick<AnimeLite, "malId" | "title" | "image">) => {
    70→    70→    const current = (() => {
    71→    71→      try {
    72→    72→        return JSON.parse(window.localStorage.getItem(MYLIST_KEY) ?? "[]") as MyListEntry[];
    73→    73→      } catch {
    74→    74→        return [] as MyListEntry[];
    75→    75→      }
    76→    76→    })();
    77→    77→    const exists = current.some((e) => e.malId === anime.malId);
    78→    78→    const next = exists
    79→    79→      ? current.filter((e) => e.malId !== anime.malId)
    80→    80→      : [{ malId: anime.malId, title: anime.title, image: anime.image, addedAt: Date.now() }, ...current];
    81→    81→    writeJson(MYLIST_KEY, next);
    82→    82→    try {
    83→    83→      window.dispatchEvent(new CustomEvent(MYLIST_EVENT));
    84→    84→    } catch {
    85→    85→      /* ignore */
    86→    86→    }
    87→    87→  }, []);
    88→    88→
    89→    89→  const malIds = list.map((e) => e.malId);
    90→    90→  return { list, malIds, has, toggle, ready: true };
    91→    91→}

   169→   169→export interface SaveProgressArgs {
   170→   170→  malId: number;
   171→   171→  episode: number;
   172→   172→  lang: "sub" | "dub";
   173→   173→  server: string;
   174→   174→  position: number;

   182→   182→/**
   183→   183→ * Persist progress locally immediately; mirror to the server. When
   184→   184→ * `keepalive` is set (page unload / tab hide) the request is sent with
   185→   185→ * keepalive so it survives navigation.
   186→   186→ */
   187→   187→export function saveProgressDetailed(args: SaveProgressArgs, opts?: { keepalive?: boolean }) {
   188→   188→  if (!Number.isFinite(args.position) || args.position < 5) return;
   189→   189→  // removed from Continue Watching — stale saves (heartbeats, beacons, the
   190→   190→  // unmount flush of a just-left player) must not resurrect it. The only
   191→   191→  // way back in is a deliberate watch (clearProgressTombstone on mount).
   192→   192→  if (isProgressRemoved(args.malId)) return;
   193→   193→  const completed = args.completed ?? (args.duration > 0 && args.position >= args.duration * 0.92);
   194→   194→  const entry: DetailedProgress = {
   195→   195→    pos: Math.floor(args.position),
   196→   196→    dur: Math.floor(args.duration || 0),
   197→   197→    completed,
   198→   198→    lang: args.lang,
   199→   199→    server: args.server,
   200→   200→    title: args.title,
   201→   201→    image: args.image,
   202→   202→    total: args.total,
   203→   203→    updatedAt: Date.now(),
   204→   204→  };
   205→   205→
   206→   206→  // local (instant — bars/resume render without a round trip)
   207→   207→  const map = readEpMap();
   208→   208→  const key = String(args.malId);
   209→   209→  map[key] = map[key] ?? {};
   210→   210→  map[key][String(args.episode)] = entry;
   211→   211→  writeJson(EP_PROGRESS_KEY, map);
   212→   212→
   213→   213→  // legacy "last watched episode" map (details badge, resume fallback)
   214→   214→  try {
   215→   215→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   216→   216→    legacy[key] = { ep: args.episode, lang: args.lang, pos: entry.pos, updatedAt: entry.updatedAt };
   217→   217→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   218→   218→  } catch {
   219→   219→    /* ignore */
   220→   220→  }
   221→   221→
   222→   222→  try {
   223→   223→    window.dispatchEvent(new CustomEvent("progress-changed"));
   224→   224→  } catch {
   225→   225→    /* ignore */
   226→   226→  }
   227→   227→
   228→   228→  // server mirror (best effort, never blocks playback)
   229→   229→  const userId = getUserId();
   230→   230→  const body = JSON.stringify({
   231→   231→    userId,
   232→   232→    malId: args.malId,
   233→   233→    episode: args.episode,
   234→   234→    lang: args.lang,
   235→   235→    server: args.server,
   236→   236→    position: entry.pos,
   237→   237→    duration: entry.dur,
   238→   238→    completed,
   239→   239→    title: args.title,
   240→   240→    image: args.image,
   241→   241→  });
   242→   242→  try {
   243→   243→    if (opts?.keepalive && "sendBeacon" in navigator) {
   244→   244→      navigator.sendBeacon("/api/progress", new Blob([body], { type: "application/json" }));
   245→   245→    } else {
   246→   246→      fetch("/api/progress", {
   247→   247→        method: "POST",
   248→   248→        headers: { "Content-Type": "application/json" },
   249→   249→        body,
   250→   250→        ...(opts?.keepalive ? { keepalive: true } : {}),
   251→   251→      }).catch(() => undefined);
   252→   252→    }
   253→   253→  } catch {
   254→   254→    /* ignore */
   255→   255→  }
   256→   256→}

   259→   259→export function removeProgress(malId: number) {
   260→   260→  // tombstone FIRST — an in-flight save must not re-add this anime between
   261→   261→  // here and the server DELETE ("deleted it but it came back" bug)
   262→   262→  markProgressRemoved(malId);
   263→   263→  const map = readEpMap();
   264→   264→  delete map[String(malId)];
   265→   265→  writeJson(EP_PROGRESS_KEY, map);
   266→   266→  try {
   267→   267→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   268→   268→    delete legacy[String(malId)];
   269→   269→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   270→   270→  } catch {
   271→   271→    /* ignore */
   272→   272→  }
   273→   273→  try {
   274→   274→    window.dispatchEvent(new CustomEvent("progress-changed"));
   275→   275→  } catch {
   276→   276→    /* ignore */
   277→   277→  }
   278→   278→  // keepalive so the removal survives the tab closing mid-request; the
   279→   279→  // server tombstones the removal too (blocks late beacon/heartbeat saves).
   280→   280→  // Retried twice — a lost DELETE (subway dead zone, app killed mid-flight)
   281→   281→  // used to leave the server row alive and the card resurrected on the next
   282→   282→  // Home visit; the retries close most of that window.
   283→   283→  const userId = getUserId();
   284→   284→  const del = (attempt: number): void => {
   285→   285→    try {
   286→   286→      fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${malId}`, {
   287→   287→        method: "DELETE",
   288→   288→        keepalive: true,
   289→   289→      }).catch(() => {
   290→   290→        if (attempt < 2) window.setTimeout(() => del(attempt + 1), 2_500 * (attempt + 1));
   291→   291→      });
   292→   292→    } catch {
   293→   293→      if (attempt < 2) window.setTimeout(() => del(attempt + 1), 2_500 * (attempt + 1));
   294→   294→    }
   295→   295→  };
   296→   296→  del(0);
   297→   297→}
   298→   298→
   299→   299→/**
   300→   300→ * Reconciliation for the "deleted it but it came back" bug: re-fire the
   301→   301→ * server DELETE for every anime this device has tombstoned in the last 48h.
   302→   302→ * A removal whose DELETE request was lost (offline moment, app killed) gets
   303→   303→ * healed the next time the Home rail loads; the local tombstone keeps the
   304→   304→ * card hidden in the meantime. Idempotent — the server DELETE just deletes
   305→   305→ * (and re-stamps) again.
   306→   306→ */
   307→   307→export function sweepRemovedProgress(): void {
   308→   308→  const map = readRemovedMap();
   309→   309→  const cutoff = Date.now() - 48 * 3600_000;
   310→   310→  const fresh = Object.entries(map).filter(([, at]) => at >= cutoff);
   311→   311→  if (!fresh.length) return;
   312→   312→  const userId = getUserId();
   313→   313→  for (const [malId, at] of fresh) {
   314→   314→    const id = Number(malId);
   315→   315→    if (!Number.isFinite(id)) continue;
   316→   316→    // pace slightly: these are keepalive DELETEs, tiny and idempotent
   317→   317→    window.setTimeout(() => {
   318→   318→      try {
   319→   319→        fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${id}`, {
   320→   320→          method: "DELETE",
   321→   321→          keepalive: true,
   322→   322→        }).catch(() => undefined);
   323→   323→      } catch {
   324→   324→        /* best-effort */
   325→   325→      }
   326→   326→      void at;
   327→   327→    }, 150 + Math.random() * 400);
   328→   328→  }
   329→   329→}

   331→   331→/* ---------------- removal tombstones (local half) ----------------
   332→   332→ * The server rejects saves for removed anime (RemovedProgress table), but
   333→   333→ * the local map must obey too: a save that raced the delete would otherwise
   334→   334→ * re-add bars/Continue entries the moment the server list is merged. */

   336→   336→const REMOVED_KEY = "animetest123.progress-removed";

   338→   338→function readRemovedMap(): Record<string, number> {
   339→   339→  return readJson<Record<string, number>>(REMOVED_KEY, {});

   342→   342→function markProgressRemoved(malId: number) {
   343→   343→  const map = readRemovedMap();
   344→   344→  map[String(malId)] = Date.now();

   348→   348→function isProgressRemoved(malId: number): boolean {

   352→   352→/**
   353→   353→ * Deliberate playback (WatchView mounted for this anime) — lift the removal
   354→   354→ * tombstone locally + on the server so progress tracking resumes.
   355→   355→ */
   356→   356→export function clearProgressTombstone(malId: number) {
   357→   357→  if (!isProgressRemoved(malId)) return;
   358→   358→  const map = readRemovedMap();
   359→   359→  delete map[String(malId)];
   360→   360→  writeJson(REMOVED_KEY, map);
   361→   361→  try {
   362→   362→    fetch("/api/progress", {
   363→   363→      method: "POST",
   364→   364→      headers: { "Content-Type": "application/json" },
   365→   365→      body: JSON.stringify({ userId: getUserId(), malId, resume: true }),
   366→   366→      keepalive: true,
   367→   367→    }).catch(() => undefined);
   368→   368→  } catch {
   369→   369→    /* ignore */
   370→   370→  }
   371→   371→}

   373→   373→/**
   374→   374→ * Server-side progress for one anime — used as a fallback so resume points
   375→   375→ * and red bars survive cleared storage / a fresh browser.
   376→   376→ */
   377→   377→export async function fetchServerProgress(malId: number): Promise<Record<string, DetailedProgress>> {
   378→   378→  try {
   379→   379→    const res = await fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`);
   380→   380→    if (!res.ok) return {};
   381→   381→    const data = (await res.json()) as {
   382→   382→      episodes: Array<{ episode: number; position: number; duration: number; completed: boolean; lang: string; server: string }>;
   383→   383→    };
   384→   384→    const out: Record<string, DetailedProgress> = {};
   385→   385→    for (const e of data.episodes ?? []) {
   386→   386→      out[String(e.episode)] = {
   387→   387→        pos: e.position,
   388→   388→        dur: e.duration,
   389→   389→        completed: e.completed,
   390→   390→        lang: e.lang === "dub" ? "dub" : "sub",
   391→   391→        server: e.server,
   392→   392→        updatedAt: Date.now(),
   393→   393→      };
   394→   394→    }
   395→   395→    // merge into the local map (server rows are authoritative for these keys)
   396→   396→    if (Object.keys(out).length) {
   397→   397→      const map = readEpMap();
   398→   398→      const key = String(malId);
   399→   399→      map[key] = { ...(map[key] ?? {}), ...out };
   400→   400→      writeJson(EP_PROGRESS_KEY, map);
   401→   401→    }
   402→   402→    return out;
   403→   403→  } catch {
   404→   404→    return {};
   405→   405→  }
   406→   406→}

   408→   408→/* ---------------- Continue Watching ---------------- */
   409→   409→
   410→   410→/**
   411→   411→ * Freshest in-progress episode per anime — server list merged with the
   412→   412→ * instant local map (so the rail renders with zero latency and still shows
   413→   413→ * entries when the server DB is cold).
   414→   414→ */
   415→   415→export function mergeContinueItems(serverItems: ContinueItem[]): ContinueItem[] {
   416→   416→  let localAll: Record<string, Record<string, DetailedProgress>> = {};
   417→   417→  try {
   418→   418→    localAll = JSON.parse(window.localStorage.getItem(EP_PROGRESS_KEY) ?? "{}");
   419→   419→  } catch {
   420→   420→    localAll = {};
   421→   421→  }
   422→   422→  // the local removal tombstone is the user's expressed intent: a server row
   423→   423→  // that slipped through the delete race (lost DELETE request, beacon that
   424→   424→  // landed a heartbeat ahead of it on another lambda) must not resurrect the
   425→   425→  // card in the UI. Deliberate playback lifts the tombstone (both halves).
   426→   426→  const removed = readRemovedMap();
   427→   427→
   428→   428→  type Row = { item: ContinueItem; at: number };
   429→   429→  const byAnime = new Map<number, Row>();
   430→   430→
   431→   431→  for (const it of serverItems) {
   432→   432→    if (it.completed) continue;
   433→   433→    if (removed[String(it.malId)]) continue; // deleted here — stays deleted
   434→   434→    byAnime.set(it.malId, { item: it, at: Date.parse(it.updatedAt) });
   435→   435→  }
   436→   436→
   437→   437→  for (const [malId, eps] of Object.entries(localAll)) {
   438→   438→    // removed from Continue Watching — a save that raced the delete must not
   439→   439→    // re-add the entry here either
   440→   440→    if (readRemovedMap()[malId]) continue;
   441→   441→    for (const [ep, e] of Object.entries(eps)) {
   442→   442→      if (e.completed || (e.pos || 0) < 5) continue;
   443→   443→      const malIdNum = Number(malId);
   444→   444→      const prev = byAnime.get(malIdNum);
   445→   445→      if (prev && prev.at >= e.updatedAt) continue;
   446→   446→      const percent = e.dur > 0 ? Math.min(100, Math.round((e.pos / e.dur) * 100)) : 0;
   447→   447→      byAnime.set(malIdNum, {
   448→   448→        at: e.updatedAt,
   449→   449→        item: {
   450→   450→          malId: malIdNum,
   451→   451→          title: e.title ?? prev?.item.title ?? "Unknown title",
   452→   452→          image: e.image ?? prev?.item.image ?? null,
   453→   453→          banner: prev?.item.banner ?? null,
   454→   454→          totalEpisodes: e.total ?? prev?.item.totalEpisodes ?? 0,
   455→   455→          type: prev?.item.type ?? null,
   456→   456→          episode: Number(ep),
   457→   457→          lang: e.lang,
   458→   458→          server: e.server,
   459→   459→          position: e.pos,
   460→   460→          duration: e.dur,
   461→   461→          percent,
   462→   462→          completed: false,
   463→   463→          updatedAt: new Date(e.updatedAt).toISOString(),
   464→   464→        },
   465→   465→      });
   466→   466→    }
   467→   467→  }
   468→   468→
   469→   469→  return [...byAnime.values()].sort((a, b) => b.at - a.at).map((r) => r.item);
   470→   470→}

   472→   472→/** GET /api/continue merged with local progress. */
   473→   473→export async function loadContinueItems(): Promise<ContinueItem[]> {
   474→   474→  let serverItems: ContinueItem[] = [];
   475→   475→  try {
   476→   476→    const res = await fetch(`/api/continue?userId=${encodeURIComponent(getUserId())}`);
   477→   477→    if (res.ok) {
   478→   478→      const data = (await res.json()) as { items: ContinueItem[] };
   479→   479→      serverItems = data.items ?? [];
   480→   480→    }
   481→   481→  } catch {

   483→   483→  }

   485→   485→}
   486→   486→
   487→   487→/* ---------------- Watch time (Top 10 Today) ---------------- *
   488→   488→ *
   489→   489→ * "Top 10 Today" ranks anime by how long people ACTUALLY watch them — the
   490→   490→ * rail stays empty until real playback seconds exist. The player records
   491→   491→ * wall-clock deltas while the video is playing; they accumulate locally
   492→   492→ * (instant, offline-safe) and flush to the server so EVERY viewer's time
   493→   493→ * lands in one global per-day ranking, Crunchyroll-style.
   494→   494→ *
   495→   495→ * Local shape: day -> malId -> { seconds, title, image, sent }.
   496→   496→ * `sent` is the amount already ACKed by the server, so flushes POST only
   497→   497→ * the un-sent remainder. Display merges server + local with max() — never
   498→   498→ * a sum — so nothing is double-counted and offline time still shows.
   499→   499→ */
   500→   500→
   501→   501→const WATCHTIME_KEY = "animetest123.watchtime";
   502→   502→const WATCHTIME_EVENT = "watchtime-changed";
   503→   503→const WATCHTIME_KEEP_DAYS = 7; // local buckets older than a week are dropped
   504→   504→
   505→   505→export interface LocalWatchEntry {
   506→   506→  seconds: number;
   507→   507→  title?: string;
   508→   508→  image?: string | null;
   509→   509→  sent?: number; // seconds already accepted by the server
   510→   510→}
   511→   511→
   512→   512→type WatchTimeMap = Record<string, Record<string, LocalWatchEntry>>;
   513→   513→
   514→   514→/**
   515→   515→ * The SHARED calendar day ("YYYY-MM-DD", UTC) — one global bucket, so every
   516→   516→ * device in every timezone contributes to (and sees) the exact same
   517→   517→ * "Top 10 Today" ranking, like Netflix's worldwide lists. Rolls over at
   518→   518→ * UTC midnight for everyone.
   519→   519→ */
   520→   520→export function localDayKey(d = new Date()): string {
   521→   521→  return d.toISOString().slice(0, 10);
   522→   522→}
   523→   523→
   524→   524→function readWatchTime(): WatchTimeMap {
   525→   525→  return readJson<WatchTimeMap>(WATCHTIME_KEY, {});
   526→   526→}
   527→   527→
   528→   528→function writeWatchTime(map: WatchTimeMap) {
   529→   529→  // prune: keep only the newest KEEP_DAYS day buckets
   530→   530→  const days = Object.keys(map).sort();
   531→   531→  for (const stale of days.slice(0, Math.max(0, days.length - WATCHTIME_KEEP_DAYS))) delete map[stale];
   532→   532→  writeJson(WATCHTIME_KEY, map);
   533→   533→}
   534→   534→
   535→   535→/**
   536→   536→ * Add `delta` seconds of real playback to today's bucket for this anime and
   537→   537→ * flush the un-sent remainder to the server (best effort, never throws).
   538→   538→ * `keepalive` switches to sendBeacon for pagehide/unload flushes.
   539→   539→ */
   540→   540→export function recordWatchSeconds(
   541→   541→  args: { malId: number; title?: string; image?: string | null; delta: number; day?: string },
   542→   542→  opts?: { keepalive?: boolean },
   543→   543→) {
   544→   544→  const delta = Math.min(120, Math.max(0, Number(args.delta) || 0));
   545→   545→  if (delta < 1 || !Number.isFinite(args.malId)) return;
   546→   546→  const day = args.day ?? localDayKey(); // shared UTC day — one global ranking
   547→   547→
   548→   548→  let pending = 0;
   549→   549→  try {
   550→   550→    const map = readWatchTime();
   551→   551→    const dayBucket = (map[day] = map[day] ?? {});
   552→   552→    const entry = dayBucket[String(args.malId)] ?? { seconds: 0, sent: 0 };
   553→   553→    entry.seconds += delta;
   554→   554→    if (args.title) entry.title = args.title;
   555→   555→    if (args.image !== undefined) entry.image = args.image;
   556→   556→    dayBucket[String(args.malId)] = entry;
   557→   557→    pending = entry.seconds - (entry.sent ?? 0);
   558→   558→    if (pending >= 5) {
   559→   559→      entry.sent = entry.seconds; // optimistic — reverted below if the send fails outright
   560→   560→      writeWatchTime(map);
   561→   561→    } else {
   562→   562→      pending = 0;
   563→   563→      writeWatchTime(map);
   564→   564→    }
   565→   565→  } catch {
   566→   566→    return; // storage blocked — ranking is decorative, never fatal
   567→   567→  }
   568→   568→
   569→   569→  if (pending < 5) {
   570→   570→    // still notify listeners so Home can re-rank from the local bucket alone
   571→   571→    try {
   572→   572→      window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   573→   573→    } catch {
   574→   574→      /* ignore */
   575→   575→    }
   576→   576→    return;
   577→   577→  }
   578→   578→
   579→   579→  const body = JSON.stringify({
   580→   580→    userId: getUserId(),
   581→   581→    malId: args.malId,
   582→   582→    day,
   583→   583→    delta: Math.round(pending),
   584→   584→    title: args.title,
   585→   585→    image: args.image,
   586→   586→  });
   587→   587→  const revert = () => {
   588→   588→    try {
   589→   589→      const map = readWatchTime();
   590→   590→      const entry = map[day]?.[String(args.malId)];
   591→   591→      if (entry) {
   592→   592→        entry.sent = Math.max(0, (entry.sent ?? 0) - pending);
   593→   593→        writeWatchTime(map);
   594→   594→      }
   595→   595→    } catch {
   596→   596→      /* ignore */
   597→   597→    }
   598→   598→  };
   599→   599→  try {
   600→   600→    if (opts?.keepalive && "sendBeacon" in navigator) {
   601→   601→      navigator.sendBeacon("/api/watchtime", new Blob([body], { type: "application/json" }));
   602→   602→    } else {
   603→   603→      fetch("/api/watchtime", {
   604→   604→        method: "POST",
   605→   605→        headers: { "Content-Type": "application/json" },
   606→   606→        body,
   607→   607→      })
   608→   608→        .then((res) => {
   609→   609→          if (!res.ok) revert(); // server rejected — retry on a later tick
   610→   610→        })
   611→   611→        .catch(revert);
   612→   612→    }
   613→   613→  } catch {
   614→   614→    revert();
   615→   615→  }
   616→   616→
   617→   617→  try {
   618→   618→    window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   619→   619→  } catch {
   620→   620→    /* ignore */
   621→   621→  }
   622→   622→}
   623→   623→
   624→   624→/** Subscribe to watch-time changes (local accrual + cross-tab). */
   625→   625→export function subscribeWatchTime(onChange: () => void) {
   626→   626→  window.addEventListener(WATCHTIME_EVENT, onChange);
   627→   627→  window.addEventListener("storage", onChange);
   628→   628→  return () => {
   629→   629→    window.removeEventListener(WATCHTIME_EVENT, onChange);
   630→   630→    window.removeEventListener("storage", onChange);
   631→   631→  };

   762→   762→export async function fetchWatchTimeTop(day: string, limit = 10): Promise<TopWatchEntry[]> {
   763→   763→  let serverItems: TopWatchEntry[] = [];
   764→   764→  try {
   765→   765→    const res = await fetch(`/api/watchtime?day=${encodeURIComponent(day)}&limit=${limit}`);
   766→   766→    if (res.ok) {
   767→   767→      const data = (await res.json()) as { items?: TopWatchEntry[] };
   768→   768→      serverItems = data.items ?? [];
   769→   769→    }
   770→   770→  } catch {
   771→   771→    /* offline / cold server — local buckets only */
   772→   772→  }
   773→   773→
   774→   774→  const byId = new Map<number, TopWatchEntry>();
   775→   775→  for (const it of serverItems) {
   776→   776→    if (Number.isFinite(it?.malId)) byId.set(it.malId, it);
   777→   777→  }
   778→   778→  try {
   779→   779→    const local = readWatchTime()[day] ?? {};
   780→   780→    for (const [key, entry] of Object.entries(local)) {
   781→   781→      const malId = Number(key);
   782→   782→      if (!Number.isFinite(malId)) continue;
   783→   783→      const prev = byId.get(malId);
   784→   784→      const seconds = Math.max(prev?.seconds ?? 0, entry.seconds || 0);
   785→   785→      if (seconds < 5) continue; // a stray couple of seconds isn't "watched"
   786→   786→      byId.set(malId, {
   787→   787→        ...(prev ?? minimalLite(malId, entry.title || `Anime ${malId}`, entry.image ?? null)),
   788→   788→        // local snapshots keep the rail informative when the catalog row is gone
   789→   789→        title: prev?.title || entry.title || `Anime ${malId}`,
   790→   790→        image: prev?.image ?? entry.image ?? null,
   791→   791→        malId,
   792→   792→        seconds,
   793→   793→      });
   794→   794→    }
   795→   795→  } catch {
   796→   796→    /* ignore */
   797→   797→  }
   798→   798→
   799→   799→  return [...byId.values()].sort((a, b) => b.seconds - a.seconds).slice(0, limit);
   800→   800→}

   914→   914→export function timeAgo(iso: string | number): string {
   915→   915→  const t = typeof iso === "number" ? iso : Date.parse(iso);
   916→   916→  const diff = Date.now() - t;
   917→   917→  const min = Math.floor(diff / 60_000);
   918→   918→  if (min < 1) return "just now";
   919→   919→  if (min < 60) return `${min}m ago`;
   920→   920→  const hr = Math.floor(min / 60);
   921→   921→  if (hr < 24) return `${hr}h ago`;
   922→   922→  const day = Math.floor(hr / 24);
   923→   923→  if (day < 30) return `${day}d ago`;
   924→   924→  return new Date(t).toLocaleDateString();
   925→   925→}

  1030→  1030→/**
  1031→  1031→ * Push the current settings into the document so EVERYTHING not explicitly
  1032→  1032→ * styled by a component follows along:
  1033→  1033→ *   - accent  -> CSS custom props --accent / --accent-deep (nav, buttons,
  1034→  1034→ *                switches, the settings screen itself)
  1035→  1035→ *   - subtitle look -> a `video::cue` rule (size / color / backdrop /
  1036→  1036→ *                outline) that native subtitle tracks pick up
  1037→  1037→ *   - app icon -> favicon + apple-touch-icon + a blob manifest so a fresh
  1038→  1038→ *                "Add to Home Screen" carries the custom icon (iOS applies
  1039→  1039→ *                the icon at add time; Android at install time — best effort
  1040→  1040→ *                by platform design)
  1041→  1041→ */
  1042→  1042→export function applySettingsToDocument(s: AppSettings): void {
  1043→  1043→  if (typeof document === "undefined") return;
  1044→  1044→
  1045→  1045→  const { color, deep } = accentColorOf(s.accent);
  1046→  1046→  document.documentElement.style.setProperty("--app-accent", color);
  1047→  1047→  document.documentElement.style.setProperty("--app-accent-deep", deep);
  1048→  1048→
  1049→  1049→  const px = (SUB_SIZES.find((x) => x.id === s.subSize) ?? SUB_SIZES[1]).px;
  1050→  1050→  const bg = Math.max(0, Math.min(100, s.subBg)) / 100;
  1051→  1051→  const shadows =
  1052→  1052→    s.subOutline === "none"
  1053→  1053→      ? "none"
  1054→  1054→      : s.subOutline === "outline"
  1055→  1055→        ? `-1px -1px 0 #000, 1px -1px 0 #000, -1px 1px 0 #000, 1px 1px 0 #000, 0 2px 3px rgba(0,0,0,.9)`
  1056→  1056→        : `0 1px 4px rgba(0,0,0,.9), 0 0 8px rgba(0,0,0,.55)`;
  1057→  1057→  const css = `video::cue { color: ${s.subColor}; background: rgba(0, 0, 0, ${bg}); text-shadow: ${shadows}; font-size: ${px}px; }`;
  1058→  1058→  let tag = document.getElementById("sub-cue-style") as HTMLStyleElement | null;
  1059→  1059→  if (!tag) {
  1060→  1060→    tag = document.createElement("style");
  1061→  1061→    tag.id = "sub-cue-style";
  1062→  1062→    document.head.appendChild(tag);
  1063→  1063→  }
  1064→  1064→  tag.textContent = css;
  1065→  1065→
  1066→  1066→  // custom app icon — swap every icon-ish link head tag
  1067→  1067→  try {
  1068→  1068→    if (s.iconDataUrl) {
  1069→  1069→      for (const [rel, sizes] of [
  1070→  1070→        ["icon", "32x32"],
  1071→  1071→        ["apple-touch-icon", "180x180"],
  1072→  1072→      ] as const) {
  1073→  1073→        let link = document.head.querySelector<HTMLLinkElement>(`link[rel="${rel}"]`);
  1074→  1074→        if (!link) {
  1075→  1075→          link = document.createElement("link");
  1076→  1076→          link.rel = rel;
  1077→  1077→          link.sizes = sizes;
  1078→  1078→          document.head.appendChild(link);
  1079→  1079→        }
  1080→  1080→        link.href = s.iconDataUrl;
  1081→  1081→      }
  1082→  1082→      // Android reads icons from the manifest at install time — build a
  1083→  1083→      // one-off blob manifest carrying the custom image at both sizes
  1084→  1084→      const manifest = {
  1085→  1085→        name: "Clyle",
  1086→  1086→        short_name: "Clyle",
  1087→  1087→        start_url: "/",
  1088→  1088→        display: "standalone",
  1089→  1089→        background_color: "#0a0a0a",
  1090→  1090→        theme_color: "#0a0a0a",
  1091→  1091→        icons: [
  1092→  1092→          { src: s.iconDataUrl, sizes: "192x192", type: "image/png" },
  1093→  1093→          { src: s.iconDataUrl, sizes: "512x512", type: "image/png" },
  1094→  1094→        ],
  1095→  1095→      };
  1096→  1096→      const url = URL.createObjectURL(new Blob([JSON.stringify(manifest)], { type: "application/manifest+json" }));
  1097→  1097→      const manLink = document.head.querySelector<HTMLLinkElement>('link[rel="manifest"]');
  1098→  1098→      if (manLink) manLink.href = url;
  1099→  1099→    }
  1100→  1100→  } catch {
  1101→  1101→    /* icon theming is cosmetic — never break the app over it */
  1102→  1102→  }
  1103→  1103→}
