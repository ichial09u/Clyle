    37→function fmtTime(s: number): string {
    38→  if (!Number.isFinite(s) || s < 0) return "0:00";
    39→  const h = Math.floor(s / 3600);
    40→  const m = Math.floor((s % 3600) / 60);
    41→  const sec = Math.floor(s % 60);
    42→  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${String(sec).padStart(2, "0")}` : `${m}:${String(sec).padStart(2, "0")}`;
    43→}
