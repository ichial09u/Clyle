  1821→                  {(stream?.subtitles ?? []).length === 0 ? (
  1822→                    <li className="text-[11.5px] text-gray-500">No captions for this episode</li>
  1823→                  ) : null}
  1824→                </ul>
  1825→              </div>
  1826→            </div>
  1827→
  1828→            {/* stream server — switch mid-episode, playback resumes at the same second */}
  1829→            <div className="mt-4 pt-3 border-t border-[#242424]">
  1830→              <h4 className="font-slab font-bold text-[14px] mb-2">Server</h4>
  1831→              <div className="flex gap-2">
  1832→                {(["zoko", "kiwi"] as const).map((s) => (
  1833→                  <button
  1834→                    key={s}
  1835→                    onClick={() => switchServer(s)}
  1836→                    disabled={s === server}
  1837→                    className={`flex-1 rounded-lg py-2 text-[12.5px] font-bold transition active:scale-[0.98] ${
  1838→                      s === server
  1839→                        ? "bg-[#d32f2f] text-white cursor-default"
  1840→                        : "bg-[#1c1c1c] hover:bg-[#282828] text-gray-200"
  1841→                    }`}

  1843→                    {s === "zoko" ? "Zoko (Server #1)" : "Kiwi (Server #2)"}

  1851→              {(backupDub ?? [])
  1852→                .filter((b) => (b.covered ?? []).includes(episode))
  1853→                .map((b) => {

  1855→                  return (

  1857→                      key={b.server}

  1902→      {/* download format dialog */}
  1903→      {downloadOpen ? (

  1925→                CANCEL
  1926→              </button>
  1927→            </div>
  1928→          </div>
  1929→        </div>
  1930→      ) : null}
  1931→    </div>
  1932→  );
  1933→}
