    19→function readJson<T>(key: string, fallback: T): T {
    20→  if (typeof window === "undefined") return fallback;
    21→  try {
    22→    const raw = window.localStorage.getItem(key);
    23→    return raw ? (JSON.parse(raw) as T) : fallback;
    24→  } catch {
    25→    return fallback;
    26→  }
    27→}
    28→
    29→function writeJson(key: string, value: unknown) {
    30→  try {
    31→    window.localStorage.setItem(key, JSON.stringify(value));
    32→  } catch {
    33→    /* storage full/blocked — ignore */
    34→  }
    35→}

    37→const MYLIST_EVENT = "mylist-changed";
    38→
    39→function subscribeMyList(onChange: () => void) {
    40→  window.addEventListener(MYLIST_EVENT, onChange);
    41→  window.addEventListener("storage", onChange);
    42→  return () => {
    43→    window.removeEventListener(MYLIST_EVENT, onChange);
    44→    window.removeEventListener("storage", onChange);
    45→  };
    46→}
    47→
    48→function getMyListSnapshot(): string {
    49→  return window.localStorage.getItem(MYLIST_KEY) ?? "[]";
    50→}
    51→
    52→function getMyListServerSnapshot(): string {
    53→  return "[]";
    54→}
    55→
    56→export function useMyList() {
    57→  const raw = useSyncExternalStore(subscribeMyList, getMyListSnapshot, getMyListServerSnapshot);
    58→
    59→  const list: MyListEntry[] = (() => {
    60→    try {
    61→      return JSON.parse(raw) as MyListEntry[];
    62→    } catch {
    63→      return [];
    64→    }
    65→  })();
    66→
    67→  const has = useCallback((malId: number) => list.some((e) => e.malId === malId), [list]);
    68→
    69→  const toggle = useCallback((anime: Pick<AnimeLite, "malId" | "title" | "image">) => {
    70→    const current = (() => {
    71→      try {
    72→        return JSON.parse(window.localStorage.getItem(MYLIST_KEY) ?? "[]") as MyListEntry[];
    73→      } catch {
    74→        return [] as MyListEntry[];
    75→      }
    76→    })();
    77→    const exists = current.some((e) => e.malId === anime.malId);
    78→    const next = exists
    79→      ? current.filter((e) => e.malId !== anime.malId)
    80→      : [{ malId: anime.malId, title: anime.title, image: anime.image, addedAt: Date.now() }, ...current];
    81→    writeJson(MYLIST_KEY, next);
    82→    try {
    83→      window.dispatchEvent(new CustomEvent(MYLIST_EVENT));
    84→    } catch {
    85→      /* ignore */
    86→    }
    87→  }, []);
    88→
    89→  const malIds = list.map((e) => e.malId);
    90→  return { list, malIds, has, toggle, ready: true };
    91→}

   182→/**
   183→ * Persist progress locally immediately; mirror to the server. When
   184→ * `keepalive` is set (page unload / tab hide) the request is sent with
   185→ * keepalive so it survives navigation.
   186→ */
   187→export function saveProgressDetailed(args: SaveProgressArgs, opts?: { keepalive?: boolean }) {
   188→  if (!Number.isFinite(args.position) || args.position < 5) return;
   189→  // removed from Continue Watching — stale saves (heartbeats, beacons, the
   190→  // unmount flush of a just-left player) must not resurrect it. The only
   191→  // way back in is a deliberate watch (clearProgressTombstone on mount).
   192→  if (isProgressRemoved(args.malId)) return;
   193→  const completed = args.completed ?? (args.duration > 0 && args.position >= args.duration * 0.92);
   194→  const entry: DetailedProgress = {
   195→    pos: Math.floor(args.position),
   196→    dur: Math.floor(args.duration || 0),
   197→    completed,
   198→    lang: args.lang,
   199→    server: args.server,
   200→    title: args.title,
   201→    image: args.image,
   202→    total: args.total,
   203→    updatedAt: Date.now(),
   204→  };
   205→
   206→  // local (instant — bars/resume render without a round trip)
   207→  const map = readEpMap();
   208→  const key = String(args.malId);
   209→  map[key] = map[key] ?? {};
   210→  map[key][String(args.episode)] = entry;
   211→  writeJson(EP_PROGRESS_KEY, map);
   212→
   213→  // legacy "last watched episode" map (details badge, resume fallback)
   214→  try {
   215→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   216→    legacy[key] = { ep: args.episode, lang: args.lang, pos: entry.pos, updatedAt: entry.updatedAt };
   217→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   218→  } catch {
   219→    /* ignore */
   220→  }
   221→
   222→  try {
   223→    window.dispatchEvent(new CustomEvent("progress-changed"));
   224→  } catch {
   225→    /* ignore */
   226→  }
   227→
   228→  // server mirror (best effort, never blocks playback)
   229→  const userId = getUserId();
   230→  const body = JSON.stringify({
   231→    userId,
   232→    malId: args.malId,
   233→    episode: args.episode,
   234→    lang: args.lang,
   235→    server: args.server,
   236→    position: entry.pos,
   237→    duration: entry.dur,
   238→    completed,
   239→    title: args.title,
   240→    image: args.image,
   241→  });
   242→  try {
   243→    if (opts?.keepalive && "sendBeacon" in navigator) {
   244→      navigator.sendBeacon("/api/progress", new Blob([body], { type: "application/json" }));
   245→    } else {
   246→      fetch("/api/progress", {
   247→        method: "POST",
   248→        headers: { "Content-Type": "application/json" },
   249→        body,
   250→        ...(opts?.keepalive ? { keepalive: true } : {}),
   251→      }).catch(() => undefined);
   252→    }
   253→  } catch {
   254→    /* ignore */
   255→  }
   256→}

   258→/** Forget everything watched for one anime (Continue Watching "X" button). */
   259→export function removeProgress(malId: number) {
   260→  // tombstone FIRST — an in-flight save must not re-add this anime between
   261→  // here and the server DELETE ("deleted it but it came back" bug)
   262→  markProgressRemoved(malId);
   263→  const map = readEpMap();
   264→  delete map[String(malId)];
   265→  writeJson(EP_PROGRESS_KEY, map);
   266→  try {
   267→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   268→    delete legacy[String(malId)];
   269→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   270→  } catch {
   271→    /* ignore */
   272→  }
   273→  try {
   274→    window.dispatchEvent(new CustomEvent("progress-changed"));
   275→  } catch {
   276→    /* ignore */
   277→  }
   278→  // keepalive so the removal survives the tab closing mid-request; the
   279→  // server tombstones the removal too (blocks late beacon/heartbeat saves).
   280→  // Retried twice — a lost DELETE (subway dead zone, app killed mid-flight)
   281→  // used to leave the server row alive and the card resurrected on the next
   282→  // Home visit; the retries close most of that window.
   283→  const userId = getUserId();
   284→  const del = (attempt: number): void => {
   285→    try {
   286→      fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${malId}`, {
   287→        method: "DELETE",
   288→        keepalive: true,
   289→      }).catch(() => {
   290→        if (attempt < 2) window.setTimeout(() => del(attempt + 1), 2_500 * (attempt + 1));
   291→      });
   292→    } catch {
   293→      if (attempt < 2) window.setTimeout(() => del(attempt + 1), 2_500 * (attempt + 1));
   294→    }
   295→  };
   296→  del(0);
   297→}
   298→
   299→/**
   300→ * Reconciliation for the "deleted it but it came back" bug: re-fire the
   301→ * server DELETE for every anime this device has tombstoned in the last 48h.
   302→ * A removal whose DELETE request was lost (offline moment, app killed) gets
   303→ * healed the next time the Home rail loads; the local tombstone keeps the
   304→ * card hidden in the meantime. Idempotent — the server DELETE just deletes
   305→ * (and re-stamps) again.
   306→ */
   307→export function sweepRemovedProgress(): void {
   308→  const map = readRemovedMap();
   309→  const cutoff = Date.now() - 48 * 3600_000;
   310→  const fresh = Object.entries(map).filter(([, at]) => at >= cutoff);
   311→  if (!fresh.length) return;
   312→  const userId = getUserId();
   313→  for (const [malId, at] of fresh) {
   314→    const id = Number(malId);
   315→    if (!Number.isFinite(id)) continue;
   316→    // pace slightly: these are keepalive DELETEs, tiny and idempotent
   317→    window.setTimeout(() => {
   318→      try {
   319→        fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${id}`, {
   320→          method: "DELETE",
   321→          keepalive: true,
   322→        }).catch(() => undefined);
   323→      } catch {
   324→        /* best-effort */
   325→      }
   326→      void at;
   327→    }, 150 + Math.random() * 400);
   328→  }
   329→}

   331→/* ---------------- removal tombstones (local half) ----------------
   332→ * The server rejects saves for removed anime (RemovedProgress table), but
   333→ * the local map must obey too: a save that raced the delete would otherwise
   334→ * re-add bars/Continue entries the moment the server list is merged. */
   335→
   336→const REMOVED_KEY = "animetest123.progress-removed";
   337→
   338→function readRemovedMap(): Record<string, number> {
   339→  return readJson<Record<string, number>>(REMOVED_KEY, {});
   340→}
   341→
   342→function markProgressRemoved(malId: number) {
   343→  const map = readRemovedMap();
   344→  map[String(malId)] = Date.now();
   345→  writeJson(REMOVED_KEY, map);
   346→}
   347→
   348→function isProgressRemoved(malId: number): boolean {
   349→  return Boolean(readRemovedMap()[String(malId)]);
   350→}

   352→/**
   353→ * Deliberate playback (WatchView mounted for this anime) — lift the removal
   354→ * tombstone locally + on the server so progress tracking resumes.
   355→ */
   356→export function clearProgressTombstone(malId: number) {
   357→  if (!isProgressRemoved(malId)) return;
   358→  const map = readRemovedMap();
   359→  delete map[String(malId)];
   360→  writeJson(REMOVED_KEY, map);
   361→  try {
   362→    fetch("/api/progress", {
   363→      method: "POST",
   364→      headers: { "Content-Type": "application/json" },
   365→      body: JSON.stringify({ userId: getUserId(), malId, resume: true }),
   366→      keepalive: true,
   367→    }).catch(() => undefined);
   368→  } catch {
   369→    /* ignore */
   370→  }
   371→}

   373→/**
   374→ * Server-side progress for one anime — used as a fallback so resume points
   375→ * and red bars survive cleared storage / a fresh browser.
   376→ */
   377→export async function fetchServerProgress(malId: number): Promise<Record<string, DetailedProgress>> {
   378→  try {
   379→    const res = await fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`);
   380→    if (!res.ok) return {};
   381→    const data = (await res.json()) as {
   382→      episodes: Array<{ episode: number; position: number; duration: number; completed: boolean; lang: string; server: string }>;
   383→    };
   384→    const out: Record<string, DetailedProgress> = {};
   385→    for (const e of data.episodes ?? []) {
   386→      out[String(e.episode)] = {
   387→        pos: e.position,
   388→        dur: e.duration,
   389→        completed: e.completed,
   390→        lang: e.lang === "dub" ? "dub" : "sub",
   391→        server: e.server,
   392→        updatedAt: Date.now(),
   393→      };
   394→    }
   395→    // merge into the local map (server rows are authoritative for these keys)
   396→    if (Object.keys(out).length) {
   397→      const map = readEpMap();
   398→      const key = String(malId);
   399→      map[key] = { ...(map[key] ?? {}), ...out };
   400→      writeJson(EP_PROGRESS_KEY, map);
   401→    }
   402→    return out;
   403→  } catch {
   404→    return {};
   405→  }
   406→}

   408→/* ---------------- Continue Watching ---------------- */
   409→
   410→/**
   411→ * Freshest in-progress episode per anime — server list merged with the
   412→ * instant local map (so the rail renders with zero latency and still shows
   413→ * entries when the server DB is cold).
   414→ */
   415→export function mergeContinueItems(serverItems: ContinueItem[]): ContinueItem[] {
   416→  let localAll: Record<string, Record<string, DetailedProgress>> = {};
   417→  try {
   418→    localAll = JSON.parse(window.localStorage.getItem(EP_PROGRESS_KEY) ?? "{}");
   419→  } catch {
   420→    localAll = {};
   421→  }
   422→  // the local removal tombstone is the user's expressed intent: a server row
   423→  // that slipped through the delete race (lost DELETE request, beacon that
   424→  // landed a heartbeat ahead of it on another lambda) must not resurrect the
   425→  // card in the UI. Deliberate playback lifts the tombstone (both halves).
   426→  const removed = readRemovedMap();
   427→
   428→  type Row = { item: ContinueItem; at: number };
   429→  const byAnime = new Map<number, Row>();
   430→
   431→  for (const it of serverItems) {
   432→    if (it.completed) continue;
   433→    if (removed[String(it.malId)]) continue; // deleted here — stays deleted
   434→    byAnime.set(it.malId, { item: it, at: Date.parse(it.updatedAt) });
   435→  }
   436→
   437→  for (const [malId, eps] of Object.entries(localAll)) {
   438→    // removed from Continue Watching — a save that raced the delete must not
   439→    // re-add the entry here either
   440→    if (readRemovedMap()[malId]) continue;
   441→    for (const [ep, e] of Object.entries(eps)) {
   442→      if (e.completed || (e.pos || 0) < 5) continue;
   443→      const malIdNum = Number(malId);
   444→      const prev = byAnime.get(malIdNum);
   445→      if (prev && prev.at >= e.updatedAt) continue;
   446→      const percent = e.dur > 0 ? Math.min(100, Math.round((e.pos / e.dur) * 100)) : 0;
   447→      byAnime.set(malIdNum, {
   448→        at: e.updatedAt,
   449→        item: {
   450→          malId: malIdNum,
   451→          title: e.title ?? prev?.item.title ?? "Unknown title",
   452→          image: e.image ?? prev?.item.image ?? null,
   453→          banner: prev?.item.banner ?? null,
   454→          totalEpisodes: e.total ?? prev?.item.totalEpisodes ?? 0,
   455→          type: prev?.item.type ?? null,
   456→          episode: Number(ep),
   457→          lang: e.lang,
   458→          server: e.server,
   459→          position: e.pos,
   460→          duration: e.dur,
   461→          percent,
   462→          completed: false,
   463→          updatedAt: new Date(e.updatedAt).toISOString(),
   464→        },
   465→      });
   466→    }
   467→  }
   468→
   469→  return [...byAnime.values()].sort((a, b) => b.at - a.at).map((r) => r.item);
   470→}

   487→/* ---------------- Watch time (Top 10 Today) ---------------- *
   488→ *
   489→ * "Top 10 Today" ranks anime by how long people ACTUALLY watch them — the
   490→ * rail stays empty until real playback seconds exist. The player records
   491→ * wall-clock deltas while the video is playing; they accumulate locally
   492→ * (instant, offline-safe) and flush to the server so EVERY viewer's time
   493→ * lands in one global per-day ranking, Crunchyroll-style.
   494→ *
   495→ * Local shape: day -> malId -> { seconds, title, image, sent }.
   496→ * `sent` is the amount already ACKed by the server, so flushes POST only
   497→ * the un-sent remainder. Display merges server + local with max() — never
   498→ * a sum — so nothing is double-counted and offline time still shows.
   499→ */
   500→
   501→const WATCHTIME_KEY = "animetest123.watchtime";
   502→const WATCHTIME_EVENT = "watchtime-changed";
   503→const WATCHTIME_KEEP_DAYS = 7; // local buckets older than a week are dropped
   504→
   505→export interface LocalWatchEntry {
   506→  seconds: number;
   507→  title?: string;
   508→  image?: string | null;
   509→  sent?: number; // seconds already accepted by the server
   510→}
   511→
   512→type WatchTimeMap = Record<string, Record<string, LocalWatchEntry>>;
   513→
   514→/**
   515→ * The SHARED calendar day ("YYYY-MM-DD", UTC) — one global bucket, so every
   516→ * device in every timezone contributes to (and sees) the exact same
   517→ * "Top 10 Today" ranking, like Netflix's worldwide lists. Rolls over at
   518→ * UTC midnight for everyone.
   519→ */
   520→export function localDayKey(d = new Date()): string {
   521→  return d.toISOString().slice(0, 10);
   522→}
   523→
   524→function readWatchTime(): WatchTimeMap {
   525→  return readJson<WatchTimeMap>(WATCHTIME_KEY, {});
   526→}
   527→
   528→function writeWatchTime(map: WatchTimeMap) {
   529→  // prune: keep only the newest KEEP_DAYS day buckets
   530→  const days = Object.keys(map).sort();
   531→  for (const stale of days.slice(0, Math.max(0, days.length - WATCHTIME_KEEP_DAYS))) delete map[stale];
   532→  writeJson(WATCHTIME_KEY, map);
   533→}
   534→
   535→/**
   536→ * Add `delta` seconds of real playback to today's bucket for this anime and
   537→ * flush the un-sent remainder to the server (best effort, never throws).
   538→ * `keepalive` switches to sendBeacon for pagehide/unload flushes.
   539→ */
   540→export function recordWatchSeconds(
   541→  args: { malId: number; title?: string; image?: string | null; delta: number; day?: string },
   542→  opts?: { keepalive?: boolean },
   543→) {
   544→  const delta = Math.min(120, Math.max(0, Number(args.delta) || 0));
   545→  if (delta < 1 || !Number.isFinite(args.malId)) return;
   546→  const day = args.day ?? localDayKey(); // shared UTC day — one global ranking
   547→
   548→  let pending = 0;
   549→  try {
   550→    const map = readWatchTime();
   551→    const dayBucket = (map[day] = map[day] ?? {});
   552→    const entry = dayBucket[String(args.malId)] ?? { seconds: 0, sent: 0 };
   553→    entry.seconds += delta;
   554→    if (args.title) entry.title = args.title;
   555→    if (args.image !== undefined) entry.image = args.image;
   556→    dayBucket[String(args.malId)] = entry;
   557→    pending = entry.seconds - (entry.sent ?? 0);
   558→    if (pending >= 5) {
   559→      entry.sent = entry.seconds; // optimistic — reverted below if the send fails outright
   560→      writeWatchTime(map);
   561→    } else {
   562→      pending = 0;
   563→      writeWatchTime(map);
   564→    }
   565→  } catch {
   566→    return; // storage blocked — ranking is decorative, never fatal
   567→  }
   568→
   569→  if (pending < 5) {
   570→    // still notify listeners so Home can re-rank from the local bucket alone
   571→    try {
   572→      window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   573→    } catch {
   574→      /* ignore */
   575→    }
   576→    return;
   577→  }
   578→
   579→  const body = JSON.stringify({
   580→    userId: getUserId(),
   581→    malId: args.malId,
   582→    day,
   583→    delta: Math.round(pending),
   584→    title: args.title,
   585→    image: args.image,
   586→  });
   587→  const revert = () => {
   588→    try {
   589→      const map = readWatchTime();
   590→      const entry = map[day]?.[String(args.malId)];
   591→      if (entry) {
   592→        entry.sent = Math.max(0, (entry.sent ?? 0) - pending);
   593→        writeWatchTime(map);
   594→      }
   595→    } catch {
   596→      /* ignore */
   597→    }
   598→  };
   599→  try {
   600→    if (opts?.keepalive && "sendBeacon" in navigator) {
   601→      navigator.sendBeacon("/api/watchtime", new Blob([body], { type: "application/json" }));
   602→    } else {
   603→      fetch("/api/watchtime", {
   604→        method: "POST",
   605→        headers: { "Content-Type": "application/json" },
   606→        body,
   607→      })
   608→        .then((res) => {
   609→          if (!res.ok) revert(); // server rejected — retry on a later tick
   610→        })
   611→        .catch(revert);
   612→    }
   613→  } catch {
   614→    revert();
   615→  }
   616→
   617→  try {
   618→    window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   619→  } catch {
   620→    /* ignore */
   621→  }
   622→}

   757→/**
   758→ * The day's most-watched anime: the server's global ranking merged with this
   759→ * device's instant local buckets (max, not sum — no double counting; local
   760→ * covers the un-flushed tail and offline/served-cold cases).
   761→ */
   762→export async function fetchWatchTimeTop(day: string, limit = 10): Promise<TopWatchEntry[]> {
   763→  let serverItems: TopWatchEntry[] = [];
   764→  try {
   765→    const res = await fetch(`/api/watchtime?day=${encodeURIComponent(day)}&limit=${limit}`);
   766→    if (res.ok) {
   767→      const data = (await res.json()) as { items?: TopWatchEntry[] };
   768→      serverItems = data.items ?? [];
   769→    }
   770→  } catch {
   771→    /* offline / cold server — local buckets only */
   772→  }
   773→
   774→  const byId = new Map<number, TopWatchEntry>();
   775→  for (const it of serverItems) {
   776→    if (Number.isFinite(it?.malId)) byId.set(it.malId, it);
   777→  }
   778→  try {
   779→    const local = readWatchTime()[day] ?? {};
   780→    for (const [key, entry] of Object.entries(local)) {
   781→      const malId = Number(key);
   782→      if (!Number.isFinite(malId)) continue;
   783→      const prev = byId.get(malId);
   784→      const seconds = Math.max(prev?.seconds ?? 0, entry.seconds || 0);
   785→      if (seconds < 5) continue; // a stray couple of seconds isn't "watched"
   786→      byId.set(malId, {
   787→        ...(prev ?? minimalLite(malId, entry.title || `Anime ${malId}`, entry.image ?? null)),
   788→        // local snapshots keep the rail informative when the catalog row is gone
   789→        title: prev?.title || entry.title || `Anime ${malId}`,
   790→        image: prev?.image ?? entry.image ?? null,
   791→        malId,
   792→        seconds,
   793→      });
   794→    }
   795→  } catch {
   796→    /* ignore */
   797→  }
   798→
   799→  return [...byId.values()].sort((a, b) => b.seconds - a.seconds).slice(0, limit);
   800→}

   914→export function timeAgo(iso: string | number): string {
   915→  const t = typeof iso === "number" ? iso : Date.parse(iso);
   916→  const diff = Date.now() - t;
   917→  const min = Math.floor(diff / 60_000);
   918→  if (min < 1) return "just now";
   919→  if (min < 60) return `${min}m ago`;
   920→  const hr = Math.floor(min / 60);
   921→  if (hr < 24) return `${hr}h ago`;
   922→  const day = Math.floor(hr / 24);
   923→  if (day < 30) return `${day}d ago`;
   924→  return new Date(t).toLocaleDateString();
   925→}

  1032→ * styled by a component follows along:

  1034→ *                switches, the settings screen itself)
  1035→ *   - subtitle look -> a `video::cue` rule (size / color / backdrop /
  1036→ *                outline) that native subtitle tracks pick up
  1037→ *   - app icon -> favicon + apple-touch-icon + a blob manifest so a fresh

  1050→  const bg = Math.max(0, Math.min(100, s.subBg)) / 100;
  1051→  const shadows =
  1052→    s.subOutline === "none"

  1054→      : s.subOutline === "outline"

  1056→        : `0 1px 4px rgba(0,0,0,.9), 0 0 8px rgba(0,0,0,.55)`;
  1057→  const css = `video::cue { color: ${s.subColor}; background: rgba(0, 0, 0, ${bg}); text-shadow: ${shadows}; font-size: ${px}px; }`;

  1090→        theme_color: "#0a0a0a",
  1091→        icons: [
  1092→          { src: s.iconDataUrl, sizes: "192x192", type: "image/png" },
  1093→          { src: s.iconDataUrl, sizes: "512x512", type: "image/png" },
  1094→        ],
