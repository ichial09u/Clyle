    19→function readJson<T>(key: string, fallback: T): T {
    20→  if (typeof window === "undefined") return fallback;
    21→  try {
    22→    const raw = window.localStorage.getItem(key);
    23→    return raw ? (JSON.parse(raw) as T) : fallback;
    24→  } catch {
    25→    return fallback;
    26→  }
    27→}
    28→
    29→function writeJson(key: string, value: unknown) {
    30→  try {
    31→    window.localStorage.setItem(key, JSON.stringify(value));
    32→  } catch {
    33→    /* storage full/blocked — ignore */
    34→  }
    35→}

    37→/* ---------------- offline outbox (sync when back online) ----------------
    38→ * Every server write the app makes while offline (progress saves, watch
    39→ * time, removal deletes, my-list hydrate) is parked in localStorage under a
    40→ * dedupe key and replayed the moment the connection is back — either by the
    41→ * browser's `online` event or the next time the app opens. Same-key items
    42→ * replace each other (latest position wins), so a whole offline session of
    43→ * heartbeats collapses into one fresh write per episode.
    44→ */
    45→
    46→const OUTBOX_KEY = "animetest123.outbox";
    47→const OUTBOX_MAX = 120;
    48→
    49→interface OutboxItem {
    50→  k: string; // dedupe key
    51→  url: string;
    52→  method: "POST" | "DELETE";
    53→  body?: string;
    54→  at: number;
    55→}
    56→
    57→function readOutbox(): OutboxItem[] {
    58→  return readJson<OutboxItem[]>(OUTBOX_KEY, []);
    59→}
    60→
    61→function writeOutbox(items: OutboxItem[]) {
    62→  writeJson(OUTBOX_KEY, items.slice(-OUTBOX_MAX));
    63→}
    64→
    65→function queueOutbox(key: string, url: string, method: "POST" | "DELETE", body?: string) {
    66→  try {
    67→    const items = readOutbox().filter((i) => i.k !== key);
    68→    items.push({ k: key, url, method, body, at: Date.now() });
    69→    writeOutbox(items);
    70→  } catch {
    71→    /* ignore */
    72→  }
    73→}
    74→
    75→function dequeueOutbox(key: string) {
    76→  try {
    77→    const items = readOutbox();
    78→    if (!items.some((i) => i.k === key)) return;
    79→    writeOutbox(items.filter((i) => i.k !== key));
    80→  } catch {
    81→    /* ignore */
    82→  }
    83→}
    84→
    85→/**
    86→ * Send a write request, or park it in the outbox when the network is down.
    87→ * Returns the Response, or null when the item was queued (or the request
    88→ * failed outright and is now queued). `queueFirst` parks the item BEFORE
    89→ * attempting — for must-eventually-land writes (progress deletes) that
    90→ * survive the page dying mid-request.
    91→ */
    92→export function outboxSend(
    93→  key: string,
    94→  url: string,
    95→  method: "POST" | "DELETE",
    96→  body?: string,
    97→  opts?: { queueFirst?: boolean },
    98→): Promise<Response | null> {
    99→  if (opts?.queueFirst) queueOutbox(key, url, method, body);
   100→  if (typeof navigator !== "undefined" && navigator.onLine === false) {
   101→    if (!opts?.queueFirst) queueOutbox(key, url, method, body);
   102→    return Promise.resolve(null);
   103→  }
   104→  return fetch(url, {
   105→    method,
   106→    headers: body ? { "Content-Type": "application/json" } : undefined,
   107→    body,
   108→    ...(opts?.queueFirst ? { keepalive: true } : {}),
   109→  })
   110→    .then((res) => {
   111→      if (res.ok && opts?.queueFirst) dequeueOutbox(key);
   112→      return res;
   113→    })
   114→    .catch(() => {
   115→      if (!opts?.queueFirst) queueOutbox(key, url, method, body);
   116→      return null;
   117→    });
   118→}
   119→
   120→let flushingOutbox = false;
   121→
   122→/**
   123→ * Replay the outbox in order. Stops at the first failure (still offline /
   124→ * server refusing) so later items don't jump the queue. Dispatches
   125→ * `clyle-synced` (detail = number of synced writes) when anything landed.
   126→ */
   127→export async function flushOutbox(): Promise<number> {
   128→  if (flushingOutbox) return 0;
   129→  if (typeof navigator !== "undefined" && navigator.onLine === false) return 0;
   130→  flushingOutbox = true;
   131→  let synced = 0;
   132→  try {
   133→    let items = readOutbox();
   134→    while (items.length) {
   135→      const item = items[0];
   136→      try {
   137→        const res = await fetch(item.url, {
   138→          method: item.method,
   139→          headers: item.body ? { "Content-Type": "application/json" } : undefined,
   140→          body: item.body,
   141→        });
   142→        if (!res.ok) break; // server refused — retry on a later flush
   143→        items = items.slice(1);
   144→        writeOutbox(items);
   145→        synced++;
   146→      } catch {
   147→        break; // still offline — stop
   148→      }
   149→    }
   150→  } finally {
   151→    flushingOutbox = false;
   152→  }
   153→  if (synced > 0) {
   154→    try {
   155→      window.dispatchEvent(new CustomEvent("clyle-synced", { detail: synced }));
   156→      window.dispatchEvent(new CustomEvent("progress-changed"));
   157→    } catch {
   158→      /* ignore */
   159→    }
   160→  }
   161→  return synced;
   162→}

   166→  return readOutbox().length;

   188→export function useMyList() {
   189→  const raw = useSyncExternalStore(subscribeMyList, getMyListSnapshot, getMyListServerSnapshot);
   190→
   191→  const list: MyListEntry[] = (() => {
   192→    try {
   193→      return JSON.parse(raw) as MyListEntry[];
   194→    } catch {
   195→      return [];
   196→    }
   197→  })();
   198→
   199→  const has = useCallback((malId: number) => list.some((e) => e.malId === malId), [list]);
   200→
   201→  const toggle = useCallback((anime: Pick<AnimeLite, "malId" | "title" | "image">) => {
   202→    const current = (() => {
   203→      try {
   204→        return JSON.parse(window.localStorage.getItem(MYLIST_KEY) ?? "[]") as MyListEntry[];
   205→      } catch {
   206→        return [] as MyListEntry[];
   207→      }
   208→    })();
   209→    const exists = current.some((e) => e.malId === anime.malId);
   210→    const next = exists
   211→      ? current.filter((e) => e.malId !== anime.malId)
   212→      : [{ malId: anime.malId, title: anime.title, image: anime.image, addedAt: Date.now() }, ...current];
   213→    writeJson(MYLIST_KEY, next);
   214→    try {
   215→      window.dispatchEvent(new CustomEvent(MYLIST_EVENT));
   216→    } catch {
   217→      /* ignore */
   218→    }
   219→  }, []);
   220→
   221→  const malIds = list.map((e) => e.malId);
   222→  return { list, malIds, has, toggle, ready: true };
   223→}

   314→/**
   315→ * Persist progress locally immediately; mirror to the server. When
   316→ * `keepalive` is set (page unload / tab hide) the request is sent with
   317→ * keepalive so it survives navigation.
   318→ */
   319→export function saveProgressDetailed(args: SaveProgressArgs, opts?: { keepalive?: boolean }) {
   320→  if (!Number.isFinite(args.position) || args.position < 5) return;
   321→  // removed from Continue Watching — stale saves (heartbeats, beacons, the
   322→  // unmount flush of a just-left player) must not resurrect it. The only
   323→  // way back in is a deliberate watch (clearProgressTombstone on mount).
   324→  if (isProgressRemoved(args.malId)) return;
   325→  const completed = args.completed ?? (args.duration > 0 && args.position >= args.duration * 0.92);
   326→  const entry: DetailedProgress = {
   327→    pos: Math.floor(args.position),
   328→    dur: Math.floor(args.duration || 0),
   329→    completed,
   330→    lang: args.lang,
   331→    server: args.server,
   332→    title: args.title,
   333→    image: args.image,
   334→    total: args.total,
   335→    updatedAt: Date.now(),
   336→  };
   337→
   338→  // local (instant — bars/resume render without a round trip)
   339→  const map = readEpMap();
   340→  const key = String(args.malId);
   341→  map[key] = map[key] ?? {};
   342→  map[key][String(args.episode)] = entry;
   343→  writeJson(EP_PROGRESS_KEY, map);
   344→
   345→  // legacy "last watched episode" map (details badge, resume fallback)
   346→  try {
   347→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   348→    legacy[key] = { ep: args.episode, lang: args.lang, pos: entry.pos, updatedAt: entry.updatedAt };
   349→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   350→  } catch {
   351→    /* ignore */
   352→  }
   353→
   354→  try {
   355→    window.dispatchEvent(new CustomEvent("progress-changed"));
   356→  } catch {
   357→    /* ignore */
   358→  }
   359→
   360→  // server mirror (best effort, never blocks playback)
   361→  const userId = getUserId();
   362→  const body = JSON.stringify({
   363→    userId,
   364→    malId: args.malId,
   365→    episode: args.episode,
   366→    lang: args.lang,
   367→    server: args.server,
   368→    position: entry.pos,
   369→    duration: entry.dur,
   370→    completed,
   371→    title: args.title,
   372→    image: args.image,
   373→  });
   374→  try {
   375→    if (opts?.keepalive && "sendBeacon" in navigator) {
   376→      navigator.sendBeacon("/api/progress", new Blob([body], { type: "application/json" }));
   377→    } else {
   378→      // offline-safe: a failed save parks in the outbox and lands when the
   379→      // connection returns (same dedupe key — the latest position wins)
   380→      void outboxSend(`prog:${args.malId}:${args.episode}:${args.lang}`, "/api/progress", "POST", body);
   381→    }
   382→  } catch {
   383→    /* ignore */
   384→  }
   385→}

   387→/** Forget everything watched for one anime (Continue Watching "X" button). */
   388→export function removeProgress(malId: number) {
   389→  // tombstone FIRST — an in-flight save must not re-add this anime between
   390→  // here and the server DELETE ("deleted it but it came back" bug)
   391→  markProgressRemoved(malId);
   392→  const map = readEpMap();
   393→  delete map[String(malId)];
   394→  writeJson(EP_PROGRESS_KEY, map);
   395→  try {
   396→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   397→    delete legacy[String(malId)];
   398→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   399→  } catch {
   400→    /* ignore */
   401→  }
   402→  try {
   403→    window.dispatchEvent(new CustomEvent("progress-changed"));
   404→  } catch {
   405→    /* ignore */
   406→  }
   407→  // queueFirst: the DELETE is parked BEFORE it is attempted, so a lost
   408→  // request (subway dead zone, app killed mid-flight) is replayed by the
   409→  // next flush instead of leaving the server row alive — the old retry
   410→  // loop could never cover "app died before the request fired".
   411→  const userId = getUserId();
   412→  void outboxSend(
   413→    `progd:${malId}`,
   414→    `/api/progress?userId=${encodeURIComponent(userId)}&malId=${malId}`,
   415→    "DELETE",
   416→    undefined,
   417→    { queueFirst: true },
   418→  );
   419→}
   420→
   421→/**
   422→ * Reconciliation for the "deleted it but it came back" bug: re-fire the
   423→ * server DELETE for every anime this device has tombstoned in the last 48h.
   424→ * A removal whose DELETE request was lost (offline moment, app killed) gets
   425→ * healed the next time the Home rail loads; the local tombstone keeps the
   426→ * card hidden in the meantime. Idempotent — the server DELETE just deletes
   427→ * (and re-stamps) again.
   428→ */
   429→export function sweepRemovedProgress(): void {
   430→  const map = readRemovedMap();
   431→  const cutoff = Date.now() - 48 * 3600_000;
   432→  const fresh = Object.entries(map).filter(([, at]) => at >= cutoff);
   433→  if (!fresh.length) return;
   434→  const userId = getUserId();
   435→  for (const [malId, at] of fresh) {
   436→    const id = Number(malId);
   437→    if (!Number.isFinite(id)) continue;
   438→    // pace slightly: these are keepalive DELETEs, tiny and idempotent
   439→    window.setTimeout(() => {
   440→      try {
   441→        fetch(`/api/progress?userId=${encodeURIComponent(userId)}&malId=${id}`, {
   442→          method: "DELETE",
   443→          keepalive: true,
   444→        }).catch(() => undefined);
   445→      } catch {
   446→        /* best-effort */
   447→      }
   448→      void at;
   449→    }, 150 + Math.random() * 400);
   450→  }
   451→}

   453→/* ---------------- removal tombstones (local half) ----------------
   454→ * The server rejects saves for removed anime (RemovedProgress table), but
   455→ * the local map must obey too: a save that raced the delete would otherwise
   456→ * re-add bars/Continue entries the moment the server list is merged. */

   458→const REMOVED_KEY = "animetest123.progress-removed";

   460→function readRemovedMap(): Record<string, number> {
   461→  return readJson<Record<string, number>>(REMOVED_KEY, {});
   462→}

   464→function markProgressRemoved(malId: number) {
   465→  const map = readRemovedMap();
   466→  map[String(malId)] = Date.now();
   467→  writeJson(REMOVED_KEY, map);
   468→}

   470→function isProgressRemoved(malId: number): boolean {
   471→  return Boolean(readRemovedMap()[String(malId)]);

   474→/**
   475→ * Deliberate playback (WatchView mounted for this anime) — lift the removal
   476→ * tombstone locally + on the server so progress tracking resumes.
   477→ */
   478→export function clearProgressTombstone(malId: number) {
   479→  if (!isProgressRemoved(malId)) return;
   480→  const map = readRemovedMap();
   481→  delete map[String(malId)];
   482→  writeJson(REMOVED_KEY, map);
   483→  try {
   484→    void outboxSend(`tomb:${malId}`, "/api/progress", "POST", JSON.stringify({ userId: getUserId(), malId, resume: true }));
   485→  } catch {
   486→    /* ignore */
   487→  }
   488→}

   490→/**
   491→ * Server-side progress for one anime — used as a fallback so resume points
   492→ * and red bars survive cleared storage / a fresh browser.
   493→ */
   494→export async function fetchServerProgress(malId: number): Promise<Record<string, DetailedProgress>> {
   495→  try {
   496→    const res = await fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`);
   497→    if (!res.ok) return {};
   498→    const data = (await res.json()) as {
   499→      episodes: Array<{ episode: number; position: number; duration: number; completed: boolean; lang: string; server: string }>;
   500→    };
   501→    const out: Record<string, DetailedProgress> = {};
   502→    for (const e of data.episodes ?? []) {
   503→      out[String(e.episode)] = {
   504→        pos: e.position,
   505→        dur: e.duration,
   506→        completed: e.completed,
   507→        lang: e.lang === "dub" ? "dub" : "sub",
   508→        server: e.server,
   509→        updatedAt: Date.now(),
   510→      };
   511→    }
   512→    // merge into the local map (server rows are authoritative for these keys)
   513→    if (Object.keys(out).length) {
   514→      const map = readEpMap();
   515→      const key = String(malId);
   516→      map[key] = { ...(map[key] ?? {}), ...out };
   517→      writeJson(EP_PROGRESS_KEY, map);
   518→    }
   519→    return out;
   520→  } catch {
   521→    return {};
   522→  }
   523→}

   525→/* ---------------- Continue Watching ---------------- */
   526→
   527→/**
   528→ * Freshest in-progress episode per anime — server list merged with the
   529→ * instant local map (so the rail renders with zero latency and still shows
   530→ * entries when the server DB is cold).
   531→ */
   532→export function mergeContinueItems(serverItems: ContinueItem[]): ContinueItem[] {
   533→  let localAll: Record<string, Record<string, DetailedProgress>> = {};
   534→  try {
   535→    localAll = JSON.parse(window.localStorage.getItem(EP_PROGRESS_KEY) ?? "{}");
   536→  } catch {
   537→    localAll = {};
   538→  }
   539→  // the local removal tombstone is the user's expressed intent: a server row
   540→  // that slipped through the delete race (lost DELETE request, beacon that
   541→  // landed a heartbeat ahead of it on another lambda) must not resurrect the
   542→  // card in the UI. Deliberate playback lifts the tombstone (both halves).
   543→  const removed = readRemovedMap();
   544→
   545→  type Row = { item: ContinueItem; at: number };
   546→  const byAnime = new Map<number, Row>();
   547→
   548→  for (const it of serverItems) {
   549→    if (it.completed) continue;
   550→    if (removed[String(it.malId)]) continue; // deleted here — stays deleted
   551→    byAnime.set(it.malId, { item: it, at: Date.parse(it.updatedAt) });
   552→  }
   553→
   554→  for (const [malId, eps] of Object.entries(localAll)) {
   555→    // removed from Continue Watching — a save that raced the delete must not
   556→    // re-add the entry here either
   557→    if (readRemovedMap()[malId]) continue;
   558→    for (const [ep, e] of Object.entries(eps)) {
   559→      if (e.completed || (e.pos || 0) < 5) continue;
   560→      const malIdNum = Number(malId);
   561→      const prev = byAnime.get(malIdNum);
   562→      if (prev && prev.at >= e.updatedAt) continue;
   563→      const percent = e.dur > 0 ? Math.min(100, Math.round((e.pos / e.dur) * 100)) : 0;
   564→      byAnime.set(malIdNum, {
   565→        at: e.updatedAt,
   566→        item: {
   567→          malId: malIdNum,
   568→          title: e.title ?? prev?.item.title ?? "Unknown title",
   569→          image: e.image ?? prev?.item.image ?? null,
   570→          banner: prev?.item.banner ?? null,
   571→          totalEpisodes: e.total ?? prev?.item.totalEpisodes ?? 0,
   572→          type: prev?.item.type ?? null,
   573→          episode: Number(ep),
   574→          lang: e.lang,
   575→          server: e.server,
   576→          position: e.pos,
   577→          duration: e.dur,
   578→          percent,
   579→          completed: false,
   580→          updatedAt: new Date(e.updatedAt).toISOString(),
   581→        },
   582→      });
   583→    }
   584→  }
   585→
   586→  return [...byAnime.values()].sort((a, b) => b.at - a.at).map((r) => r.item);

   604→/* ---------------- Watch time (Top 10 Today) ---------------- *
   605→ *
   606→ * "Top 10 Today" ranks anime by how long people ACTUALLY watch them — the
   607→ * rail stays empty until real playback seconds exist. The player records
   608→ * wall-clock deltas while the video is playing; they accumulate locally
   609→ * (instant, offline-safe) and flush to the server so EVERY viewer's time
   610→ * lands in one global per-day ranking, Crunchyroll-style.
   611→ *
   612→ * Local shape: day -> malId -> { seconds, title, image, sent }.
   613→ * `sent` is the amount already ACKed by the server, so flushes POST only
   614→ * the un-sent remainder. Display merges server + local with max() — never
   615→ * a sum — so nothing is double-counted and offline time still shows.
   616→ */
   617→
   618→const WATCHTIME_KEY = "animetest123.watchtime";
   619→const WATCHTIME_EVENT = "watchtime-changed";
   620→const WATCHTIME_KEEP_DAYS = 7; // local buckets older than a week are dropped
   621→
   622→export interface LocalWatchEntry {
   623→  seconds: number;
   624→  title?: string;
   625→  image?: string | null;
   626→  sent?: number; // seconds already accepted by the server
   627→}
   628→
   629→type WatchTimeMap = Record<string, Record<string, LocalWatchEntry>>;
   630→
   631→/**
   632→ * The SHARED calendar day ("YYYY-MM-DD", UTC) — one global bucket, so every
   633→ * device in every timezone contributes to (and sees) the exact same
   634→ * "Top 10 Today" ranking, like Netflix's worldwide lists. Rolls over at
   635→ * UTC midnight for everyone.
   636→ */
   637→export function localDayKey(d = new Date()): string {
   638→  return d.toISOString().slice(0, 10);
   639→}
   640→
   641→function readWatchTime(): WatchTimeMap {
   642→  return readJson<WatchTimeMap>(WATCHTIME_KEY, {});
   643→}
   644→
   645→function writeWatchTime(map: WatchTimeMap) {
   646→  // prune: keep only the newest KEEP_DAYS day buckets
   647→  const days = Object.keys(map).sort();
   648→  for (const stale of days.slice(0, Math.max(0, days.length - WATCHTIME_KEEP_DAYS))) delete map[stale];
   649→  writeJson(WATCHTIME_KEY, map);
   650→}
   651→
   652→/**
   653→ * Add `delta` seconds of real playback to today's bucket for this anime and
   654→ * flush the un-sent remainder to the server (best effort, never throws).
   655→ * `keepalive` switches to sendBeacon for pagehide/unload flushes.
   656→ */
   657→export function recordWatchSeconds(
   658→  args: { malId: number; title?: string; image?: string | null; delta: number; day?: string },
   659→  opts?: { keepalive?: boolean },
   660→) {
   661→  const delta = Math.min(120, Math.max(0, Number(args.delta) || 0));
   662→  if (delta < 1 || !Number.isFinite(args.malId)) return;
   663→  const day = args.day ?? localDayKey(); // shared UTC day — one global ranking
   664→
   665→  let pending = 0;
   666→  try {
   667→    const map = readWatchTime();
   668→    const dayBucket = (map[day] = map[day] ?? {});
   669→    const entry = dayBucket[String(args.malId)] ?? { seconds: 0, sent: 0 };
   670→    entry.seconds += delta;
   671→    if (args.title) entry.title = args.title;
   672→    if (args.image !== undefined) entry.image = args.image;
   673→    dayBucket[String(args.malId)] = entry;
   674→    pending = entry.seconds - (entry.sent ?? 0);
   675→    if (pending >= 5) {
   676→      entry.sent = entry.seconds; // optimistic — reverted below if the send fails outright
   677→      writeWatchTime(map);
   678→    } else {
   679→      pending = 0;
   680→      writeWatchTime(map);
   681→    }
   682→  } catch {
   683→    return; // storage blocked — ranking is decorative, never fatal
   684→  }
   685→
   686→  if (pending < 5) {
   687→    // still notify listeners so Home can re-rank from the local bucket alone
   688→    try {
   689→      window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   690→    } catch {
   691→      /* ignore */
   692→    }
   693→    return;
   694→  }
   695→
   696→  const body = JSON.stringify({
   697→    userId: getUserId(),
   698→    malId: args.malId,
   699→    day,
   700→    delta: Math.round(pending),
   701→    title: args.title,
   702→    image: args.image,
   703→  });
   704→  const revert = () => {
   705→    try {
   706→      const map = readWatchTime();
   707→      const entry = map[day]?.[String(args.malId)];
   708→      if (entry) {
   709→        entry.sent = Math.max(0, (entry.sent ?? 0) - pending);
   710→        writeWatchTime(map);
   711→      }
   712→    } catch {
   713→      /* ignore */
   714→    }
   715→  };
   716→  try {
   717→    if (opts?.keepalive && "sendBeacon" in navigator) {
   718→      navigator.sendBeacon("/api/watchtime", new Blob([body], { type: "application/json" }));
   719→    } else {
   720→      void outboxSend(`wt:${day}:${args.malId}`, "/api/watchtime", "POST", body).then((res) => {
   721→        // null = the delta was parked in the offline outbox — treat as sent
   722→        // (the flush replays it); only a live server rejection reverts
   723→        if (res !== null && !res.ok) revert();
   724→      });
   725→    }
   726→  } catch {
   727→    revert();
   728→  }
   729→
   730→  try {
   731→    window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   732→  } catch {
   733→    /* ignore */
   734→  }
   735→}

   737→/** Subscribe to watch-time changes (local accrual + cross-tab). */
   738→export function subscribeWatchTime(onChange: () => void) {
   739→  window.addEventListener(WATCHTIME_EVENT, onChange);
   740→  window.addEventListener("storage", onChange);
   741→  return () => {
   742→    window.removeEventListener(WATCHTIME_EVENT, onChange);

   873→/**
   874→ * The day's most-watched anime: the server's global ranking merged with this
   875→ * device's instant local buckets (max, not sum — no double counting; local
   876→ * covers the un-flushed tail and offline/served-cold cases).
   877→ */
   878→export async function fetchWatchTimeTop(day: string, limit = 10): Promise<TopWatchEntry[]> {
   879→  let serverItems: TopWatchEntry[] = [];
   880→  try {
   881→    const res = await fetch(`/api/watchtime?day=${encodeURIComponent(day)}&limit=${limit}`);
   882→    if (res.ok) {
   883→      const data = (await res.json()) as { items?: TopWatchEntry[] };
   884→      serverItems = data.items ?? [];
   885→    }
   886→  } catch {
   887→    /* offline / cold server — local buckets only */
   888→  }
   889→
   890→  const byId = new Map<number, TopWatchEntry>();
   891→  for (const it of serverItems) {
   892→    if (Number.isFinite(it?.malId)) byId.set(it.malId, it);
   893→  }
   894→  try {
   895→    const local = readWatchTime()[day] ?? {};
   896→    for (const [key, entry] of Object.entries(local)) {
   897→      const malId = Number(key);
   898→      if (!Number.isFinite(malId)) continue;
   899→      const prev = byId.get(malId);
   900→      const seconds = Math.max(prev?.seconds ?? 0, entry.seconds || 0);
   901→      if (seconds < 5) continue; // a stray couple of seconds isn't "watched"
   902→      byId.set(malId, {
   903→        ...(prev ?? minimalLite(malId, entry.title || `Anime ${malId}`, entry.image ?? null)),
   904→        // local snapshots keep the rail informative when the catalog row is gone
   905→        title: prev?.title || entry.title || `Anime ${malId}`,
   906→        image: prev?.image ?? entry.image ?? null,
   907→        malId,
   908→        seconds,
   909→      });
   910→    }
   911→  } catch {
   912→    /* ignore */
   913→  }
   914→
   915→  return [...byId.values()].sort((a, b) => b.seconds - a.seconds).slice(0, limit);
   916→}

  1030→export function timeAgo(iso: string | number): string {
  1031→  const t = typeof iso === "number" ? iso : Date.parse(iso);
  1032→  const diff = Date.now() - t;
  1033→  const min = Math.floor(diff / 60_000);
  1034→  if (min < 1) return "just now";
  1035→  if (min < 60) return `${min}m ago`;
  1036→  const hr = Math.floor(min / 60);
  1037→  if (hr < 24) return `${hr}h ago`;
  1038→  const day = Math.floor(hr / 24);
  1039→  if (day < 30) return `${day}d ago`;
  1040→  return new Date(t).toLocaleDateString();
  1041→}

  1125→    if (window.localStorage.getItem(SETTINGS_ICON_KEY) != null) return;
  1126→    const raw = window.localStorage.getItem(SETTINGS_KEY);
  1127→    if (!raw) return;
  1128→    const parsed = JSON.parse(raw) as Partial<AppSettings>;
  1129→    if (parsed.iconDataUrl) {
  1130→      window.localStorage.setItem(SETTINGS_ICON_KEY, JSON.stringify(parsed.iconDataUrl));
  1131→      const legacy = { ...parsed, iconDataUrl: undefined };
  1132→      window.localStorage.setItem(SETTINGS_KEY, JSON.stringify(legacy));
  1133→    }
  1134→  } catch {
  1135→    /* best effort — the icon is cosmetic */
  1136→  }
  1137→}

  1140→  migrateLegacyIcon();

  1142→    ...DEFAULT_SETTINGS,
  1143→    ...readJson<Partial<AppSettings>>(SETTINGS_KEY, {}),

  1166→/** Reactive settings for components (re-renders on every change). */
  1167→export function useAppSettings(): AppSettings {
  1168→  const raw = useSyncExternalStore(subscribeSettings, getSettingsSnapshot, getSettingsServerSnapshot);
  1169→  // one corrupted settings key used to THROW during render in every consumer
  1170→  // (player included) and white-screen the whole app — guard like readJson
  1171→  let parsed: Partial<AppSettings> = {};
  1172→  if (raw && raw !== "{}") {
  1173→    try {
  1174→      parsed = JSON.parse(raw) as Partial<AppSettings>;
  1175→    } catch {
  1176→      parsed = {};
  1177→    }
  1178→  }
  1179→  const base = { ...DEFAULT_SETTINGS, ...parsed };
  1180→  // merge the separately-stored icon reactively (it changes with the same event)
  1181→  base.iconDataUrl = readJson<string | null>(SETTINGS_ICON_KEY, null) ?? null;
  1182→  return base;
  1183→}

  1185→export function updateAppSettings(patch: Partial<AppSettings>): AppSettings {
  1186→  // icon moves are handled in their own storage slot — keep them OUT of the
  1187→  // settings record so a big data URL can never blow the quota for a toggle
  1188→  const { iconDataUrl: iconPatch, ...rest } = patch;
  1189→  if (iconPatch !== undefined) {
  1190→    try {
  1191→      if (iconPatch == null) window.localStorage.removeItem(SETTINGS_ICON_KEY);
  1192→      else window.localStorage.setItem(SETTINGS_ICON_KEY, JSON.stringify(iconPatch));
  1193→    } catch {
  1194→      /* icon storage full/blocked — the icon is cosmetic; everything else
  1195→         below still applies and the UI still updates */
  1196→    }
  1197→  }
  1198→  const next = { ...readAppSettings(), ...rest, iconDataUrl: iconPatch !== undefined ? iconPatch : readAppSettings().iconDataUrl };
  1199→  writeJson(SETTINGS_KEY, { ...next, iconDataUrl: undefined });
  1200→  window.dispatchEvent(new Event(SETTINGS_EVENT));
  1201→  applySettingsToDocument(next);
  1202→  return next;
  1203→}

  1211→/**
  1212→ * Push the current settings into the document so EVERYTHING not explicitly
  1213→ * styled by a component follows along:
  1214→ *   - accent  -> CSS custom props --accent / --accent-deep (nav, buttons,
  1215→ *                switches, the settings screen itself)
  1216→ *   - subtitle look -> a `video::cue` rule (size / color / backdrop /
  1217→ *                outline) that native subtitle tracks pick up
  1218→ *   - app icon -> favicon + apple-touch-icon + a blob manifest so a fresh
  1219→ *                "Add to Home Screen" carries the custom icon (iOS applies
  1220→ *                the icon at add time; Android at install time — best effort
  1221→ *                by platform design)
  1222→ */
  1223→export function applySettingsToDocument(s: AppSettings): void {
  1224→  if (typeof document === "undefined") return;
  1225→
  1226→  const { color, deep } = accentColorOf(s.accent);
  1227→  document.documentElement.style.setProperty("--app-accent", color);
  1228→  document.documentElement.style.setProperty("--app-accent-deep", deep);
  1229→
  1230→  const px = (SUB_SIZES.find((x) => x.id === s.subSize) ?? SUB_SIZES[1]).px;
  1231→  const bg = Math.max(0, Math.min(100, s.subBg)) / 100;
  1232→  const shadows =
  1233→    s.subOutline === "none"
  1234→      ? "none"
  1235→      : s.subOutline === "outline"
  1236→        ? `-1px -1px 0 #000, 1px -1px 0 #000, -1px 1px 0 #000, 1px 1px 0 #000, 0 2px 3px rgba(0,0,0,.9)`
  1237→        : `0 1px 4px rgba(0,0,0,.9), 0 0 8px rgba(0,0,0,.55)`;
  1238→  const css = `video::cue { color: ${s.subColor}; background: rgba(0, 0, 0, ${bg}); text-shadow: ${shadows}; font-size: ${px}px; }`;
  1239→  let tag = document.getElementById("sub-cue-style") as HTMLStyleElement | null;
  1240→  if (!tag) {
  1241→    tag = document.createElement("style");
  1242→    tag.id = "sub-cue-style";
  1243→    document.head.appendChild(tag);
  1244→  }
  1245→  tag.textContent = css;
  1246→
  1247→  // custom app icon — swap every icon-ish link head tag
  1248→  try {
  1249→    if (s.iconDataUrl) {
  1250→      for (const [rel, sizes] of [
  1251→        ["icon", "32x32"],
  1252→        ["apple-touch-icon", "180x180"],
  1253→      ] as const) {
  1254→        let link = document.head.querySelector<HTMLLinkElement>(`link[rel="${rel}"]`);
  1255→        if (!link) {
  1256→          link = document.createElement("link");
  1257→          link.rel = rel;
  1258→          link.sizes = sizes;
  1259→          document.head.appendChild(link);
  1260→        }
  1261→        link.href = s.iconDataUrl;
  1262→      }
  1263→      // Android reads icons from the manifest at install time — build a
  1264→      // one-off blob manifest carrying the custom image at both sizes
  1265→      const manifest = {
  1266→        name: "Clyle",
  1267→        short_name: "Clyle",
  1268→        start_url: "/",
  1269→        display: "standalone",
  1270→        background_color: "#0a0a0a",
  1271→        theme_color: "#0a0a0a",
  1272→        icons: [
  1273→          { src: s.iconDataUrl, sizes: "192x192", type: "image/png" },
  1274→          { src: s.iconDataUrl, sizes: "512x512", type: "image/png" },
  1275→        ],
  1276→      };
  1277→      const url = URL.createObjectURL(new Blob([JSON.stringify(manifest)], { type: "application/manifest+json" }));
  1278→      const manLink = document.head.querySelector<HTMLLinkElement>('link[rel="manifest"]');
  1279→      if (manLink) manLink.href = url;
  1280→    }
  1281→  } catch {
  1282→    /* icon theming is cosmetic — never break the app over it */
  1283→  }
  1284→}

  1286→/**
  1287→ * Turn a user-chosen image into a home-screen icon data URL, DOWNSCALED on a
  1288→ * canvas so it can never blow the localStorage quota again (a full-res phone
  1289→ * photo as a data URL is multiple MB — that used to be stored raw INSIDE the
  1290→ * settings record, silently killing every settings write afterwards).
  1291→ * PNG keeps its transparency at 384px; everything else becomes a 512px JPEG.
  1292→ */
  1293→export async function fileToAppIconDataUrl(file: File): Promise<string> {
  1294→  if (!file.type.startsWith("image/")) throw new Error("Not an image");
  1295→  if (file.size > 25_000_000) throw new Error("Image too large (max ~25 MB)");
  1296→
  1297→  const url = URL.createObjectURL(file);
  1298→  let img: HTMLImageElement;
  1299→  try {
  1300→    img = await new Promise<HTMLImageElement>((res, rej) => {
  1301→      const el = new Image();
  1302→      el.onload = () => res(el);
  1303→      el.onerror = () => rej(new Error("Could not read image"));
  1304→      el.src = url;
  1305→    });
  1306→  } finally {
  1307→    setTimeout(() => URL.revokeObjectURL(url), 5_000);
  1308→  }
  1309→
  1310→  const png = file.type.includes("png");
  1311→  const S = png ? 384 : 512; // covers the 192/512 manifest slots
  1312→  const side = Math.min(img.naturalWidth || img.width, img.naturalHeight || img.height) || S;
  1313→  const canvas = document.createElement("canvas");
  1314→  canvas.width = S;
  1315→  canvas.height = S;
  1316→  const ctx = canvas.getContext("2d");
  1317→  if (!ctx) throw new Error("Canvas unavailable");
  1318→  const sw = side, sh = side;
  1319→  const sx = ((img.naturalWidth || img.width) - sw) / 2;
  1320→  const sy = ((img.naturalHeight || img.height) - sh) / 2;
  1321→  ctx.drawImage(img, sx, sy, sw, sh, 0, 0, S, S);
  1322→  return png ? canvas.toDataURL("image/png") : canvas.toDataURL("image/jpeg", 0.85);
  1323→}
