    37→    37→ * CAF's requestSession() rejection is often an EMPTY object (no code, no
    38→    38→ * message) — the real reason rides on the CastContext events instead
    39→    39→ * (SESSION_START_FAILED / CAST_STATE_CHANGED carry e.errorCode). We keep

    43→    43→let lastEventError = "";
    44→    44→const castTrail: string[] = [];
    45→    45→function noteTrail(s: string, code?: unknown) {
    46→    46→  const c = typeof code === "string" && code ? code : "";
    47→    47→  if (c) lastEventError = c;
    48→    48→  castTrail.push(c ? `${s}(${c})` : s);
    49→    49→  if (castTrail.length > 12) castTrail.shift();
    50→    50→}

    64→    64→/* ---- cast progress ownership --------------------------------------------
    65→    65→ * While casting, the TV is the playback surface — the phone's <video> is a
    66→    66→ * paused bystander. WatchView used to own progress saving, but its heartbeat
    67→    67→ * died the moment the user navigated away from the watch page (very common:
    68→    68→ * cast, then browse), and the pagehide/unmount flush read the STALE local
    69→    69→ * position and REGRESSED the saved TV progress. Now cast.ts owns it: every
    70→    70→ * load carries a progress descriptor and a module-level tracker keeps
    71→    71→ * writing Continue Watching for as long as the session lives — page
    72→    72→ * navigation inside the SPA can't touch it.
    73→    73→ * ----------------------------------------------------------------------- */
    74→    74→export interface CastProgressDescriptor {

    84→    84→   * resumed session (the TV might be on something else). */
    85→    85→  castTitle?: string;

   108→   108→function stopProgressTracker() {

   115→   115→/**
   116→   116→ * (Re)acquire the receiver's media client from a live session. THE fix for
   117→   117→ * the "remote controls dead while the TV plays" report: a page reload (or
   118→   118→ * PWA re-open) RESUMES the live cast session, but the media client was only
   119→   119→ * ever acquired inside the load paths — on a resumed session refs.media
   120→   120→ * stayed null, so position/duration never reached the remote and every
   121→   121→ * control (pause, ±10s, scrub) was a silent no-op while the TV played on.
   122→   122→ */
   123→   123→function wireMediaErrorListener(client: any) {
   124→   124→  client?.addUpdateListener?.(() => {
   125→   125→    const st = statusOf(client);
   126→   126→    if (st?.playerState === "IDLE" && st?.idleReason === "ERROR") {
   127→   127→      noteTrail("receiver-media-error");
   128→   128→      notifyMediaError("The TV couldn't play this stream — try another server or quality");
   129→   129→    }
   130→   130→  });
   131→   131→}

   133→   133→function attachRemoteMedia(session: any): boolean {
   134→   134→  try {
   135→   135→    const client = session?.getMediaSession?.();
   136→   136→    if (!client) return false;
   137→   137→    const isNew = refs.media !== client;
   138→   138→    refs.media = client;
   139→   139→    if (isNew) wireMediaErrorListener(client);
   140→   140→    return true;
   141→   141→  } catch {
   142→   142→    return false;
   143→   143→  }
   144→   144→}
   145→   145→
   146→   146→/**
   147→   147→ * Self-heal the remote from OUTSIDE a load: healthy sessions are untouched;
   148→   148→ * a casting session whose media client is missing/stale gets re-acquired
   149→   149→ * from CastContext. Called from the UI poll (fast) and the tracker (10s).
   150→   150→ */
   151→   151→export function castReattach(): boolean {
   152→   152→  if (currentState !== "casting") return false;
   153→   153→  if (refs.media && statusOf(refs.media)) return true; // healthy — nothing to do
   154→   154→  try {
   155→   155→    const live = cfw()?.CastContext.getInstance?.()?.getCurrentSession?.();
   156→   156→    if (live) refs.session = live;
   157→   157→    return attachRemoteMedia(refs.session);
   158→   158→  } catch {
   159→   159→    return false;
   160→   160→  }
   161→   161→}
   162→   162→
   163→   163→/**
   164→   164→ * Continue-Watching adoption for RESUMED sessions: the TV is already playing
   165→   165→ * (loaded by the previous page life) and this fresh page has no descriptor.
   166→   166→ * Once the receiver reports a real status — and its title matches OUR open
   167→   167→ * episode — adopt the caller's descriptor so TV progress keeps saving.
   168→   168→ */
   169→   169→export function castEnsureTracking(desc: CastProgressDescriptor): void {
   170→   170→  if (currentState !== "casting" || progressDesc || !desc) return;
   171→   171→  try {
   172→   172→    const st = statusOf(refs.media);
   173→   173→    if (!st) return; // no confirmed receiver state — never guess
   174→   174→    const receiverTitle = st?.media?.metadata?.title;
   175→   175→    if (desc.castTitle && receiverTitle && receiverTitle !== desc.castTitle) return;
   176→   176→    progressDesc = desc;
   177→   177→    startProgressTracker();
   178→   178→  } catch {
   179→   179→    /* ignore */
   180→   180→  }
   181→   181→}
   182→   182→
   183→   183→function startProgressTracker() {
   184→   184→  if (typeof window === "undefined" || demoHook) return;
   185→   185→  if (progressTimer != null) return; // already watching this session
   186→   186→  trackerLastPos = castPosition();
   187→   187→  trackerActiveAt = Date.now();
   188→   188→  progressTimer = window.setInterval(() => {
   189→   189→    if (currentState !== "casting") return;
   190→   190→    // ---- self-heal + sync watchdog (covers RESUMED sessions) ----
   191→   191→    // a page reload rejoins the live session with a null/stale media client;
   192→   192→    // re-acquire it, and if the receiver never reports a status, say so once
   193→   193→    if (!refs.media || !statusOf(refs.media)) {
   194→   194→      try {
   195→   195→        const live = cfw()?.CastContext.getInstance?.()?.getCurrentSession?.();
   196→   196→        if (live) {
   197→   197→          refs.session = live;
   198→   198→          attachRemoteMedia(live);
   199→   199→        }
   200→   200→      } catch {
   201→   201→        /* ignore */
   202→   202→      }
   203→   203→      if (noStatusSince == null) noStatusSince = Date.now();
   204→   204→      else if (!syncToastShown && Date.now() - noStatusSince > 12_000) {
   205→   205→        syncToastShown = true;
   206→   206→        noteTrail("no-status");
   207→   207→        notifyMediaError("Can't sync with the TV's playback — Stop casting, then cast this episode again");
   208→   208→      }
   209→   209→    } else {
   210→   210→      noStatusSince = null;
   211→   211→      syncToastShown = false;
   212→   212→    }
   213→   213→    if (!progressDesc) return; // resumed session not yet adopted — controls only
   214→   214→    const pos = castPosition();
   215→   215→    const now = Date.now();
   216→   216→    if (pos == null || pos === trackerLastPos) {
   217→   217→      // paused, buffering or unreachable — wall time between these ticks
   218→   218→      // must not accrue as watch time
   219→   219→      trackerLastPos = pos;
   220→   220→      trackerActiveAt = now;
   221→   221→    } else {
   222→   222→      const delta = Math.min(120, (now - trackerActiveAt) / 1000);
   223→   223→      trackerLastPos = pos;
   224→   224→      trackerActiveAt = now;
   225→   225→      if (delta >= 1) {
   226→   226→        recordWatchSeconds({ malId: progressDesc.malId, title: progressDesc.title ?? "", image: progressDesc.image ?? null, delta });
   227→   227→      }
   228→   228→      saveProgressDetailed({
   229→   229→        malId: progressDesc.malId,
   230→   230→        episode: progressDesc.episode,
   231→   231→        lang: progressDesc.lang,
   232→   232→        server: progressDesc.server,
   233→   233→        position: pos,
   234→   234→        duration: castDuration() || 0,
   235→   235→        title: progressDesc.title,
   236→   236→        image: progressDesc.image,
   237→   237→        total: progressDesc.total,
   238→   238→      });
   239→   239→    }
   240→   240→    // ---- auto next episode (mirrors the in-player ended handler) ----
   241→   241→    if (progressDesc && autoNextHandler) {
   242→   242→      const dur = castDuration() ?? 0;
   243→   243→      const st = statusOf(refs.media);
   244→   244→      const finished = st?.playerState === "IDLE" && st?.idleReason === "FINISHED";
   245→   245→      const atEnd = dur > 0 && pos != null && pos >= dur - 2;
   246→   246→      if ((finished || atEnd) && autoNextFiredEpoch !== autoNextEpoch) {
   247→   247→        if (readAppSettings().autoNext) {
   248→   248→          autoNextFiredEpoch = autoNextEpoch;
   249→   249→          try {
   250→   250→            autoNextHandler();
   251→   251→          } catch {
   252→   252→            /* navigation is best-effort */
   253→   253→          }
   254→   254→        }
   255→   255→      }
   256→   256→    }
   257→   257→  }, 10_000);
   258→   258→}

   276→   276→/* ---- stuck-load watchdog ------------------------------------------------
   277→   277→ * loadMedia() resolving only means the receiver ACCEPTED the request. If
   278→   278→ * the media URL can't actually be fetched by the TV (CDN blocks the
   279→   279→ * proxy's datacenter IP, dead gateway), the receiver sits on its loading
   280→   280→ * screen forever while the remote shows dead 0:00/0:00 controls — exactly
   281→   281→ * the owner's cast report (2026-10-09). Watch the status briefly after
   282→   282→ * each load: no duration and no position progress within 12s = the stream
   283→   283→ * never loaded — say so with an actionable message instead of leaving
   284→   284→ * silent dead controls. Disarms itself the moment the media is real.
   285→   285→ * ----------------------------------------------------------------------- */
   286→   286→let stuckWatchdog: number | null = null;
   287→   287→function disarmStuckWatchdog() {
   288→   288→  if (stuckWatchdog != null) {
   289→   289→    window.clearInterval(stuckWatchdog);
   290→   290→    stuckWatchdog = null;
   291→   291→  }
   292→   292→}
   293→   293→function armStuckLoadWatchdog() {
   294→   294→  if (typeof window === "undefined" || demoHook) return;
   295→   295→  disarmStuckWatchdog();
   296→   296→  const startedAt = Date.now();
   297→   297→  stuckWatchdog = window.setInterval(() => {
   298→   298→    if (currentState !== "casting") {
   299→   299→      disarmStuckWatchdog();
   300→   300→      return;
   301→   301→    }
   302→   302→    const dur = castDuration();
   303→   303→    const pos = castPosition();
   304→   304→    if ((dur != null && dur > 0) || (pos != null && pos > 0)) {
   305→   305→      disarmStuckWatchdog(); // media is real and moving — normal remote operation
   306→   306→      return;
   307→   307→    }
   308→   308→    if (Date.now() - startedAt > 12_000) {
   309→   309→      disarmStuckWatchdog();
   310→   310→      noteTrail("stuck-loading");
   311→   311→      notifyMediaError("The TV can't load this stream — try Server 2, or stop casting and cast again");
   312→   312→    }
   313→   313→  }, 1_000);
   314→   314→}

   316→   316→/**
   317→   317→ * Squeeze every byte of meaning out of a cast failure. CAF rejections come
   318→   318→ * in many shapes: undefined, empty object, {errorCode}, {code, description},
   319→   319→ * DOMException, Error... Serialize whichever fields actually exist.
   320→   320→ */
   321→   321→function describeError(err: unknown): string {
   322→   322→  if (err == null) return "";
   323→   323→  if (typeof err === "string") return err.trim();
   324→   324→  const e = err as Record<string, unknown>;
   325→   325→  const parts: string[] = [];
   326→   326→  for (const key of ["message", "description", "errorCode", "code", "detail", "reason", "name"]) {
   327→   327→    const v = e[key];
   328→   328→    if (typeof v === "string" && v.trim()) parts.push(`${key}:${v.trim().replace(/_/g, " ")}`);
   329→   329→    else if (typeof v === "number") parts.push(`${key}:${v}`);
   330→   330→  }
   331→   331→  // last resort: every own property, JSON-shaped (truncated)
   332→   332→  if (parts.length === 0) {
   333→   333→    try {
   334→   334→      const keys = Object.getOwnPropertyNames(e);
   335→   335→      const snap = JSON.stringify(e, keys as never);
   336→   336→      if (snap && snap !== "{}" && snap !== '""') parts.push(snap.slice(0, 140));
   337→   337→      else if (keys.length > 0) parts.push(keys.join(","));
   338→   338→    } catch {
   339→   339→      /* ignore */
   340→   340→    }
   341→   341→  }
   342→   342→  return parts.join(" · ").slice(0, 180);
   343→   343→}

   370→   370→/**
   371→   371→ * Inject the CAF sender SDK once and configure the cast context with the
   372→   372→ * Default Media Receiver. Safe to call from every player mount.
   373→   373→ */
   374→   374→export function ensureCastSdk(): void {
   375→   375→  if (typeof window === "undefined" || sdkReady || sdkLoading) return;
   376→   376→
   377→   377→  // SDK already there (another component loaded it)
   378→   378→  if (cfw() && (window as any).chrome?.cast?.media) {
   379→   379→    configureContext();
   380→   380→    return;
   381→   381→  }
   382→   382→
   383→   383→  sdkLoading = true;
   384→   384→  const done = (ok: boolean) => {
   385→   385→    sdkLoading = false;
   386→   386→    if (!ok) {
   387→   387→      setState("unavailable");
   388→   388→      return;
   389→   389→    }
   390→   390→    configureContext();
   391→   391→  };
   392→   392→
   393→   393→  (window as any).__onGCastApiAvailable = (available: boolean) => done(available !== false);
   394→   394→
   395→   395→  const existing = document.querySelector<HTMLScriptElement>(`script[src="${CAST_SDK_SRC}"]`);
   396→   396→  if (existing) return; // callback above will fire
   397→   397→  const script = document.createElement("script");
   398→   398→  script.src = CAST_SDK_SRC;
   399→   399→  script.async = true;
   400→   400→  script.onerror = () => done(false);
   401→   401→  document.head.appendChild(script);
   402→   402→
   403→   403→  // belt & braces: some builds never call __onGCastApiAvailable
   404→   404→  window.setTimeout(() => {
   405→   405→    if (!sdkReady && cfw() && (window as any).chrome?.cast?.media) configureContext();
   406→   406→  }, 6000);
   407→   407→}
   408→   408→
   409→   409→function configureContext() {
   410→   410→  try {
   411→   411→    const framework = cfw();
   412→   412→    if (!framework || !(window as any).chrome?.cast?.media) {
   413→   413→      setState("unavailable");
   414→   414→      return;
   415→   415→    }
   416→   416→    const context = framework.CastContext.getInstance();
   417→   417→    context.setOptions({
   418→   418→      receiverApplicationId: (window as any).chrome.cast.media.DEFAULT_MEDIA_RECEIVER_APP_ID,
   419→   419→      autoJoinPolicy: (window as any).chrome.cast.AutoJoinPolicy.ORIGIN_SCOPED,
   420→   420→    });
   421→   421→    context.addEventListener(
   422→   422→      framework.CastContextEventType.SESSION_STATE_CHANGED,
   423→   423→      (e: any) => {
   424→   424→        const s = e?.sessionState;
   425→   425→        const S = framework.SessionState ?? {};
   426→   426→        if (s === S.SESSION_STARTED || s === S.SESSION_RESUMED) {
   427→   427→          noteTrail("session-started");
   428→   428→          refs.session = e.session ?? context.getCurrentSession();
   429→   429→          // RESUMED (page reload / PWA re-open while the TV plays): adopt the
   430→   430→          // receiver's CURRENT media client right here — waiting for a load
   431→   431→          // left the remote dead at 0:00/0:00 with no-op controls
   432→   432→          attachRemoteMedia(refs.session);
   433→   433→          startProgressTracker();
   434→   434→          setState("casting");
   435→   435→        } else if (s === S.SESSION_START_FAILED) {
   436→   436→          // THIS is where the real reason hides when requestSession rejects empty
   437→   437→          noteTrail("session-failed", e?.errorCode);
   438→   438→          refs.session = null;
   439→   439→          refs.media = null;
   440→   440→          setState("idle");
   441→   441→        } else if (s === S.SESSION_ENDED || s === S.SESSION_END_FAILED) {
   442→   442→          noteTrail("session-ended", s === S.SESSION_END_FAILED ? e?.errorCode : undefined);
   443→   443→          refs.session = null;
   444→   444→          refs.media = null;
   445→   445→          progressDesc = null;
   446→   446→          captionsTrackId = null;
   447→   447→          captionsActive = false;
   448→   448→          noStatusSince = null;
   449→   449→          syncToastShown = false;
   450→   450→          stopProgressTracker();
   451→   451→          setState("idle");
   452→   452→        }
   453→   453→      },
   454→   454→    );
   455→   455→    context.addEventListener(framework.CastContextEventType.CAST_STATE_CHANGED, (e: any) => {
   456→   456→      noteTrail(`cast:${e?.castState ?? "?"}`, e?.errorCode);
   457→   457→    });
   458→   458→    // a previous page load may still hold a live session
   459→   459→    if (context.getCurrentSession()) {
   460→   460→      refs.session = context.getCurrentSession();
   461→   461→      setState("casting");
   462→   462→    } else {
   463→   463→      setState("idle");
   464→   464→    }
   465→   465→    sdkReady = true;
   466→   466→  } catch {
   467→   467→    setState("unavailable");
   468→   468→  }
   469→   469→}

   572→   572→const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
   573→   573→
   574→   574→/**
   575→   575→ * Request a (possibly resumed) session. Rejects on user cancel; returns null
   576→   576→ * when the retry after a transient failure also failed.
   577→   577→ */
   578→   578→async function requestSession(context: any): Promise<any | null> {
   579→   579→  try {
   580→   580→    const session = await context.requestSession(); // rejects on cancel/failure
   581→   581→    return session ?? context.getCurrentSession();
   582→   582→  } catch (err) {
   583→   583→    const text = describeError(err).toLowerCase();
   584→   584→    // user closed the chooser — do not retry, just report
   585→   585→    if (text.includes("cancel")) throw err;
   586→   586→    // first connect attempts fail transiently on flaky mesh/2.4GHz links;
   587→   587→    // one quiet retry catches a good chunk of those
   588→   588→    noteTrail("retry");
   589→   589→    await sleep(1200);
   590→   590→    try {
   591→   591→      const session = await context.requestSession();
   592→   592→      return session ?? context.getCurrentSession();
   593→   593→    } catch {
   594→   594→      return null;
   595→   595→    }
   596→   596→  }
   597→   597→}
   598→   598→
   599→   599→/**
   600→   600→ * The session wrapper the SDK handed us earlier can go STALE while the
   601→   601→ * module-level ref survives (phone locked, app backgrounded, Wi-Fi blip —
   602→   602→ * the receiver drops the socket, CAF tears the session down internally).
   603→   603→ * Calling loadMedia() on the stale wrapper then explodes deep inside the SDK
   604→   604→ * with e.g. "Cannot read properties of undefined (reading
   605→   605→ * 'addUpdateListener')". Re-hydrate from CastContext before every reuse:
   606→   606→ * it returns null/undefined once the session is really gone.
   607→   607→ */
   608→   608→function refreshLiveSession(context: any): boolean {
   609→   609→  if (!refs.session) return false;
   610→   610→  try {
   611→   611→    const live = context.getCurrentSession?.();
   612→   612→    if (live) {
   613→   613→      refs.session = live;
   614→   614→      return true;
   615→   615→    }
   616→   616→  } catch {
   617→   617→    /* fall through — treat as dead */
   618→   618→  }
   619→   619→  refs.session = null;
   620→   620→  refs.media = null;
   621→   621→  setState("idle");
   622→   622→  return false;
   623→   623→}
   624→   624→
   625→   625→/**
   626→   626→ * CAF's CastSession.loadMedia() resolves VOID — the controllable object is
   627→   627→ * the RemoteMediaClient handed out by session.getMediaSession(). Treating
   628→   628→ * the resolution itself as the media object is what produced the owner's
   629→   629→ * "Cannot read properties of undefined (reading 'addUpdateListener')"
   630→   630→ * cast failure (the media HAD loaded on the TV; the sender then crashed on
   631→   631→ * undefined). Legacy v2 callers get the Media object back, so accept both
   632→   632→ * shapes: prefer the session's client, fall back to whatever loadMedia
   633→   633→ * returned.
   634→   634→ */
   635→   635→function resolveClient(session: any, loaded: any): any {
   636→   636→  try {
   637→   637→    const viaSession = session?.getMediaSession?.();
   638→   638→    if (viaSession) return viaSession;
   639→   639→  } catch {
   640→   640→    /* fall through to the loaded shape */
   641→   641→  }
   642→   642→  return loaded ?? null;
   643→   643→}
   644→   644→
   645→   645→/** MediaStatus accessor for both API shapes (RemoteMediaClient.getMediaStatus()
   646→   646→ * vs the v2 Media object's .status property). */
   647→   647→function statusOf(media: any): any {
   648→   648→  if (!media) return null;
   649→   649→  try {
   650→   650→    const st = media.getMediaStatus?.();
   651→   651→    if (st) return st;
   652→   652→  } catch {
   653→   653→    /* fall through */
   654→   654→  }
   655→   655→  return media.status ?? null;
   656→   656→}

   658→   658→/** Ask the user to pick a device, then load the episode. Returns success. */
   659→   659→export async function castLoad(req: CastMediaRequest): Promise<{ ok: boolean; error?: string }> {
   660→   660→  try {
   661→   661→    const framework = cfw();
   662→   662→    const chromeCast = (window as any).chrome?.cast;
   663→   663→    if (!framework || !chromeCast) return { ok: false, error: "Cast is not available in this browser" };
   664→   664→
   665→   665→    setState("connecting");
   666→   666→    const context = framework.CastContext.getInstance();
   667→   667→
   668→   668→    if (refs.session) refreshLiveSession(context);
   669→   669→    if (!refs.session) {
   670→   670→      refs.session = await requestSession(context);
   671→   671→    }
   672→   672→    if (!refs.session) throw new Error("No Chromecast selected");
   673→   673→
   674→   674→    progressDesc = req.progress ?? null;
   675→   675→    autoNextEpoch++;
   676→   676→    const loaded = await loadMediaWithRecovery(context, chromeCast, req);
   677→   677→    if (!attachRemoteMedia(refs.session)) {
   678→   678→      // loadMedia resolving only proves the REQUEST was accepted (CAF resolves
   679→   679→      // VOID) — the controllable client rides on the session
   680→   680→      refs.media = resolveClient(refs.session, loaded);
   681→   681→      wireMediaErrorListener(refs.media);
   682→   682→    }
   683→   683→    armStuckLoadWatchdog();
   684→   684→    startProgressTracker();
   685→   685→    setState("casting");
   686→   686→    return { ok: true };
   687→   687→  } catch (err: unknown) {
   688→   688→    // CAF errors arrive in mixed shapes (often EMPTY) — describeError pulls
   689→   689→    // whatever exists, and lastEventError holds the reason the context events
   690→   690→    // saw (the promise rejection alone is frequently {} on session failures).
   691→   691→    const detail = [describeError(err), lastEventError ? `event:${lastEventError}` : ""]
   692→   692→      .filter(Boolean)
   693→   693→      .join(" · ");
   694→   694→    const flat = detail.toLowerCase();
   695→   695→    console.warn("[cast] session/media failure:", err, "| eventError:", lastEventError, "| trail:", castTrail.join(" > "));
   696→   696→    let msg: string;
   697→   697→    if (flat.includes("cancel")) msg = "Cast cancelled";
   698→   698→    else if (flat.includes("no device") || flat.includes("not found") || flat.includes("no compatible")) {
   699→   699→      msg = "No cast device found — make sure your TV/Chromecast is on the same Wi-Fi, then try again";
   700→   700→    } else if (flat.includes("session error") || flat.includes("timeout") || flat.includes("unreachable")) {
   701→   701→      msg = `Couldn't reach the TV (${detail}) — your phone finds the TV but can't open a connection. Router AP/client isolation or a guest network usually causes this`;
   702→   702→    } else if (flat.includes("load") || flat.includes("media") || flat.includes("invalid")) {
   703→   703→      msg = "The TV couldn't load this stream — try another server for this episode";
   704→   704→    } else {
   705→   705→      // unknown failure — show everything we captured so it can be reported
   706→   706→      msg = `Cast failed (${detail || "no details"}) — tell me this text to fix it`;
   707→   707→    }
   708→   708→    if (!refs.session) setState("idle");
   709→   709→    return { ok: false, error: msg };
   710→   710→  }
   711→   711→}
   712→   712→
   713→   713→/**
   714→   714→ * loadMedia with one stale-session recovery: the SDK throws opaque TypeErrors
   715→   715→ * (addUpdateListener of undefined) when the cached CastSession wrapper outlived
   716→   716→ * its underlying connection. Discard the dead wrapper, re-ask for a session,
   717→   717→ * retry once — the user just sees the cast succeed.
   718→   718→ */
   719→   719→async function loadMediaWithRecovery(context: any, chromeCast: any, req: CastMediaRequest): Promise<any> {
   720→   720→  const build = () => buildLoadRequest(chromeCast, req);
   721→   721→  try {
   722→   722→    return await refs.session.loadMedia(build());
   723→   723→  } catch (err) {
   724→   724→    // no live session underneath — drop it and retry once with a fresh one
   725→   725→    noteTrail("stale-session");
   726→   726→    refs.session = null;
   727→   727→    refs.media = null;
   728→   728→    setState("idle");
   729→   729→    const fresh = await requestSession(context);
   730→   730→    if (!fresh) throw err;
   731→   731→    refs.session = fresh;
   732→   732→    return await refs.session.loadMedia(build());
   733→   733→  }
   734→   734→}
   735→   735→
   736→   736→function buildLoadRequest(chromeCast: any, req: CastMediaRequest): any {
   737→   737→  // the Default Media Receiver sniffs better with the honest type: our SP
   738→   738→  // sources are progressive MP4s, everything else is an HLS master
   739→   739→  const type = /\.m3u8($|\?)/i.test(req.src) ? "application/x-mpegurl" : "video/mp4";
   740→   740→  const mediaInfo = new chromeCast.media.MediaInfo(req.src, type);
   741→   741→  const metadata = new chromeCast.media.GenericMediaMetadata();
   742→   742→  metadata.title = req.title;
   743→   743→  if (req.subtitle) metadata.subtitle = req.subtitle;
   744→   744→  if (req.imageUrl) metadata.images = [new chromeCast.Image(req.imageUrl)];
   745→   745→  mediaInfo.metadata = metadata;
   746→   746→  mediaInfo.streamType = chromeCast.media.StreamType.BUFFERED;
   747→   747→  const request = new chromeCast.media.LoadRequest(mediaInfo);
   748→   748→  request.autoplay = true;
   749→   749→  request.currentTime = Math.max(0, req.startTime ?? 0);
   750→   750→  attachCaptions(chromeCast, mediaInfo, request, req.captions, req.captionsOn ?? true);
   751→   751→  return request;
   752→   752→}

   754→   754→/** Load another episode into the live cast session (no re-prompt). */
   755→   755→export async function castLoadNext(req: CastMediaRequest): Promise<boolean> {
   756→   756→  if (demoHook) return demoHook.loadNext();
   757→   757→  try {
   758→   758→    const framework = cfw();
   759→   759→    const chromeCast = (window as any).chrome?.cast;
   760→   760→    if (!framework || !chromeCast) return false;
   761→   761→    const context = framework.CastContext.getInstance();
   762→   762→    // a session that died since the last episode (phone slept, Wi-Fi blip)
   763→   763→    // must not explode inside loadMedia — verify it is still alive first
   764→   764→    if (!refreshLiveSession(context) || !refs.session) return false;
   765→   765→    progressDesc = req.progress ?? progressDesc; // keep the old descriptor when none is given
   766→   766→    autoNextEpoch++;
   767→   767→    // loadMedia resolves VOID in CAF — prefer the session's client
   768→   768→    const loadedNext = await refs.session.loadMedia(buildLoadRequest(chromeCast, req));
   769→   769→    if (!attachRemoteMedia(refs.session)) {
   770→   770→      refs.media = resolveClient(refs.session, loadedNext);
   771→   771→      wireMediaErrorListener(refs.media);
   772→   772→    }
   773→   773→    armStuckLoadWatchdog();
   774→   774→    startProgressTracker();
   775→   775→    return true;
   776→   776→  } catch {
   777→   777→    // the session died mid-series — drop it so the next Cast tap re-prompts
   778→   778→    refs.session = null;
   779→   779→    refs.media = null;
   780→   780→    setState("idle");
   781→   781→    return false;
   782→   782→  }
   783→   783→}

   794→   794→        malId: progressDesc.malId,

   796→   796→        lang: progressDesc.lang,
   797→   797→        server: progressDesc.server,
   798→   798→        position: pos,
   799→   799→        duration: castDuration() || 0,
   800→   800→        title: progressDesc.title,
   801→   801→        image: progressDesc.image,
   802→   802→        total: progressDesc.total,
   803→   803→      });
   804→   804→    }
   805→   805→  }
   806→   806→  progressDesc = null;
   807→   807→  captionsTrackId = null;
   808→   808→  captionsActive = false;
   809→   809→  noStatusSince = null;
   810→   810→  syncToastShown = false;
   811→   811→  stopProgressTracker();
   812→   812→  try {
   813→   813→    refs.media = null;
   814→   814→    refs.session?.endCurrentSession?.(true);
   815→   815→    cfw()?.CastContext.getInstance().endCurrentSession(true);
   816→   816→  } catch {
   817→   817→    /* ignore */
   818→   818→  }
   819→   819→  refs.session = null;
   820→   820→  setState("idle");
   821→   821→}
