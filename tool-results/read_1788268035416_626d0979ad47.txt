    25→function fmtTime(s: number): string {
    26→  if (!Number.isFinite(s) || s < 0) return "0:00";
    27→  const h = Math.floor(s / 3600);
    28→  const m = Math.floor((s % 3600) / 60);
    29→  const sec = Math.floor(s % 60);
    30→  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${String(sec).padStart(2, "0")}` : `${m}:${String(sec).padStart(2, "0")}`;
    31→}

   468→      pendingSeekRef.current = null;

   495→          setStreamError(true);
   496→        }
   497→      });

   499→      // Safari native HLS
   500→      video.src = proxyUrl(stream.src, "m3u8");
   501→      video.addEventListener("loadedmetadata", startPlayback, { once: true });
   502→    } else if (!destroyed) {
   503→      setStreamError(true);
   504→    }
   505→
   506→    return () => {
   507→      destroyed = true;
   508→      hlsRef.current?.destroy();
   509→      hlsRef.current = null;
   510→      video.removeAttribute("src");
   511→      video.load();
   512→    };
   513→  }, [stream, iframeFallback]);
