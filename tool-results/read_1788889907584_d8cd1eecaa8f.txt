    31→function fmtTime(s: number): string {
    32→  if (!Number.isFinite(s) || s < 0) return "0:00";
    33→  const h = Math.floor(s / 3600);
    34→  const m = Math.floor((s % 3600) / 60);
    35→  const sec = Math.floor(s % 60);
    36→  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${String(sec).padStart(2, "0")}` : `${m}:${String(sec).padStart(2, "0")}`;
    37→}
