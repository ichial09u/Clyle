     6→ * Chromecast sender via Google's Cast Application Framework (CAF).

    36→/* ---- failure forensics ------------------------------------------------
    37→ * CAF's requestSession() rejection is often an EMPTY object (no code, no
    38→ * message) — the real reason rides on the CastContext events instead
    39→ * (SESSION_START_FAILED / CAST_STATE_CHANGED carry e.errorCode). We keep
    40→ * both the last event errorCode and a short trail so a failed cast can be
    41→ * diagnosed from the toast alone, without a debugger.
    42→ * -------------------------------------------------------------------- */
    43→let lastEventError = "";
    44→const castTrail: string[] = [];
    45→function noteTrail(s: string, code?: unknown) {
    46→  const c = typeof code === "string" && code ? code : "";
    47→  if (c) lastEventError = c;
    48→  castTrail.push(c ? `${s}(${c})` : s);
    49→  if (castTrail.length > 12) castTrail.shift();
    50→}

    64→/* ---- cast progress ownership --------------------------------------------
    65→ * While casting, the TV is the playback surface — the phone's <video> is a
    66→ * paused bystander. WatchView used to own progress saving, but its heartbeat
    67→ * died the moment the user navigated away from the watch page (very common:
    68→ * cast, then browse), and the pagehide/unmount flush read the STALE local
    69→ * position and REGRESSED the saved TV progress. Now cast.ts owns it: every
    70→ * load carries a progress descriptor and a module-level tracker keeps
    71→ * writing Continue Watching for as long as the session lives — page
    72→ * navigation inside the SPA can't touch it.
    73→ * ----------------------------------------------------------------------- */
    74→export interface CastProgressDescriptor {
    75→  malId: number;
    76→  episode: number;
    77→  lang: "sub" | "dub";
    78→  server: string;
    79→  title?: string;
    80→  image?: string | null;
    81→  total?: number;
    82→  /** The exact title string we load with — used to verify the receiver is
    83→   * still playing OUR episode before adopting progress tracking on a
    84→   * resumed session (the TV might be on something else). */
    85→  castTitle?: string;
    86→}

   108→function stopProgressTracker() {

   115→/**
   116→ * (Re)acquire the receiver's media client from a live session. THE fix for
   117→ * the "remote controls dead while the TV plays" report: a page reload (or
   118→ * PWA re-open) RESUMES the live cast session, but the media client was only
   119→ * ever acquired inside the load paths — on a resumed session refs.media
   120→ * stayed null, so position/duration never reached the remote and every
   121→ * control (pause, ±10s, scrub) was a silent no-op while the TV played on.
   122→ */
   123→function wireMediaErrorListener(client: any) {
   124→  client?.addUpdateListener?.(() => {
   125→    const st = statusOf(client);
   126→    if (st?.playerState === "IDLE" && st?.idleReason === "ERROR") {
   127→      noteTrail("receiver-media-error");
   128→      notifyMediaError("The TV couldn't play this stream — try another server or quality");
   129→    }
   130→  });
   131→}

   133→function attachRemoteMedia(session: any): boolean {
   134→  try {
   135→    const client = session?.getMediaSession?.();
   136→    if (!client) return false;
   137→    const isNew = refs.media !== client;
   138→    refs.media = client;
   139→    if (isNew) wireMediaErrorListener(client);
   140→    return true;
   141→  } catch {
   142→    return false;
   143→  }
   144→}
   145→
   146→/**
   147→ * Self-heal the remote from OUTSIDE a load: healthy sessions are untouched;
   148→ * a casting session whose media client is missing/stale gets re-acquired
   149→ * from CastContext. Called from the UI poll (fast) and the tracker (10s).
   150→ */
   151→export function castReattach(): boolean {
   152→  if (currentState !== "casting") return false;
   153→  if (refs.media && statusOf(refs.media)) return true; // healthy — nothing to do
   154→  try {
   155→    const live = cfw()?.CastContext.getInstance?.()?.getCurrentSession?.();
   156→    if (live) refs.session = live;
   157→    return attachRemoteMedia(refs.session);
   158→  } catch {
   159→    return false;
   160→  }
   161→}
   162→
   163→/**
   164→ * Continue-Watching adoption for RESUMED sessions: the TV is already playing
   165→ * (loaded by the previous page life) and this fresh page has no descriptor.
   166→ * Once the receiver reports a real status — and its title matches OUR open
   167→ * episode — adopt the caller's descriptor so TV progress keeps saving.
   168→ */
   169→export function castEnsureTracking(desc: CastProgressDescriptor): void {
   170→  if (currentState !== "casting" || progressDesc || !desc) return;
   171→  try {
   172→    const st = statusOf(refs.media);
   173→    if (!st) return; // no confirmed receiver state — never guess
   174→    const receiverTitle = st?.media?.metadata?.title;
   175→    if (desc.castTitle && receiverTitle && receiverTitle !== desc.castTitle) return;
   176→    progressDesc = desc;
   177→    startProgressTracker();
   178→  } catch {
   179→    /* ignore */
   180→  }
   181→}
   182→
   183→function startProgressTracker() {
   184→  if (typeof window === "undefined" || demoHook) return;
   185→  if (progressTimer != null) return; // already watching this session
   186→  trackerLastPos = castPosition();
   187→  trackerActiveAt = Date.now();
   188→  progressTimer = window.setInterval(() => {
   189→    if (currentState !== "casting") return;
   190→    // ---- self-heal + sync watchdog (covers RESUMED sessions) ----
   191→    // a page reload rejoins the live session with a null/stale media client;
   192→    // re-acquire it, and if the receiver never reports a status, say so once
   193→    if (!refs.media || !statusOf(refs.media)) {
   194→      try {
   195→        const live = cfw()?.CastContext.getInstance?.()?.getCurrentSession?.();
   196→        if (live) {
   197→          refs.session = live;
   198→          attachRemoteMedia(live);
   199→        }
   200→      } catch {
   201→        /* ignore */
   202→      }
   203→      if (noStatusSince == null) noStatusSince = Date.now();
   204→      else if (!syncToastShown && Date.now() - noStatusSince > 12_000) {
   205→        syncToastShown = true;
   206→        noteTrail("no-status");
   207→        notifyMediaError("Can't sync with the TV's playback — Stop casting, then cast this episode again");
   208→      }
   209→    } else {
   210→      noStatusSince = null;
   211→      syncToastShown = false;
   212→    }
   213→    if (!progressDesc) return; // resumed session not yet adopted — controls only
   214→    const pos = castPosition();
   215→    const now = Date.now();
   216→    if (pos == null || pos === trackerLastPos) {
   217→      // paused, buffering or unreachable — wall time between these ticks
   218→      // must not accrue as watch time
   219→      trackerLastPos = pos;
   220→      trackerActiveAt = now;
   221→    } else {
   222→      const delta = Math.min(120, (now - trackerActiveAt) / 1000);
   223→      trackerLastPos = pos;
   224→      trackerActiveAt = now;
   225→      if (delta >= 1) {
   226→        recordWatchSeconds({ malId: progressDesc.malId, title: progressDesc.title ?? "", image: progressDesc.image ?? null, delta });
   227→      }
   228→      saveProgressDetailed({
   229→        malId: progressDesc.malId,
   230→        episode: progressDesc.episode,
   231→        lang: progressDesc.lang,
   232→        server: progressDesc.server,
   233→        position: pos,
   234→        duration: castDuration() || 0,
   235→        title: progressDesc.title,
   236→        image: progressDesc.image,
   237→        total: progressDesc.total,
   238→      });
   239→    }
   240→    // ---- auto next episode (mirrors the in-player ended handler) ----
   241→    if (progressDesc && autoNextHandler) {
   242→      const dur = castDuration() ?? 0;
   243→      const st = statusOf(refs.media);
   244→      const finished = st?.playerState === "IDLE" && st?.idleReason === "FINISHED";
   245→      const atEnd = dur > 0 && pos != null && pos >= dur - 2;
   246→      if ((finished || atEnd) && autoNextFiredEpoch !== autoNextEpoch) {
   247→        if (readAppSettings().autoNext) {
   248→          autoNextFiredEpoch = autoNextEpoch;
   249→          try {
   250→            autoNextHandler();
   251→          } catch {
   252→            /* navigation is best-effort */
   253→          }
   254→        }
   255→      }
   256→    }
   257→  }, 10_000);
   258→}

   276→/* ---- stuck-load watchdog ------------------------------------------------
   277→ * loadMedia() resolving only means the receiver ACCEPTED the request. If
   278→ * the media URL can't actually be fetched by the TV (CDN blocks the
   279→ * proxy's datacenter IP, dead gateway), the receiver sits on its loading
   280→ * screen forever while the remote shows dead 0:00/0:00 controls — exactly
   281→ * the owner's cast report (2026-10-09). Watch the status briefly after
   282→ * each load: no duration and no position progress within 12s = the stream
   283→ * never loaded — say so with an actionable message instead of leaving
   284→ * silent dead controls. Disarms itself the moment the media is real.
   285→ * ----------------------------------------------------------------------- */
   286→let stuckWatchdog: number | null = null;
   287→function disarmStuckWatchdog() {
   288→  if (stuckWatchdog != null) {
   289→    window.clearInterval(stuckWatchdog);
   290→    stuckWatchdog = null;
   291→  }
   292→}
   293→function armStuckLoadWatchdog() {
   294→  if (typeof window === "undefined" || demoHook) return;
   295→  disarmStuckWatchdog();
   296→  const startedAt = Date.now();
   297→  stuckWatchdog = window.setInterval(() => {
   298→    if (currentState !== "casting") {
   299→      disarmStuckWatchdog();
   300→      return;
   301→    }
   302→    const dur = castDuration();
   303→    const pos = castPosition();
   304→    if ((dur != null && dur > 0) || (pos != null && pos > 0)) {
   305→      disarmStuckWatchdog(); // media is real and moving — normal remote operation
   306→      return;
   307→    }
   308→    if (Date.now() - startedAt > 12_000) {
   309→      disarmStuckWatchdog();
   310→      noteTrail("stuck-loading");
   311→      notifyMediaError("The TV can't load this stream — try Server 2, or stop casting and cast again");
   312→    }
   313→  }, 1_000);
   314→}

   316→/**
   317→ * Squeeze every byte of meaning out of a cast failure. CAF rejections come
   318→ * in many shapes: undefined, empty object, {errorCode}, {code, description},
   319→ * DOMException, Error... Serialize whichever fields actually exist.
   320→ */
   321→function describeError(err: unknown): string {
   322→  if (err == null) return "";
   323→  if (typeof err === "string") return err.trim();
   324→  const e = err as Record<string, unknown>;
   325→  const parts: string[] = [];
   326→  for (const key of ["message", "description", "errorCode", "code", "detail", "reason", "name"]) {
   327→    const v = e[key];
   328→    if (typeof v === "string" && v.trim()) parts.push(`${key}:${v.trim().replace(/_/g, " ")}`);
   329→    else if (typeof v === "number") parts.push(`${key}:${v}`);
   330→  }
   331→  // last resort: every own property, JSON-shaped (truncated)
   332→  if (parts.length === 0) {
   333→    try {
   334→      const keys = Object.getOwnPropertyNames(e);
   335→      const snap = JSON.stringify(e, keys as never);
   336→      if (snap && snap !== "{}" && snap !== '""') parts.push(snap.slice(0, 140));
   337→      else if (keys.length > 0) parts.push(keys.join(","));
   338→    } catch {
   339→      /* ignore */
   340→    }
   341→  }
   342→  return parts.join(" · ").slice(0, 180);
   343→}

   370→/**
   371→ * Inject the CAF sender SDK once and configure the cast context with the
   372→ * Default Media Receiver. Safe to call from every player mount.
   373→ */
   374→export function ensureCastSdk(): void {
   375→  if (typeof window === "undefined" || sdkReady || sdkLoading) return;
   376→
   377→  // SDK already there (another component loaded it)
   378→  if (cfw() && (window as any).chrome?.cast?.media) {
   379→    configureContext();
   380→    return;
   381→  }
   382→
   383→  sdkLoading = true;
   384→  const done = (ok: boolean) => {
   385→    sdkLoading = false;
   386→    if (!ok) {
   387→      setState("unavailable");
   388→      return;
   389→    }
   390→    configureContext();
   391→  };
   392→
   393→  (window as any).__onGCastApiAvailable = (available: boolean) => done(available !== false);
   394→
   395→  const existing = document.querySelector<HTMLScriptElement>(`script[src="${CAST_SDK_SRC}"]`);
   396→  if (existing) return; // callback above will fire
   397→  const script = document.createElement("script");
   398→  script.src = CAST_SDK_SRC;
   399→  script.async = true;
   400→  script.onerror = () => done(false);
   401→  document.head.appendChild(script);
   402→
   403→  // belt & braces: some builds never call __onGCastApiAvailable
   404→  window.setTimeout(() => {
   405→    if (!sdkReady && cfw() && (window as any).chrome?.cast?.media) configureContext();
   406→  }, 6000);
   407→}

   409→function configureContext() {
   410→  try {
   411→    const framework = cfw();
   412→    if (!framework || !(window as any).chrome?.cast?.media) {
   413→      setState("unavailable");
   414→      return;
   415→    }
   416→    const context = framework.CastContext.getInstance();
   417→    context.setOptions({
   418→      receiverApplicationId: (window as any).chrome.cast.media.DEFAULT_MEDIA_RECEIVER_APP_ID,
   419→      autoJoinPolicy: (window as any).chrome.cast.AutoJoinPolicy.ORIGIN_SCOPED,
   420→    });
   421→    context.addEventListener(
   422→      framework.CastContextEventType.SESSION_STATE_CHANGED,
   423→      (e: any) => {
   424→        const s = e?.sessionState;
   425→        const S = framework.SessionState ?? {};
   426→        if (s === S.SESSION_STARTED || s === S.SESSION_RESUMED) {
   427→          noteTrail("session-started");
   428→          refs.session = e.session ?? context.getCurrentSession();
   429→          // RESUMED (page reload / PWA re-open while the TV plays): adopt the
   430→          // receiver's CURRENT media client right here — waiting for a load
   431→          // left the remote dead at 0:00/0:00 with no-op controls
   432→          attachRemoteMedia(refs.session);
   433→          startProgressTracker();
   434→          setState("casting");
   435→        } else if (s === S.SESSION_START_FAILED) {
   436→          // THIS is where the real reason hides when requestSession rejects empty
   437→          noteTrail("session-failed", e?.errorCode);
   438→          refs.session = null;
   439→          refs.media = null;
   440→          setState("idle");
   441→        } else if (s === S.SESSION_ENDED || s === S.SESSION_END_FAILED) {
   442→          noteTrail("session-ended", s === S.SESSION_END_FAILED ? e?.errorCode : undefined);
   443→          refs.session = null;
   444→          refs.media = null;
   445→          progressDesc = null;
   446→          captionsTrackId = null;
   447→          captionsActive = false;
   448→          noStatusSince = null;
   449→          syncToastShown = false;
   450→          stopProgressTracker();
   451→          setState("idle");
   452→        }
   453→      },
   454→    );
   455→    context.addEventListener(framework.CastContextEventType.CAST_STATE_CHANGED, (e: any) => {
   456→      noteTrail(`cast:${e?.castState ?? "?"}`, e?.errorCode);
   457→    });
   458→    // a previous page load may still hold a live session
   459→    if (context.getCurrentSession()) {
   460→      refs.session = context.getCurrentSession();
   461→      setState("casting");
   462→    } else {
   463→      setState("idle");
   464→    }
   465→    sdkReady = true;
   466→  } catch {
   467→    setState("unavailable");
   468→  }
   469→}

   507→export async function castSetCaptions(on: boolean): Promise<boolean> {

   550→/** Side-load a WebVTT caption track onto the Default Media Receiver. */
   551→function attachCaptions(chromeCast: any, mediaInfo: any, request: any, captions?: { url: string; label?: string }, on = true) {
   552→  captionsTrackId = null;
   553→  captionsActive = false;
   554→  if (!captions?.url || !chromeCast.media.Track) return;
   555→  try {
   556→    const track = new chromeCast.media.Track(0, chromeCast.media.TrackType.TEXT);
   557→    track.subtype = chromeCast.media.TextTrackType?.SUBTITLES ?? "SUBTITLES";
   558→    track.name = captions.label || "English";
   559→    track.language = "en";
   560→    track.contentId = absoluteUrl(captions.url);
   561→    track.contentType = "text/vtt";
   562→    mediaInfo.tracks = [track];
   563→    mediaInfo.textTrackStyle = tvTrackStyle(chromeCast);
   564→    request.activeTrackIds = on ? [0] : [];
   565→    captionsTrackId = 0;
   566→    captionsActive = on;
   567→  } catch {
   568→    /* captions are best-effort — video still casts without them */
   569→  }

   574→/**
   575→ * Request a (possibly resumed) session. Rejects on user cancel; returns null
   576→ * when the retry after a transient failure also failed.
   577→ */
   578→async function requestSession(context: any): Promise<any | null> {
   579→  try {
   580→    const session = await context.requestSession(); // rejects on cancel/failure
   581→    return session ?? context.getCurrentSession();
   582→  } catch (err) {
   583→    const text = describeError(err).toLowerCase();
   584→    // user closed the chooser — do not retry, just report
   585→    if (text.includes("cancel")) throw err;
   586→    // first connect attempts fail transiently on flaky mesh/2.4GHz links;
   587→    // one quiet retry catches a good chunk of those
   588→    noteTrail("retry");
   589→    await sleep(1200);
   590→    try {
   591→      const session = await context.requestSession();
   592→      return session ?? context.getCurrentSession();
   593→    } catch {
   594→      return null;
   595→    }
   596→  }
   597→}
   598→
   599→/**
   600→ * The session wrapper the SDK handed us earlier can go STALE while the
   601→ * module-level ref survives (phone locked, app backgrounded, Wi-Fi blip —
   602→ * the receiver drops the socket, CAF tears the session down internally).
   603→ * Calling loadMedia() on the stale wrapper then explodes deep inside the SDK
   604→ * with e.g. "Cannot read properties of undefined (reading
   605→ * 'addUpdateListener')". Re-hydrate from CastContext before every reuse:
   606→ * it returns null/undefined once the session is really gone.
   607→ */
   608→function refreshLiveSession(context: any): boolean {
   609→  if (!refs.session) return false;
   610→  try {
   611→    const live = context.getCurrentSession?.();
   612→    if (live) {
   613→      refs.session = live;
   614→      return true;
   615→    }
   616→  } catch {
   617→    /* fall through — treat as dead */
   618→  }
   619→  refs.session = null;
   620→  refs.media = null;
   621→  setState("idle");
   622→  return false;
   623→}
   624→
   625→/**
   626→ * CAF's CastSession.loadMedia() resolves VOID — the controllable object is
   627→ * the RemoteMediaClient handed out by session.getMediaSession(). Treating
   628→ * the resolution itself as the media object is what produced the owner's
   629→ * "Cannot read properties of undefined (reading 'addUpdateListener')"
   630→ * cast failure (the media HAD loaded on the TV; the sender then crashed on
   631→ * undefined). Legacy v2 callers get the Media object back, so accept both
   632→ * shapes: prefer the session's client, fall back to whatever loadMedia
   633→ * returned.
   634→ */
   635→function resolveClient(session: any, loaded: any): any {
   636→  try {
   637→    const viaSession = session?.getMediaSession?.();
   638→    if (viaSession) return viaSession;
   639→  } catch {
   640→    /* fall through to the loaded shape */
   641→  }
   642→  return loaded ?? null;
   643→}
   644→
   645→/** MediaStatus accessor for both API shapes (RemoteMediaClient.getMediaStatus()
   646→ * vs the v2 Media object's .status property). */
   647→function statusOf(media: any): any {
   648→  if (!media) return null;
   649→  try {
   650→    const st = media.getMediaStatus?.();
   651→    if (st) return st;
   652→  } catch {
   653→    /* fall through */
   654→  }
   655→  return media.status ?? null;
   656→}

   658→/** Ask the user to pick a device, then load the episode. Returns success. */
   659→export async function castLoad(req: CastMediaRequest): Promise<{ ok: boolean; error?: string }> {
   660→  try {
   661→    const framework = cfw();
   662→    const chromeCast = (window as any).chrome?.cast;
   663→    if (!framework || !chromeCast) return { ok: false, error: "Cast is not available in this browser" };
   664→
   665→    setState("connecting");
   666→    const context = framework.CastContext.getInstance();
   667→
   668→    if (refs.session) refreshLiveSession(context);
   669→    if (!refs.session) {
   670→      refs.session = await requestSession(context);
   671→    }
   672→    if (!refs.session) throw new Error("No Chromecast selected");
   673→
   674→    progressDesc = req.progress ?? null;
   675→    autoNextEpoch++;
   676→    const loaded = await loadMediaWithRecovery(context, chromeCast, req);
   677→    if (!attachRemoteMedia(refs.session)) {
   678→      // loadMedia resolving only proves the REQUEST was accepted (CAF resolves
   679→      // VOID) — the controllable client rides on the session
   680→      refs.media = resolveClient(refs.session, loaded);
   681→      wireMediaErrorListener(refs.media);
   682→    }
   683→    armStuckLoadWatchdog();
   684→    startProgressTracker();
   685→    setState("casting");
   686→    return { ok: true };
   687→  } catch (err: unknown) {
   688→    // CAF errors arrive in mixed shapes (often EMPTY) — describeError pulls
   689→    // whatever exists, and lastEventError holds the reason the context events
   690→    // saw (the promise rejection alone is frequently {} on session failures).
   691→    const detail = [describeError(err), lastEventError ? `event:${lastEventError}` : ""]
   692→      .filter(Boolean)
   693→      .join(" · ");
   694→    const flat = detail.toLowerCase();
   695→    console.warn("[cast] session/media failure:", err, "| eventError:", lastEventError, "| trail:", castTrail.join(" > "));
   696→    let msg: string;
   697→    if (flat.includes("cancel")) msg = "Cast cancelled";
   698→    else if (flat.includes("no device") || flat.includes("not found") || flat.includes("no compatible")) {
   699→      msg = "No cast device found — make sure your TV/Chromecast is on the same Wi-Fi, then try again";
   700→    } else if (flat.includes("session error") || flat.includes("timeout") || flat.includes("unreachable")) {
   701→      msg = `Couldn't reach the TV (${detail}) — your phone finds the TV but can't open a connection. Router AP/client isolation or a guest network usually causes this`;
   702→    } else if (flat.includes("load") || flat.includes("media") || flat.includes("invalid")) {
   703→      msg = "The TV couldn't load this stream — try another server for this episode";
   704→    } else {
   705→      // unknown failure — show everything we captured so it can be reported
   706→      msg = `Cast failed (${detail || "no details"}) — tell me this text to fix it`;
   707→    }
   708→    if (!refs.session) setState("idle");
   709→    return { ok: false, error: msg };
   710→  }
   711→}
   712→
   713→/**
   714→ * loadMedia with one stale-session recovery: the SDK throws opaque TypeErrors
   715→ * (addUpdateListener of undefined) when the cached CastSession wrapper outlived
   716→ * its underlying connection. Discard the dead wrapper, re-ask for a session,
   717→ * retry once — the user just sees the cast succeed.
   718→ */
   719→async function loadMediaWithRecovery(context: any, chromeCast: any, req: CastMediaRequest): Promise<any> {
   720→  const build = () => buildLoadRequest(chromeCast, req);
   721→  try {
   722→    return await refs.session.loadMedia(build());
   723→  } catch (err) {
   724→    // no live session underneath — drop it and retry once with a fresh one
   725→    noteTrail("stale-session");
   726→    refs.session = null;
   727→    refs.media = null;
   728→    setState("idle");
   729→    const fresh = await requestSession(context);
   730→    if (!fresh) throw err;
   731→    refs.session = fresh;
   732→    return await refs.session.loadMedia(build());
   733→  }
   734→}
   735→
   736→function buildLoadRequest(chromeCast: any, req: CastMediaRequest): any {
   737→  // the Default Media Receiver sniffs better with the honest type: our SP
   738→  // sources are progressive MP4s, everything else is an HLS master
   739→  const type = /\.m3u8($|\?)/i.test(req.src) ? "application/x-mpegurl" : "video/mp4";
   740→  const mediaInfo = new chromeCast.media.MediaInfo(req.src, type);
   741→  const metadata = new chromeCast.media.GenericMediaMetadata();
   742→  metadata.title = req.title;
   743→  if (req.subtitle) metadata.subtitle = req.subtitle;
   744→  if (req.imageUrl) metadata.images = [new chromeCast.Image(req.imageUrl)];
   745→  mediaInfo.metadata = metadata;
   746→  mediaInfo.streamType = chromeCast.media.StreamType.BUFFERED;
   747→  const request = new chromeCast.media.LoadRequest(mediaInfo);
   748→  request.autoplay = true;
   749→  request.currentTime = Math.max(0, req.startTime ?? 0);
   750→  attachCaptions(chromeCast, mediaInfo, request, req.captions, req.captionsOn ?? true);
   751→  return request;
   752→}

   754→/** Load another episode into the live cast session (no re-prompt). */
   755→export async function castLoadNext(req: CastMediaRequest): Promise<boolean> {
   756→  if (demoHook) return demoHook.loadNext();
   757→  try {
   758→    const framework = cfw();
   759→    const chromeCast = (window as any).chrome?.cast;
   760→    if (!framework || !chromeCast) return false;
   761→    const context = framework.CastContext.getInstance();
   762→    // a session that died since the last episode (phone slept, Wi-Fi blip)
   763→    // must not explode inside loadMedia — verify it is still alive first
   764→    if (!refreshLiveSession(context) || !refs.session) return false;
   765→    progressDesc = req.progress ?? progressDesc; // keep the old descriptor when none is given
   766→    autoNextEpoch++;
   767→    // loadMedia resolves VOID in CAF — prefer the session's client
   768→    const loadedNext = await refs.session.loadMedia(buildLoadRequest(chromeCast, req));
   769→    if (!attachRemoteMedia(refs.session)) {
   770→      refs.media = resolveClient(refs.session, loadedNext);
   771→      wireMediaErrorListener(refs.media);
   772→    }
   773→    armStuckLoadWatchdog();
   774→    startProgressTracker();
   775→    return true;
   776→  } catch {
   777→    // the session died mid-series — drop it so the next Cast tap re-prompts
   778→    refs.session = null;
   779→    refs.media = null;
   780→    setState("idle");
   781→    return false;
   782→  }
   783→}

   785→/** Stop casting and give playback back to the device. */
   786→export function castStop(): void {
   787→  disarmStuckWatchdog();
   788→  // final Continue-Watching write at the exact second the TV was at — the
   789→  // tracker dies with the session, this flush must not be skipped
   790→  if (progressDesc && !demoHook) {
   791→    const pos = castPosition();
   792→    if (pos != null) {
   793→      saveProgressDetailed({
   794→        malId: progressDesc.malId,
   795→        episode: progressDesc.episode,
   796→        lang: progressDesc.lang,
   797→        server: progressDesc.server,
   798→        position: pos,
   799→        duration: castDuration() || 0,
   800→        title: progressDesc.title,
   801→        image: progressDesc.image,
   802→        total: progressDesc.total,
   803→      });
   804→    }
   805→  }
   806→  progressDesc = null;
   807→  captionsTrackId = null;
   808→  captionsActive = false;
   809→  noStatusSince = null;
   810→  syncToastShown = false;
   811→  stopProgressTracker();
   812→  try {
   813→    refs.media = null;
   814→    refs.session?.endCurrentSession?.(true);
   815→    cfw()?.CastContext.getInstance().endCurrentSession(true);
   816→  } catch {
   817→    /* ignore */
   818→  }
   819→  refs.session = null;
   820→  setState("idle");
   821→}

   823→/** Best-effort current position inside the cast session (seconds). */
   824→export function castPosition(): number | null {
   825→  if (demoHook) return demoHook.position();
   826→  try {
   827→    const t = refs.media?.getEstimatedTime?.();
   828→    if (typeof t === "number" && Number.isFinite(t)) return t;
   829→    // v2 Media shape: .currentTime property
   830→    const legacy = refs.media?.currentTime;
   831→    return typeof legacy === "number" && Number.isFinite(legacy) ? legacy : null;
   832→  } catch {
   833→    return null;
   834→  }
   835→}

   837→export function castDuration(): number | null {
   838→  if (demoHook) return demoHook.duration();
   839→  try {
   840→    // RemoteMediaClient: status.media.duration · legacy v2 Media: .duration
   841→    const d = statusOf(refs.media)?.media?.duration ?? refs.media?.duration;
   842→    return typeof d === "number" && Number.isFinite(d) ? d : null;
   843→  } catch {
   844→    return null;
   845→  }
   846→}
