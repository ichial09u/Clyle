    24→function fmtTime(s: number): string {
    25→  if (!Number.isFinite(s) || s < 0) return "0:00";
    26→  const h = Math.floor(s / 3600);
    27→  const m = Math.floor((s % 3600) / 60);
    28→  const sec = Math.floor(s % 60);
    29→  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${String(sec).padStart(2, "0")}` : `${m}:${String(sec).padStart(2, "0")}`;
    30→}

   538→  const subTrack = stream?.subtitles?.[0];
   539→
   540→  const pct = duration > 0 ? (currentTime / duration) * 100 : 0;
   541→  const bufPct = duration > 0 ? (buffered / duration) * 100 : 0;
   542→  const casting = castState === "casting";
   543→  const castUsable = castState !== "unavailable";
   544→  const ui = uiVisible; // shorthand for JSX below
   545→
   546→  return (
   547→    <div className="fixed inset-0 z-50 bg-black select-none" ref={shellRef}>

   888→      {/* download format dialog */}

   914→          </div>
   915→        </div>

   917→    </div>
   918→  );
   919→}

   921→function ControlButton({
   922→  icon: Icon,
   923→  label,
   924→  onClick,
   925→  disabled,
   926→  active,
   927→}: {
   928→  icon: React.ComponentType<{ className?: string }>;
   929→  label: string;
   930→  onClick: () => void;
   931→  disabled?: boolean;
   932→  active?: boolean;
   933→}) {
   934→  // reference style: icon and label side-by-side in one row inside the player
   935→  return (
   936→    <button
   937→      onClick={onClick}
   938→      disabled={disabled}
   939→      className={`flex items-center gap-1 disabled:opacity-30 active:scale-95 transition px-0.5 py-1 ${active ? "text-[#ff5252]" : "text-gray-200 hover:text-white"}`}
   940→    >
   941→      <Icon className="w-3 h-3 shrink-0" />
   942→      <span className="text-[10.5px] leading-none whitespace-nowrap">{label}</span>
   943→    </button>
   944→  );
   945→}
   946→
