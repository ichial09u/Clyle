     1→     1→/**
     2→     2→ * Skip-time lookup: OP/ED intervals for every catalog anime.
     3→     3→ *
     4→     4→ * Four tiers, best data wins:
     5→     5→ *   1. Zoko's own per-episode `skip` payload (frame-accurate, but rare —
     6→     6→ *      only some titles carry it)
     7→     7→ *   2. The vendored AniSkip community dataset (skip-times.data.json,
     8→     8→ *      built from aniskip's public dump — covers 42/43 catalog titles,
     9→     9→ *      3,290 OP entries, highest-vote row per episode)
    10→    10→ *   3. Per-anime consensus: openings are the SAME video across episodes,
    11→    11→ *      so episodes missing from the dump reuse the median interval of
    12→    12→ *      the anime's other episodes
    13→    13→ *   4. LIVE AniSkip query — for brand-new episodes (and freshly imported
    14→    14→ *      anime) that predate the vendored dump. Fetched once on first open,
    15→    15→ *      cached on disk (positive entries forever, misses 7 days) and warmed
    16→    16→ *      proactively by the catalog sync, so a newly imported episode
    17→    17→ *      effectively "imports" its own skip intro with it. The player still
    18→    18→ *      has its own last-resort 8-90s heuristic window for TV-length
    19→    19→ *      episodes with no data at all.
    20→    20→ *   4.5 SkipDB fills (skipdb.tv) — AniSkip doesn't cover everything (HxH
    21→    21→ *      2011, Code Geass R2, Oshi no Ko, Naruto Shippuden…). SkipDB is the
    22→    22→ *      open crowdsourced skip-timestamps database (ODbL 1.0), keyed by IMDb
    23→    23→ *      ids + real season/episode numbers. The backfill bot maps MAL->IMDb
    24→    24→ *      (Wikidata P4086+P345, Jikan backup), vendors the dump into the
    25→    25→ *      SkipDbFill table and refreshes weekly; this tier reads it for any
    26→    26→ *      episode the tiers above left blank.
    27→    27→ *
    28→    28→ * Server-side only (the dataset ships inside the server bundle).
    29→    29→ */
    30→    30→
    31→    31→import { createHash } from "node:crypto";
    32→    32→import { mkdir, readFile, writeFile } from "node:fs/promises";
    33→    33→import { tmpdir } from "node:os";
    34→    34→import { join } from "node:path";
    35→    35→
    36→    36→import { db } from "@/lib/db";

    39→    39→export interface SkipInterval {
    40→    40→  start: number;
    41→    41→  end: number;
    42→    42→}
    43→    43→
    44→    44→export interface SkipData {
    45→    45→  intro?: SkipInterval;
    46→    46→  outro?: SkipInterval;
    47→    47→  /** where the data came from — handy for debugging */
    48→    48→  source?: "zoko" | "scout" | "aniskip" | "aniskip-live" | "skipdb" | "crowd";
    49→    49→}
    50→    50→
    51→    51→type EpisodeEntry = { op?: [number, number]; ed?: [number, number] };
    52→    52→type Dataset = Record<string, Record<string, EpisodeEntry>>;
    53→    53→
    54→    54→const data = skipDataset as unknown as Dataset;
    55→    55→
    56→    56→/** median across an anime's known episodes, per type */
    57→    57→const consensusCache = new Map<string, { op?: [number, number]; ed?: [number, number] }>();
    58→    58→
    59→    59→function median(values: number[]): number | null {
    60→    60→  if (!values.length) return null;
    61→    61→  const s = [...values].sort((a, b) => a - b);
    62→    62→  return s[Math.floor(s.length / 2)];
    63→    63→}
    64→    64→
    65→    65→function consensus(malId: number): { op?: [number, number]; ed?: [number, number] } {
    66→    66→  const key = String(malId);
    67→    67→  const hit = consensusCache.get(key);
    68→    68→  if (hit) return hit;
    69→    69→
    70→    70→  const out: { op?: [number, number]; ed?: [number, number] } = {};
    71→    71→  const byEp = data[key];
    72→    72→  if (byEp) {
    73→    73→    for (const type of ["op", "ed"] as const) {
    74→    74→      const starts: number[] = [];
    75→    75→      const ends: number[] = [];
    76→    76→      for (const entry of Object.values(byEp)) {
    77→    77→        const iv = entry[type];
    78→    78→        if (iv) {
    79→    79→          starts.push(iv[0]);
    80→    80→          ends.push(iv[1]);
    81→    81→        }
    82→    82→      }
    83→    83→      const s = median(starts);
    84→    84→      const e = median(ends);
    85→    85→      if (s != null && e != null) out[type] = [s, e];
    86→    86→    }
    87→    87→  }
    88→    88→  consensusCache.set(key, out);
    89→    89→  return out;
    90→    90→}

    92→    92→/** Look up OP/ED intervals for one episode: exact entry, else per-anime consensus. */
    93→    93→export function lookupSkipTimes(malId: number, ep: number): { intro?: SkipInterval; outro?: SkipInterval } {
    94→    94→  const byEp = data[String(malId)];
    95→    95→  const entry = byEp?.[String(ep)] ?? consensus(malId);
    96→    96→  if (!entry) return {};
    97→    97→  const out: { intro?: SkipInterval; outro?: SkipInterval } = {};
    98→    98→  if (entry.op) out.intro = { start: entry.op[0], end: entry.op[1] };
    99→    99→  if (entry.ed) out.outro = { start: entry.ed[0], end: entry.ed[1] };
   100→   100→  return out;
   101→   101→}
   102→   102→
   103→   103→/**
   104→   104→ * The vendored dataset's EXACT op row for this episode — no consensus
   105→   105→ * fallback. Used to defend known openings against the scout tier's
   106→   106→ * "couldn't measure" verdicts: a fleet nodetect row means the bots failed
   107→   107→ * to cross-match THIS scan (audio quirks, odd encodes), not that the show
   108→   108→ * has no opening. When the community has a vote-backed op for the exact
   109→   109→ * episode (Kimetsu ep1: 41.3–131.3 at 10k votes), it must never be
   110→   110→ * silenced by a failed measurement. Consensus has no such protection —
   111→   111→ * finale episodes genuinely drop their OP, and the anime-wide median
   112→   112→ * would wrongly button them.
   113→   113→ */
   114→   114→export function exactDatasetIntro(malId: number, ep: number): SkipInterval | null {
   115→   115→  const entry = data[String(malId)]?.[String(ep)];
   116→   116→  if (!entry?.op) return null;
   117→   117→  return { start: entry.op[0], end: entry.op[1] };
   118→   118→}
   119→   119→
   120→   120→function valid(iv: unknown): iv is SkipInterval {
   121→   121→  if (!iv || typeof iv !== "object") return false;
   122→   122→  const v = iv as SkipInterval;
   123→   123→  return Number.isFinite(v.start) && Number.isFinite(v.end) && v.end > v.start;
   124→   124→}
   125→   125→
   126→   126→/**
   127→   127→ * Plausibility gate — a "skip intro" that STARTS 10 minutes into a
   128→   128→ * 24-minute episode is not an opening (Zoko mis-tags recap blocks and
   129→   129→ * random chapters as intro/outro on plenty of shows — Grand Blue ep9
   130→   130→ * "intro 619-709"). Since the blind 8-90s guess was removed, a WRONG
   131→   131→ * interval shows a wrong button — worse than no button. Anything that
   132→   132→ * fails the sanity shape is dropped: the Skip Intro button must actually
   133→   133→ * skip the opening, or not exist.
   134→   134→ */
   135→   135→export function plausibleIntro(iv: SkipInterval | undefined | null, lenMin?: number | null): iv is SkipInterval {
   136→   136→  if (!iv || !valid(iv)) return false;
   137→   137→  const cap = lenMin && lenMin > 30 ? 420 : 320; // hour-long formats may open late
   138→   138→  return iv.start <= cap && iv.end - iv.start <= 180;
   139→   139→}
   140→   140→
   141→   141→export function plausibleOutro(iv: SkipInterval | undefined | null): iv is SkipInterval {
   142→   142→  return !!iv && valid(iv) && iv.end - iv.start <= 300;
   143→   143→}
   144→   144→
   145→   145→/**
   146→   146→ * Merge Zoko's native skip payload with the AniSkip dataset.
   147→   147→ * Zoko wins per-field (it is frame-accurate for its own encodes);
   148→   148→ * the dataset fills whatever Zoko left blank. BOTH sources pass the
   149→   149→ * plausibility gate here — this is where early-returning callers
   150→   150→ * (withSkipDataLive's "nothing left to ask") would otherwise leak a
   151→   151→ * mis-tagged interval (Grand Blue ep9 "intro 619-709") into the player.
   152→   152→ */
   153→   153→export function withSkipData<T extends { skip?: SkipData }>(
   154→   154→  malId: number,
   155→   155→  ep: number,
   156→   156→  stream: T,
   157→   157→  opts: { episodeLengthMin?: number | null } = {},
   158→   158→): T & { skip?: SkipData } {
   159→   159→  const zoko = stream.skip ?? {};
   160→   160→  const filled = lookupSkipTimes(malId, ep);
   161→   161→
   162→   162→  const merged: SkipData = {};
   163→   163→  const zi = valid(zoko.intro) && plausibleIntro(zoko.intro, opts.episodeLengthMin) ? zoko.intro : undefined;
   164→   164→  const zo = valid(zoko.outro) && plausibleOutro(zoko.outro) ? zoko.outro : undefined;
   165→   165→  const fi = valid(filled.intro) && plausibleIntro(filled.intro, opts.episodeLengthMin) ? filled.intro : undefined;
   166→   166→  const fo = valid(filled.outro) && plausibleOutro(filled.outro) ? filled.outro : undefined;
   167→   167→  const intro = zi ?? fi;
   168→   168→  const outro = zo ?? fo;
   169→   169→  if (intro) merged.intro = intro;
   170→   170→  if (outro) merged.outro = outro;
   171→   171→  if (intro || outro) {
   172→   172→    merged.source = zi || zo ? "zoko" : "aniskip";
   173→   173→  }
   174→   174→
   175→   175→  return { ...stream, skip: Object.keys(merged).length ? merged : undefined };
   176→   176→}
   177→   177→
   178→   178→/* ------------------------------------------------------------------ *
   179→   179→ * Tier 2.5 — the intro-scout fleet. Bots watch the first minutes of
   180→   180→ * every catalog episode (the SAME streams the player serves) and
   181→   181→ * cross-match loudness envelopes across episodes: the opening is the
   182→   182→ * one part of a show that repeats identically, so where episodes align
   183→   183→ * IS the opening. Because these intervals are measured on OUR encodes,
   184→   184→ * they beat AniSkip's community timings (taken on other encodes, often
   185→   185→ * offset by recaps/stream bumps — the "skip intro is inaccurate" bug).
   186→   186→ * Priority: Zoko native (frame-accurate) > scout > AniSkip > crowd.
   187→   187→ * ---------------------------------------------------------------- */
   188→   188→
   189→   189→const SCOUT_CACHE_TTL_MS = 60_000;
   190→   190→const scoutCache = new Map<number, { rows: Array<{ ep: number; start: number; end: number; nodetect: boolean }>; at: number }>();
   191→   191→
   192→   192→async function scoutRows(malId: number): Promise<Array<{ ep: number; start: number; end: number; nodetect: boolean }>> {
   193→   193→  const hit = scoutCache.get(malId);
   194→   194→  if (hit && Date.now() - hit.at < SCOUT_CACHE_TTL_MS) return hit.rows;
   195→   195→  const rows = await db.introScout.findMany({
   196→   196→    where: { malId },
   197→   197→    orderBy: { updatedAt: "desc" },
   198→   198→    take: 60, // room for nodetect rows too — real measurements are filtered below
   199→   199→    select: { ep: true, start: true, end: true, nodetect: true },
   200→   200→  });
   201→   201→  scoutCache.set(malId, { rows, at: Date.now() });
   202→   202→  return rows;
   203→   203→}
   204→   204→
   205→   205→/**
   206→   206→ * The fleet's verdict for one episode:
   207→   207→ *   interval set    — the opening interval (exact per-episode row, or the
   208→   208→ *                     anime-wide median when 2+ episodes agree)
   209→   209→ *   confirmedNone   — bots WATCHED this exact episode and found NO opening:
   210→   210→ *                     the Skip Intro button must not appear for it (no
   211→   211→ *                     guessing, no AniSkip resurrection)
   212→   212→ *   null            — the fleet has nothing on this anime
   213→   213→ */
   214→   214→export async function scoutIntro(
   215→   215→  malId: number,
   216→   216→  ep: number,
   217→   217→): Promise<{ interval: SkipInterval | null; exact: boolean; confirmedNone?: boolean } | null> {
   218→   218→  try {
   219→   219→    const rows = await scoutRows(malId);
   220→   220→    const exact = rows.find((r) => r.ep === ep);
   221→   221→    if (exact) {
   222→   222→      if (exact.nodetect) return { interval: null, exact: true, confirmedNone: true };
   223→   223→      // degenerate row guard — merge upserts can land as {0,0} (old establish
   224→   224→      // failures came back from prod without their nodetect flag); a zero or
   225→   225→      // inverted interval is a broken row, not a measurement. Ignore it and
   226→   226→      // fall through to the anime-wide median / other tiers.
   227→   227→      if (exact.end > exact.start) return { interval: { start: exact.start, end: exact.end }, exact: true };
   228→   228→    }
   229→   229→    // degenerate {0,0} rows are excluded here too — they must not skew the median
   230→   230→    const measured = rows.filter((r) => !r.nodetect && r.end > r.start);
   231→   231→    if (measured.length >= 2) {
   232→   232→      const med = (xs: number[]) => {
   233→   233→        const s = [...xs].sort((a, b) => a - b);
   234→   234→        return s[Math.floor(s.length / 2)];
   235→   235→      };
   236→   236→      const start = med(measured.map((r) => r.start));
   237→   237→      const end = med(measured.map((r) => r.end));
   238→   238→      if (end > start && end - start <= 150 && start <= 320) {
   239→   239→        return { interval: { start, end }, exact: false };
   240→   240→      }
   241→   241→    }
   242→   242→    return null;
   243→   243→  } catch {
   244→   244→    return null;
   245→   245→  }
   246→   246→}
   247→   247→
   248→   248→/* ------------------------------------------------------------------ *
   249→   249→ * Tier 3.5 — SkipDB fills (skipdb.tv, open data under ODbL 1.0).
   250→   250→ * For anime AniSkip doesn't have: the backfill bot maps MAL->IMDb once
   251→   251→ * (Wikidata + Jikan backup), vendors SkipDB's approved intro/outro segments
   252→   252→ * per (season, episode) into the SkipDbFill table, and re-downloads the
   253→   253→ * dump weekly so new episodes keep arriving. Same precedence class as the
   254→   254→ * vendored AniSkip dataset — community timings on other encodes — so it
   255→   255→ * only fills gaps the higher tiers left blank, never overrides them.
   256→   256→ * ---------------------------------------------------------------- */
   257→   257→
   258→   258→const SKIPDB_TTL_MS = 60_000;
   259→   259→const skipDbCache = new Map<
   260→   260→  number,
   261→   261→  { rows: Array<{ ep: number; kind: string; start: number; end: number; confidence: number }>; at: number }
   262→   262→>();
   263→   263→
   264→   264→async function skipDbRows(
   265→   265→  malId: number,
   266→   266→): Promise<Array<{ ep: number; kind: string; start: number; end: number; confidence: number }>> {
   267→   267→  const hit = skipDbCache.get(malId);
   268→   268→  if (hit && Date.now() - hit.at < SKIPDB_TTL_MS) return hit.rows;
   269→   269→  const rows = await db.skipDbFill.findMany({
   270→   270→    where: { malId },
   271→   271→    select: { ep: true, kind: true, start: true, end: true, confidence: true },
   272→   272→  });
   273→   273→  skipDbCache.set(malId, { rows, at: Date.now() });
   274→   274→  return rows;
   275→   275→}
   276→   276→
   277→   277→/**
   278→   278→ * SkipDB's exact per-episode intervals for one episode (op/ed in seconds),
   279→   279→ * already sanity-gated. null when the table has nothing for this anime.
   280→   280→ */
   281→   281→export async function skipDbIntervals(
   282→   282→  malId: number,
   283→   283→  ep: number,
   284→   284→): Promise<{ op?: SkipInterval; ed?: SkipInterval } | null> {
   285→   285→  try {
   286→   286→    const rows = await skipDbRows(malId);
   287→   287→    if (!rows.length) return null;
   288→   288→    const out: { op?: SkipInterval; ed?: SkipInterval } = {};
   289→   289→    for (const kind of ["op", "ed"] as const) {
   290→   290→      const r = rows.find((x) => x.ep === ep && x.kind === kind);
   291→   291→      if (!r) continue;
   292→   292→      const iv = { start: r.start, end: r.end };
   293→   293→      if (kind === "op" && plausibleIntro(iv)) out.op = iv;
   294→   294→      if (kind === "ed" && plausibleOutro(iv)) out.ed = iv;
   295→   295→    }
   296→   296→    return out.op || out.ed ? out : null;
   297→   297→  } catch {
   298→   298→    return null; // table missing / db hiccup — SkipDB is optional, never fatal
   299→   299→  }
   300→   300→}
   301→   301→
   302→   302→/* ------------------------------------------------------------------ *
   303→   303→ * Tier 4 — LIVE AniSkip lookups (cached), for episodes the vendored
   304→   304→ * dump + consensus don't cover: brand-new episodes of airing shows and
   305→   305→ * freshly imported anime. Also used to prefetch skip data when the
   306→   306→ * catalog sync imports new episodes (see prefetchSkipTimes below).
   307→   307→ * ------------------------------------------------------------------ */
   308→   308→
   309→   309→const ANISKIP_API = "https://api.aniskip.com/v2/skip-times";
   310→   310→// Submissions trickle in over the days after an episode airs (2026-season
   311→   311→// shows were 404 on air day, populated within the week) — re-ask daily so a
   312→   312→// fresh episode picks up community times as soon as anyone submits them.
   313→   313→const ANISKIP_MISS_TTL_MS = 24 * 60 * 60 * 1000;
   314→   314→const ANISKIP_DISK_DIR =
   315→   315→  process.env.VERCEL === "1" ? join(tmpdir(), "aniskip-cache") : join(process.cwd(), "db", "aniskip-cache");
   316→   316→
   317→   317→type LiveEntry = { op?: [number, number]; ed?: [number, number]; at: number; miss?: boolean };
   318→   318→type LiveTimes = { op?: [number, number]; ed?: [number, number] };
   319→   319→type LiveFetchResult = { times: LiveTimes | null; transient: boolean };
   320→   320→type LiveFile = Record<string, LiveEntry>;
   321→   321→const liveMem = new Map<string, LiveFile>();
   322→   322→const liveInFlight = new Map<string, Promise<LiveFetchResult>>();
   323→   323→
   324→   324→function liveDiskPath(malId: number): string {
   325→   325→  const hash = createHash("sha1").update(String(malId)).digest("hex").slice(0, 16);
   326→   326→  return join(ANISKIP_DISK_DIR, `${hash}.json`);
   327→   327→}
   328→   328→
   329→   329→async function liveRead(malId: number): Promise<LiveFile> {
   330→   330→  const key = String(malId);
   331→   331→  const mem = liveMem.get(key);
   332→   332→  if (mem) return mem;
   333→   333→  try {
   334→   334→    const file = JSON.parse(await readFile(liveDiskPath(malId), "utf8")) as LiveFile;
   335→   335→    liveMem.set(key, file);
   336→   336→    return file;
   337→   337→  } catch {
   338→   338→    return {};
   339→   339→  }
   340→   340→}
   341→   341→
   342→   342→async function liveWrite(malId: number, file: LiveFile): Promise<void> {
   343→   343→  liveMem.set(String(malId), file);
   344→   344→  try {
   345→   345→    await mkdir(ANISKIP_DISK_DIR, { recursive: true });
   346→   346→    await writeFile(liveDiskPath(malId), JSON.stringify(file), "utf8");
   347→   347→  } catch {
   348→   348→    /* read-only FS (serverless) — the in-memory copy still helps this instance */
   349→   349→  }
   350→   350→}
   351→   351→
   352→   352→/**
   353→   353→ * Ask api.aniskip.com for one episode's OP/ED times. `lenMin` is the
   354→   354→ * episode's length in MINUTES — the API requires an `episodeLength` param
   355→   355→ * (400s without it); 24 is the TV standard when we don't know better.
   356→   356→ * Returns the times plus whether the failure was TRANSIENT (5xx / rate
   357→   357→ * limit / timeout) — transient failures must NOT be negative-cached.
   358→   358→ */
   359→   359→async function fetchAniSkipLive(malId: number, ep: number, lenMin?: number): Promise<LiveFetchResult> {
   360→   360→  const len = lenMin && Number.isFinite(lenMin) && lenMin > 0 ? Math.round(lenMin) : 24;
   361→   361→  try {
   362→   362→    const res = await fetch(`${ANISKIP_API}/${malId}/${ep}?types=op&types=ed&episodeLength=${len}`, {
   363→   363→      headers: { Accept: "application/json" },
   364→   364→      cache: "no-store",
   365→   365→      signal: AbortSignal.timeout(4_500),
   366→   366→    });
   367→   367→    if (res.status === 404) return { times: null, transient: false }; // genuine "not submitted yet"
   368→   368→    if (!res.ok) return { times: null, transient: true }; // 429/5xx — retry another time
   369→   369→    const body = (await res.json()) as {
   370→   370→      found?: boolean;
   371→   371→      results?: Array<{ interval?: { startTime?: number; endTime?: number }; skipType?: string }>;
   372→   372→    };
   373→   373→    if (!body?.found || !Array.isArray(body.results)) return { times: null, transient: false };
   374→   374→    const out: LiveTimes = {};
   375→   375→    for (const r of body.results) {
   376→   376→      const s = r?.interval?.startTime;
   377→   377→      const e = r?.interval?.endTime;
   378→   378→      if (!Number.isFinite(s) || !Number.isFinite(e) || e! <= s!) continue;
   379→   379→      if (r.skipType === "op") out.op = [s!, e!];
   380→   380→      else if (r.skipType === "ed") out.ed = [s!, e!];
   381→   381→    }
   382→   382→    return { times: out.op || out.ed ? out : null, transient: false };
   383→   383→  } catch {
   384→   384→    return { times: null, transient: true }; // timeout / offline — retryable
   385→   385→  }
   386→   386→}
   387→   387→
   388→   388→/**
   389→   389→ * Cached live lookup for one episode. Positive entries are kept forever
   390→   390→ * (community timings don't rot); misses are retried after 7 days.
   391→   391→ */
   392→   392→async function liveLookup(malId: number, ep: number, lenMin?: number): Promise<LiveTimes | null> {
   393→   393→  const epKey = String(ep);
   394→   394→  const file = await liveRead(malId);
   395→   395→  const hit = file[epKey];
   396→   396→  if (hit && Date.now() - hit.at < (hit.miss ? ANISKIP_MISS_TTL_MS : Number.MAX_SAFE_INTEGER)) {
   397→   397→    return hit.miss ? null : { op: hit.op, ed: hit.ed };
   398→   398→  }
   399→   399→  const inflightKey = `${malId}/${ep}`;
   400→   400→  let p: Promise<LiveFetchResult> | undefined = liveInFlight.get(inflightKey);
   401→   401→  if (!p) {
   402→   402→    p = fetchAniSkipLive(malId, ep, lenMin)
   403→   403→      .then((r) => {
   404→   404→        liveInFlight.delete(inflightKey);
   405→   405→        return r;
   406→   406→      })
   407→   407→      .catch(() => {
   408→   408→        liveInFlight.delete(inflightKey);
   409→   409→        return { times: null, transient: true } as LiveFetchResult;
   410→   410→      });
   411→   411→    liveInFlight.set(inflightKey, p);
   412→   412→  }
   413→   413→  const wrapped = await p;
   414→   414→  const data = wrapped.times;
   415→   415→  const transient = wrapped.transient;
   416→   416→  if (!transient) {
   417→   417→    const fresh = await liveRead(malId); // another request may have written meanwhile
   418→   418→    fresh[epKey] = data ? { ...data, at: Date.now() } : { at: Date.now(), miss: true };
   419→   419→    await liveWrite(malId, fresh);
   420→   420→  }
   421→   421→  return data;
   422→   422→}

   424→   424→/**
   425→   425→ * Merge like `withSkipData`, plus the live AniSkip tier: anything the
   426→   426→ * vendored data left blank gets one (cached) live query. This is what makes
   427→   427→ * EVERY episode — including ones imported after the dump was built — carry
   428→   428→ * its skip intro/ending.
   429→   429→ */
   430→   430→export async function withSkipDataLive<T extends { skip?: SkipData }>(
   431→   431→  malId: number,
   432→   432→  ep: number,
   433→   433→  stream: T,
   434→   434→  opts: { episodeLengthMin?: number | null } = {},
   435→   435→): Promise<T & { skip?: SkipData }> {
   436→   436→  const merged = withSkipData(malId, ep, stream, opts);
   437→   437→  let skip: SkipData = { ...(merged.skip ?? {}) };
   438→   438→
   439→   439→  // scout tier — measured on OUR encodes by the fleet. An exact per-episode
   440→   440→  // measurement outranks everything (including Zoko's payload, which marks
   441→   441→  // the repeated cold-open as "intro start" on many shows); the anime-wide
   442→   442→  // median only fills gaps (Zoko/frame data wins there).
   443→   443→  // A CONFIRMED-NONE verdict (the fleet watched this episode against an
   444→   444→  // established reference and found no opening) wins too — EXCEPT when this
   445→   445→  // exact episode already has concrete intro data: Zoko's native payload or
   446→   446→  // the dataset's exact op row. A nodetect is "our scan couldn't align",
   447→   447→  // which must never silence a vote-backed community interval for the same
   448→   448→  // episode (Kimetsu ep1: 10k votes vs one failed audio match). Consensus /
   449→   449→  // live / crowd intros are NOT protected — finale episodes genuinely drop
   450→   450→  // their OP and the anime-wide median must not button them.
   451→   451→  const scout = await scoutIntro(malId, ep);
   452→   452→  const confirmedNone = !!scout?.confirmedNone;
   453→   453→  const sdbExact = await skipDbIntervals(malId, ep);
   454→   454→  const exactKnown = valid(stream.skip?.intro) || !!exactDatasetIntro(malId, ep) || !!sdbExact?.op;
   455→   455→  const suppressIntro = confirmedNone && !exactKnown;
   456→   456→  if (suppressIntro) {
   457→   457→    delete skip.intro;
   458→   458→    if (!skip.outro) delete skip.source;
   459→   459→  } else if (scout?.interval && (scout.exact || !valid(stream.skip?.intro))) {
   460→   460→    skip.intro = scout.interval;
   461→   461→    skip.source = "scout";
   462→   462→  }
   463→   463→  // SkipDB fill tier — exact per-episode community rows for anime AniSkip
   464→   464→  // doesn't carry. Exact rows survive a fleet confirmed-none (same defense
   465→   465→  // as the dataset's exact op row above); they never override zoko/scout/
   466→   466→  // AniSkip data, only fill what's missing.
   467→   467→  if (!suppressIntro && (!skip.intro || !skip.outro) && sdbExact) {
   468→   468→    if (!skip.intro && sdbExact.op) {
   469→   469→      skip.intro = sdbExact.op;
   470→   470→      skip.source = "skipdb";
   471→   471→    }
   472→   472→    if (!skip.outro && sdbExact.ed) {
   473→   473→      skip.outro = sdbExact.ed;
   474→   474→      if (!skip.source) skip.source = "skipdb";
   475→   475→    }
   476→   476→  }
   477→   477→  if (skip.intro && skip.outro) return { ...merged, skip }; // nothing left to ask
   478→   478→
   479→   479→  try {
   480→   480→    const live = await liveLookup(malId, ep, opts.episodeLengthMin ?? undefined);
   481→   481→    if (live) {
   482→   482→      if (!skip.intro && live.op && !suppressIntro) {
   483→   483→        skip.intro = { start: live.op[0], end: live.op[1] };
   484→   484→        skip.source = "aniskip-live";
   485→   485→      }
   486→   486→      if (!skip.outro && live.ed) skip.outro = { start: live.ed[0], end: live.ed[1] };
   487→   487→      if (skip.intro || skip.outro) {
   488→   488→        if (!skip.source) skip.source = "aniskip-live";
   489→   489→        return { ...merged, skip };
   490→   490→      }
   491→   491→    }
   492→   492→  } catch {
   493→   493→    /* live tier is best-effort — the crowd/heuristic tiers still cover episodes */
   494→   494→  }
   495→   495→  // crowd tier — for shows nobody has submitted anywhere yet, viewers who
   496→   496→  // scrubbed past the opening have already taught us where it ends
   497→   497→  // (never applied when the fleet confirmed there IS no opening)
   498→   498→  if (!suppressIntro) {
   499→   499→    try {
   500→   500→      const crowd = await crowdIntroTarget(malId, ep);
   501→   501→      if (crowd != null) {
   502→   502→        if (!skip.intro) {
   503→   503→          skip.intro = { start: 8, end: crowd };
   504→   504→          skip.source = "crowd";
   505→   505→        }
   506→   506→        if (skip.intro || skip.outro) return { ...merged, skip };
   507→   507→      }
   508→   508→    } catch {
   509→   509→      /* votes are best-effort too */
   510→   510→    }
   511→   511→  }
   512→   512→  // final plausibility gate — the button must actually skip the opening:
   513→   513→  // drop any interval that fails the sanity shape (wrong data < no button)
   514→   514→  if (skip.intro && !plausibleIntro(skip.intro, opts.episodeLengthMin)) {
   515→   515→    delete skip.intro;
   516→   516→    if (!skip.outro) delete skip.source;
   517→   517→  }
   518→   518→  if (skip.outro && !plausibleOutro(skip.outro)) {
   519→   519→    delete skip.outro;
   520→   520→    if (!skip.intro) delete skip.source;
   521→   521→  }
   522→   522→  return { ...merged, skip };
   523→   523→}
   524→   524→
   525→   525→/**
   526→   526→ * Coverage check used by the skip audit (bot): where would this episode's
   527→   527→ * Skip Intro data come from? Checks the vendored dataset (exact row, then
   528→   528→ * per-anime consensus), then the live tier's cache — `allowLiveFetch` also
   529→   529→ * permits one cached network query to AniSkip (used when probing gaps), and
   530→   530→ * finally the crowd-learned scrub median — the audit counts anything the
   531→   531→ * player would actually show. `confirmedNone` = the fleet verified there is
   532→   532→ * NO opening on this episode (the button is suppressed by design — not a
   533→   533→ * coverage gap).
   534→   534→ */
   535→   535→export interface SkipCoverage {
   536→   536→  intro?: SkipInterval;
   537→   537→  outro?: SkipInterval;
   538→   538→  source: "scout" | "exact" | "consensus" | "skipdb" | "live" | "crowd";
   539→   539→  confirmedNone?: boolean;
   540→   540→}
   541→   541→
   542→   542→export async function skipCoverage(
   543→   543→  malId: number,
   544→   544→  ep: number,
   545→   545→  opts: { allowLiveFetch?: boolean; episodeLengthMin?: number | null } = {},
   546→   546→): Promise<SkipCoverage | null> {
   547→   547→  // scout tier first — what the player would actually use (see the priority
   548→   548→  // in withSkipDataLive). A confirmed-none yields to an exact dataset op row
   549→   549→  // for THIS episode (same precedence as the player merge).
   550→   550→  const scout = await scoutIntro(malId, ep);
   551→   551→  if (scout?.confirmedNone) {
   552→   552→    const ex = exactDatasetIntro(malId, ep);
   553→   553→    if (ex && plausibleIntro(ex, opts.episodeLengthMin)) return { source: "exact", intro: ex };
   554→   554→    // a SkipDB exact row for this episode is also vote-backed concrete data
   555→   555→    // and must not be silenced by the fleet's failed cross-match
   556→   556→    const sdbN = await skipDbIntervals(malId, ep);
   557→   557→    if (sdbN?.op && plausibleIntro(sdbN.op, opts.episodeLengthMin)) return { source: "skipdb", intro: sdbN.op };
   558→   558→    return { source: "scout", confirmedNone: true };
   559→   559→  }
   560→   560→  if (scout?.interval) return { source: "scout", intro: scout.interval };
   561→   561→  const byEp = data[String(malId)];
   562→   562→  const exact = byEp?.[String(ep)];
   563→   563→  const exOp = exact?.op ? { start: exact.op[0], end: exact.op[1] } : null;
   564→   564→  const exEd = exact?.ed ? { start: exact.ed[0], end: exact.ed[1] } : null;
   565→   565→  if (exOp || exEd) {
   566→   566→    const cov: SkipCoverage = { source: "exact" };
   567→   567→    if (plausibleIntro(exOp, opts.episodeLengthMin)) cov.intro = exOp;
   568→   568→    if (plausibleOutro(exEd)) cov.outro = exEd;
   569→   569→    if (cov.intro || cov.outro) return cov;
   570→   570→  }
   571→   571→  const cons = consensus(malId);
   572→   572→  const coOp = cons.op ? { start: cons.op[0], end: cons.op[1] } : null;
   573→   573→  const coEd = cons.ed ? { start: cons.ed[0], end: cons.ed[1] } : null;
   574→   574→  if (coOp || coEd) {
   575→   575→    const cov: SkipCoverage = { source: "consensus" };
   576→   576→    if (plausibleIntro(coOp, opts.episodeLengthMin)) cov.intro = coOp;
   577→   577→    if (plausibleOutro(coEd)) cov.outro = coEd;
   578→   578→    if (cov.intro || cov.outro) return cov;
   579→   579→  }
   580→   580→  // SkipDB tier — the vendored-fill table for anime AniSkip doesn't have
   581→   581→  const sdb = await skipDbIntervals(malId, ep);
   582→   582→  if (sdb) {
   583→   583→    const cov: SkipCoverage = { source: "skipdb" };
   584→   584→    if (plausibleIntro(sdb.op, opts.episodeLengthMin)) cov.intro = sdb.op;
   585→   585→    if (plausibleOutro(sdb.ed)) cov.outro = sdb.ed;
   586→   586→    if (cov.intro || cov.outro) return cov;
   587→   587→  }
   588→   588→  const liveCov = (op?: [number, number], ed?: [number, number]): SkipCoverage | null => {
   589→   589→    const i = op ? { start: op[0], end: op[1] } : null;
   590→   590→    const o = ed ? { start: ed[0], end: ed[1] } : null;
   591→   591→    if (!plausibleIntro(i, opts.episodeLengthMin) && !plausibleOutro(o)) return null;
   592→   592→    const cov: SkipCoverage = { source: "live" };
   593→   593→    if (plausibleIntro(i, opts.episodeLengthMin)) cov.intro = i;
   594→   594→    if (plausibleOutro(o)) cov.outro = o;
   595→   595→    return cov;
   596→   596→  };
   597→   597→  if (opts.allowLiveFetch) {
   598→   598→    const live = await liveLookup(malId, ep, opts.episodeLengthMin ?? undefined);
   599→   599→    const cov = live ? liveCov(live.op, live.ed) : null;
   600→   600→    if (cov) return cov;
   601→   601→  } else {
   602→   602→    const cached = (await liveRead(malId))[String(ep)];
   603→   603→    const cov = cached && !cached.miss ? liveCov(cached.op, cached.ed) : null;
   604→   604→    if (cov) return cov;
   605→   605→  }
   606→   606→  // crowd tier — viewers' scrub medians count as coverage too (the player
   607→   607→  // shows a working Skip Intro button for them)
   608→   608→  const crowd = await crowdIntroTarget(malId, ep);
   609→   609→  if (crowd != null) return { source: "crowd", intro: { start: 8, end: crowd } };
   610→   610→  return null;
   611→   611→}
   612→   612→
   613→   613→/**
   614→   614→ * Warm the live cache ahead of time — called after the catalog sync imports
   615→   615→ * new episodes so their skip data is already on disk before anyone presses
   616→   616→ * play. Sequential with ~1.1s spacing (AniSkip's documented rate limit is
   617→   617→ * 1 req/s); bounded so a huge import can't run away.
   618→   618→ */
   619→   619→export async function prefetchSkipTimes(
   620→   620→  pairs: Array<{ malId: number; ep: number; lenMin?: number | null }>,
   621→   621→  budget = 16,
   622→   622→): Promise<number> {
   623→   623→  let fetched = 0;
   624→   624→  for (const { malId, ep, lenMin } of pairs.slice(0, budget)) {
   625→   625→    try {
   626→   626→      const file = await liveRead(malId);
   627→   627→      const hit = file[String(ep)];
   628→   628→      if (hit && (hit.op || hit.ed || (hit.miss && Date.now() - hit.at < ANISKIP_MISS_TTL_MS))) continue;
   629→   629→      const { times, transient } = await fetchAniSkipLive(malId, ep, lenMin ?? undefined);
   630→   630→      if (!transient) {
   631→   631→        const fresh = await liveRead(malId);
   632→   632→        fresh[String(ep)] = times ? { ...times, at: Date.now() } : { at: Date.now(), miss: true };
   633→   633→        await liveWrite(malId, fresh);
   634→   634→      }
   635→   635→      if (times) fetched++;
   636→   636→    } catch {
   637→   637→      /* keep going */
   638→   638→    }
   639→   639→    await new Promise((r) => setTimeout(r, 1_100)); // respect 1 req/s
   640→   640→  }
   641→   641→  return fetched;
   642→   642→}
   643→   643→
   644→   644→/* ------------------------------------------------------------------ *
   645→   645→ * Tier 5 — crowd-learned openings. Brand-new seasons often have NO
   646→   646→ * skip data anywhere (AniSkip 404s for months on niche shows), so the
   647→   647→ * fixed 8-90s guess lands in the middle of their long cold-opens.
   648→   648→ * Instead, viewers who manually scrub forward past the opening get
   649→   649→ * recorded (see POST /api/skip-vote) and the median scrub target
   650→   650→ * becomes this anime's Skip Intro end for everyone. Self-healing:
   651→   651→ * the more people watch, the sharper the button gets.
   652→   652→ * ---------------------------------------------------------------- */
   653→   653→
   654→   654→const CROWD_MIN_VOTES_EP = 2; // per-episode median needs at least this many
   655→   655→const CROWD_MIN_VOTES_ANIME = 4; // anime-wide fallback needs more agreement
   656→   656→const CROWD_VOTE_MAX_AGE_MS = 60 * 24 * 60 * 60 * 1000; // tastes/encodes drift
   657→   657→const CROWD_KEEP_PER_ANIME = 200; // storage cap per anime
   658→   658→
   659→   659→export const SKIP_VOTE_TARGET_MIN = 30;
   660→   660→export const SKIP_VOTE_TARGET_MAX = 420;
   661→   661→
   662→   662→/**
   663→   663→ * Record one implied OP-end vote. Best-effort: validates the catalog entry
   664→   664→ * and the target range, caps rows per anime, never throws.
   665→   665→ */
   666→   666→export async function recordSkipVote(malId: number, ep: number, target: number): Promise<boolean> {
   667→   667→  try {
   668→   668→    if (!Number.isFinite(malId) || malId <= 0 || !Number.isFinite(ep) || ep < 1 || ep > 2000) return false;
   669→   669→    if (!Number.isFinite(target)) return false;
   670→   670→    const t = Math.round(target);
   671→   671→    if (t < SKIP_VOTE_TARGET_MIN || t > SKIP_VOTE_TARGET_MAX) return false;
   672→   672→    const anime = await db.anime.findUnique({ where: { malId }, select: { malId: true } });
   673→   673→    if (!anime) return false;
   674→   674→    await db.skipVote.create({ data: { malId, ep, target: t } });
   675→   675→    // prune: keep the newest KEEP votes for this anime (votes are cheap to drop)
   676→   676→    const count = await db.skipVote.count({ where: { malId } });
   677→   677→    if (count > CROWD_KEEP_PER_ANIME) {
   678→   678→      const old = await db.skipVote.findMany({
   679→   679→        where: { malId },
   680→   680→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   681→   681→        take: count - CROWD_KEEP_PER_ANIME,
   682→   682→        select: { id: true },
   683→   683→      });
   684→   684→      if (old.length) await db.skipVote.deleteMany({ where: { id: { in: old.map((r) => r.id) } } });
   685→   685→    }
   686→   686→    return true;
   687→   687→  } catch {
   688→   688→    return false;
   689→   689→  }
   690→   690→}
   691→   691→
   692→   692→function medianNum(values: number[]): number | null {
   693→   693→  if (!values.length) return null;
   694→   694→  const s = [...values].sort((a, b) => a - b);
   695→   695→  return s[Math.floor(s.length / 2)];
   696→   696→}
   697→   697→
   698→   698→/**
   699→   699→ * The crowd's OP-end for one episode: median of this episode's recent votes,
   700→   700→ * else (with more agreement) the anime-wide median — openings are the same
   701→   701→ * video across episodes. Returns whole seconds or null when thin/old.
   702→   702→ */
   703→   703→export async function crowdIntroTarget(malId: number, ep: number): Promise<number | null> {
   704→   704→  try {
   705→   705→    const from = new Date(Date.now() - CROWD_VOTE_MAX_AGE_MS);
   706→   706→    const rows = await db.skipVote.findMany({
   707→   707→      where: { malId, createdAt: { gte: from } },
   708→   708→      select: { ep: true, target: true },
   709→   709→      orderBy: { createdAt: "desc" },
   710→   710→      take: 400,
   711→   711→    });
   712→   712→    const epTargets = rows.filter((r) => r.ep === ep).map((r) => r.target);
   713→   713→    if (epTargets.length >= CROWD_MIN_VOTES_EP) return medianNum(epTargets);
   714→   714→    const all = rows.map((r) => r.target);
   715→   715→    if (all.length >= CROWD_MIN_VOTES_ANIME) return medianNum(all);
   716→   716→    return null;
   717→   717→  } catch {
   718→   718→    return null;
   719→   719→  }
   720→   720→}
