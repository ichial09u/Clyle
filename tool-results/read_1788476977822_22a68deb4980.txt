    19→function readJson<T>(key: string, fallback: T): T {
    20→  if (typeof window === "undefined") return fallback;
    21→  try {
    22→    const raw = window.localStorage.getItem(key);
    23→    return raw ? (JSON.parse(raw) as T) : fallback;
    24→  } catch {
    25→    return fallback;
    26→  }
    27→}
    28→
    29→function writeJson(key: string, value: unknown) {
    30→  try {
    31→    window.localStorage.setItem(key, JSON.stringify(value));
    32→  } catch {
    33→    /* storage full/blocked — ignore */
    34→  }
    35→}

    37→const MYLIST_EVENT = "mylist-changed";
    38→
    39→function subscribeMyList(onChange: () => void) {
    40→  window.addEventListener(MYLIST_EVENT, onChange);
    41→  window.addEventListener("storage", onChange);
    42→  return () => {
    43→    window.removeEventListener(MYLIST_EVENT, onChange);
    44→    window.removeEventListener("storage", onChange);
    45→  };
    46→}
    47→
    48→function getMyListSnapshot(): string {
    49→  return window.localStorage.getItem(MYLIST_KEY) ?? "[]";
    50→}
    51→
    52→function getMyListServerSnapshot(): string {
    53→  return "[]";
    54→}
    55→
    56→export function useMyList() {
    57→  const raw = useSyncExternalStore(subscribeMyList, getMyListSnapshot, getMyListServerSnapshot);
    58→
    59→  const list: MyListEntry[] = (() => {
    60→    try {
    61→      return JSON.parse(raw) as MyListEntry[];
    62→    } catch {
    63→      return [];
    64→    }
    65→  })();
    66→
    67→  const has = useCallback((malId: number) => list.some((e) => e.malId === malId), [list]);
    68→
    69→  const toggle = useCallback((anime: Pick<AnimeLite, "malId" | "title" | "image">) => {
    70→    const current = (() => {
    71→      try {
    72→        return JSON.parse(window.localStorage.getItem(MYLIST_KEY) ?? "[]") as MyListEntry[];
    73→      } catch {
    74→        return [] as MyListEntry[];
    75→      }
    76→    })();
    77→    const exists = current.some((e) => e.malId === anime.malId);
    78→    const next = exists
    79→      ? current.filter((e) => e.malId !== anime.malId)
    80→      : [{ malId: anime.malId, title: anime.title, image: anime.image, addedAt: Date.now() }, ...current];
    81→    writeJson(MYLIST_KEY, next);
    82→    try {
    83→      window.dispatchEvent(new CustomEvent(MYLIST_EVENT));
    84→    } catch {
    85→      /* ignore */
    86→    }
    87→  }, []);
    88→
    89→  const malIds = list.map((e) => e.malId);
    90→  return { list, malIds, has, toggle, ready: true };
    91→}

   182→/**
   183→ * Persist progress locally immediately; mirror to the server. When
   184→ * `keepalive` is set (page unload / tab hide) the request is sent with
   185→ * keepalive so it survives navigation.
   186→ */
   187→export function saveProgressDetailed(args: SaveProgressArgs, opts?: { keepalive?: boolean }) {
   188→  if (!Number.isFinite(args.position) || args.position < 5) return;
   189→  const completed = args.completed ?? (args.duration > 0 && args.position >= args.duration * 0.92);
   190→  const entry: DetailedProgress = {
   191→    pos: Math.floor(args.position),
   192→    dur: Math.floor(args.duration || 0),
   193→    completed,
   194→    lang: args.lang,
   195→    server: args.server,
   196→    title: args.title,
   197→    image: args.image,
   198→    total: args.total,
   199→    updatedAt: Date.now(),
   200→  };
   201→
   202→  // local (instant — bars/resume render without a round trip)
   203→  const map = readEpMap();
   204→  const key = String(args.malId);
   205→  map[key] = map[key] ?? {};
   206→  map[key][String(args.episode)] = entry;
   207→  writeJson(EP_PROGRESS_KEY, map);
   208→
   209→  // legacy "last watched episode" map (details badge, resume fallback)
   210→  try {
   211→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   212→    legacy[key] = { ep: args.episode, lang: args.lang, pos: entry.pos, updatedAt: entry.updatedAt };
   213→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   214→  } catch {
   215→    /* ignore */
   216→  }
   217→
   218→  try {
   219→    window.dispatchEvent(new CustomEvent("progress-changed"));
   220→  } catch {
   221→    /* ignore */
   222→  }
   223→
   224→  // server mirror (best effort, never blocks playback)
   225→  const userId = getUserId();
   226→  const body = JSON.stringify({
   227→    userId,
   228→    malId: args.malId,
   229→    episode: args.episode,
   230→    lang: args.lang,
   231→    server: args.server,
   232→    position: entry.pos,
   233→    duration: entry.dur,
   234→    completed,
   235→    title: args.title,
   236→    image: args.image,
   237→  });
   238→  try {
   239→    if (opts?.keepalive && "sendBeacon" in navigator) {
   240→      navigator.sendBeacon("/api/progress", new Blob([body], { type: "application/json" }));
   241→    } else {
   242→      fetch("/api/progress", {
   243→        method: "POST",
   244→        headers: { "Content-Type": "application/json" },
   245→        body,
   246→        ...(opts?.keepalive ? { keepalive: true } : {}),
   247→      }).catch(() => undefined);
   248→    }
   249→  } catch {
   250→    /* ignore */
   251→  }
   252→}

   254→/** Forget everything watched for one anime (Continue Watching "X" button). */
   255→export function removeProgress(malId: number) {
   256→  const map = readEpMap();
   257→  delete map[String(malId)];
   258→  writeJson(EP_PROGRESS_KEY, map);
   259→  removeEpThumbs(malId);
   260→  try {
   261→    const legacy = JSON.parse(window.localStorage.getItem(PROGRESS_KEY) ?? "{}");
   262→    delete legacy[String(malId)];
   263→    window.localStorage.setItem(PROGRESS_KEY, JSON.stringify(legacy));
   264→  } catch {
   265→    /* ignore */
   266→  }
   267→  try {
   268→    window.dispatchEvent(new CustomEvent("progress-changed"));
   269→  } catch {
   270→    /* ignore */
   271→  }
   272→  fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`, { method: "DELETE" }).catch(
   273→    () => undefined,
   274→  );
   275→}

   277→/**
   278→ * Server-side progress for one anime — used as a fallback so resume points
   279→ * and red bars survive cleared storage / a fresh browser.
   280→ */
   281→export async function fetchServerProgress(malId: number): Promise<Record<string, DetailedProgress>> {
   282→  try {
   283→    const res = await fetch(`/api/progress?userId=${encodeURIComponent(getUserId())}&malId=${malId}`);
   284→    if (!res.ok) return {};
   285→    const data = (await res.json()) as {
   286→      episodes: Array<{ episode: number; position: number; duration: number; completed: boolean; lang: string; server: string }>;
   287→    };
   288→    const out: Record<string, DetailedProgress> = {};
   289→    for (const e of data.episodes ?? []) {
   290→      out[String(e.episode)] = {
   291→        pos: e.position,
   292→        dur: e.duration,
   293→        completed: e.completed,
   294→        lang: e.lang === "dub" ? "dub" : "sub",
   295→        server: e.server,
   296→        updatedAt: Date.now(),
   297→      };
   298→    }
   299→    // merge into the local map (server rows are authoritative for these keys)
   300→    if (Object.keys(out).length) {
   301→      const map = readEpMap();
   302→      const key = String(malId);
   303→      map[key] = { ...(map[key] ?? {}), ...out };
   304→      writeJson(EP_PROGRESS_KEY, map);
   305→    }
   306→    return out;
   307→  } catch {
   308→    return {};
   309→  }
   310→}

   312→/* ---------------- Continue Watching ---------------- */
   313→
   314→/**
   315→ * Freshest in-progress episode per anime — server list merged with the
   316→ * instant local map (so the rail renders with zero latency and still shows
   317→ * entries when the server DB is cold).
   318→ */
   319→export function mergeContinueItems(serverItems: ContinueItem[]): ContinueItem[] {
   320→  let localAll: Record<string, Record<string, DetailedProgress>> = {};
   321→  try {
   322→    localAll = JSON.parse(window.localStorage.getItem(EP_PROGRESS_KEY) ?? "{}");
   323→  } catch {
   324→    localAll = {};
   325→  }
   326→
   327→  type Row = { item: ContinueItem; at: number };
   328→  const byAnime = new Map<number, Row>();
   329→
   330→  for (const it of serverItems) {
   331→    if (it.completed) continue;
   332→    byAnime.set(it.malId, { item: it, at: Date.parse(it.updatedAt) });
   333→  }
   334→
   335→  for (const [malId, eps] of Object.entries(localAll)) {
   336→    for (const [ep, e] of Object.entries(eps)) {
   337→      if (e.completed || (e.pos || 0) < 5) continue;
   338→      const malIdNum = Number(malId);
   339→      const prev = byAnime.get(malIdNum);
   340→      if (prev && prev.at >= e.updatedAt) continue;
   341→      const percent = e.dur > 0 ? Math.min(100, Math.round((e.pos / e.dur) * 100)) : 0;
   342→      byAnime.set(malIdNum, {
   343→        at: e.updatedAt,
   344→        item: {
   345→          malId: malIdNum,
   346→          title: e.title ?? prev?.item.title ?? "Unknown title",
   347→          image: e.image ?? prev?.item.image ?? null,
   348→          banner: prev?.item.banner ?? null,
   349→          totalEpisodes: e.total ?? prev?.item.totalEpisodes ?? 0,
   350→          type: prev?.item.type ?? null,
   351→          episode: Number(ep),
   352→          lang: e.lang,
   353→          server: e.server,
   354→          position: e.pos,
   355→          duration: e.dur,
   356→          percent,
   357→          completed: false,
   358→          updatedAt: new Date(e.updatedAt).toISOString(),
   359→        },
   360→      });
   361→    }
   362→  }
   363→
   364→  return [...byAnime.values()].sort((a, b) => b.at - a.at).map((r) => r.item);
   365→}

   382→/* ---------------- Episode scene thumbnails ---------------- *
   383→ *
   384→ * Crunchyroll-style Continue Watching art: while an episode plays we grab a
   385→ * frame straight off the <video> surface (canvas -> JPEG data URL) and keep
   386→ * it per episode+lang. Streams play through our own same-origin proxy via
   387→ * MSE, so the canvas is never tainted and the capture virtually always
   388→ * succeeds; anything that fails just falls back to the poster art.
   389→ *
   390→ * Each episode captures ONCE (the first session it is watched — that frame
   391→ * is the "random scene"); stored thumbs are evicted LRU-style so the
   392→ * localStorage budget stays bounded.
   393→ */
   394→
   395→const EP_THUMBS_KEY = "animetest123.epthumbs";
   396→const EP_THUMBS_MAX = 80; // ~10-14KB each at 224x126 q0.62 -> well under quota
   397→
   398→type ThumbEntry = { t: string; at: number };
   399→type ThumbMap = Record<string, Record<string, ThumbEntry>>;
   400→
   401→function readThumbs(): ThumbMap {
   402→  return readJson<ThumbMap>(EP_THUMBS_KEY, {});
   403→}
   404→
   405→function writeThumbs(map: ThumbMap) {
   406→  // global LRU cap — evict the oldest captures first
   407→  const all: Array<{ malId: string; ep: string; at: number }> = [];
   408→  for (const [malId, eps] of Object.entries(map)) {
   409→    for (const [ep, entry] of Object.entries(eps)) all.push({ malId, ep, at: entry.at });
   410→  }
   411→  if (all.length > EP_THUMBS_MAX) {
   412→    all.sort((a, b) => a.at - b.at);
   413→    for (const victim of all.slice(0, all.length - EP_THUMBS_MAX)) delete map[victim.malId][victim.ep];
   414→  }
   415→  writeJson(EP_THUMBS_KEY, map);
   416→}

   425→export function saveEpThumb(malId: number, episode: number, lang: "sub" | "dub", dataUrl: string) {

   436→    } catch {
   437→      /* ignore */
   438→    }
   439→  } catch {
   440→    /* quota / private mode — thumbnails are decorative, never fatal */
   441→  }

   454→/* ---------------- Watch time (Top 10 Today) ---------------- *
   455→ *
   456→ * "Top 10 Today" ranks anime by how long people ACTUALLY watch them — the
   457→ * rail stays empty until real playback seconds exist. The player records
   458→ * wall-clock deltas while the video is playing; they accumulate locally
   459→ * (instant, offline-safe) and flush to the server so EVERY viewer's time
   460→ * lands in one global per-day ranking, Crunchyroll-style.
   461→ *
   462→ * Local shape: day -> malId -> { seconds, title, image, sent }.
   463→ * `sent` is the amount already ACKed by the server, so flushes POST only
   464→ * the un-sent remainder. Display merges server + local with max() — never
   465→ * a sum — so nothing is double-counted and offline time still shows.
   466→ */
   467→
   468→const WATCHTIME_KEY = "animetest123.watchtime";
   469→const WATCHTIME_EVENT = "watchtime-changed";
   470→const WATCHTIME_KEEP_DAYS = 7; // local buckets older than a week are dropped
   471→
   472→export interface LocalWatchEntry {
   473→  seconds: number;
   474→  title?: string;
   475→  image?: string | null;
   476→  sent?: number; // seconds already accepted by the server
   477→}
   478→
   479→type WatchTimeMap = Record<string, Record<string, LocalWatchEntry>>;
   480→
   481→/**
   482→ * The SHARED calendar day ("YYYY-MM-DD", UTC) — one global bucket, so every
   483→ * device in every timezone contributes to (and sees) the exact same
   484→ * "Top 10 Today" ranking, like Netflix's worldwide lists. Rolls over at
   485→ * UTC midnight for everyone.
   486→ */
   487→export function localDayKey(d = new Date()): string {
   488→  return d.toISOString().slice(0, 10);
   489→}
   490→
   491→function readWatchTime(): WatchTimeMap {
   492→  return readJson<WatchTimeMap>(WATCHTIME_KEY, {});
   493→}
   494→
   495→function writeWatchTime(map: WatchTimeMap) {
   496→  // prune: keep only the newest KEEP_DAYS day buckets
   497→  const days = Object.keys(map).sort();
   498→  for (const stale of days.slice(0, Math.max(0, days.length - WATCHTIME_KEEP_DAYS))) delete map[stale];
   499→  writeJson(WATCHTIME_KEY, map);
   500→}
   501→
   502→/**
   503→ * Add `delta` seconds of real playback to today's bucket for this anime and
   504→ * flush the un-sent remainder to the server (best effort, never throws).
   505→ * `keepalive` switches to sendBeacon for pagehide/unload flushes.
   506→ */
   507→export function recordWatchSeconds(
   508→  args: { malId: number; title?: string; image?: string | null; delta: number; day?: string },
   509→  opts?: { keepalive?: boolean },
   510→) {
   511→  const delta = Math.min(120, Math.max(0, Number(args.delta) || 0));
   512→  if (delta < 1 || !Number.isFinite(args.malId)) return;
   513→  const day = args.day ?? localDayKey(); // shared UTC day — one global ranking
   514→
   515→  let pending = 0;
   516→  try {
   517→    const map = readWatchTime();
   518→    const dayBucket = (map[day] = map[day] ?? {});
   519→    const entry = dayBucket[String(args.malId)] ?? { seconds: 0, sent: 0 };
   520→    entry.seconds += delta;
   521→    if (args.title) entry.title = args.title;
   522→    if (args.image !== undefined) entry.image = args.image;
   523→    dayBucket[String(args.malId)] = entry;
   524→    pending = entry.seconds - (entry.sent ?? 0);
   525→    if (pending >= 5) {
   526→      entry.sent = entry.seconds; // optimistic — reverted below if the send fails outright
   527→      writeWatchTime(map);
   528→    } else {
   529→      pending = 0;
   530→      writeWatchTime(map);
   531→    }
   532→  } catch {
   533→    return; // storage blocked — ranking is decorative, never fatal
   534→  }
   535→
   536→  if (pending < 5) {
   537→    // still notify listeners so Home can re-rank from the local bucket alone
   538→    try {
   539→      window.dispatchEvent(new CustomEvent(WATCHTIME_EVENT));
   540→    } catch {
   541→      /* ignore */
   542→    }
   543→    return;
   544→  }
   545→
   546→  const body = JSON.stringify({
   547→    userId: getUserId(),
   548→    malId: args.malId,
   549→    day,
   550→    delta: Math.round(pending),
   551→    title: args.title,
   552→    image: args.image,
   553→  });
   554→  const revert = () => {
   555→    try {
   556→      const map = readWatchTime();
   557→      const entry = map[day]?.[String(args.malId)];
   558→      if (entry) {
   559→        entry.sent = Math.max(0, (entry.sent ?? 0) - pending);
   560→        writeWatchTime(map);
   561→      }
   562→    } catch {
   563→      /* ignore */
   564→    }
   565→  };
   566→  try {
   567→    if (opts?.keepalive && "sendBeacon" in navigator) {
   568→      navigator.sendBeacon("/api/watchtime", new Blob([body], { type: "application/json" }));
   569→    } else {
   570→      fetch("/api/watchtime", {
   571→        method: "POST",
   572→        headers: { "Content-Type": "application/json" },
   573→        body,
   574→      })
   575→        .then((res) => {
   576→          if (!res.ok) revert(); // server rejected — retry on a later tick
   577→        })
   578→        .catch(revert);
   579→    }
   580→  } catch {
   581→    revert();
   582→  }
   583→
   584→  try {

   654→/**
   655→ * The day's most-watched anime: the server's global ranking merged with this
   656→ * device's instant local buckets (max, not sum — no double counting; local
   657→ * covers the un-flushed tail and offline/served-cold cases).
   658→ */
   659→export async function fetchWatchTimeTop(day: string, limit = 10): Promise<TopWatchEntry[]> {
   660→  let serverItems: TopWatchEntry[] = [];
   661→  try {
   662→    const res = await fetch(`/api/watchtime?day=${encodeURIComponent(day)}&limit=${limit}`);
   663→    if (res.ok) {
   664→      const data = (await res.json()) as { items?: TopWatchEntry[] };
   665→      serverItems = data.items ?? [];
   666→    }
   667→  } catch {
   668→    /* offline / cold server — local buckets only */
   669→  }
   670→
   671→  const byId = new Map<number, TopWatchEntry>();
   672→  for (const it of serverItems) {
   673→    if (Number.isFinite(it?.malId)) byId.set(it.malId, it);
   674→  }
   675→  try {
   676→    const local = readWatchTime()[day] ?? {};
   677→    for (const [key, entry] of Object.entries(local)) {
   678→      const malId = Number(key);
   679→      if (!Number.isFinite(malId)) continue;
   680→      const prev = byId.get(malId);
   681→      const seconds = Math.max(prev?.seconds ?? 0, entry.seconds || 0);
   682→      if (seconds < 5) continue; // a stray couple of seconds isn't "watched"
   683→      byId.set(malId, {
   684→        ...(prev ?? minimalLite(malId, entry.title || `Anime ${malId}`, entry.image ?? null)),
   685→        // local snapshots keep the rail informative when the catalog row is gone
   686→        title: prev?.title || entry.title || `Anime ${malId}`,
   687→        image: prev?.image ?? entry.image ?? null,
   688→        malId,
   689→        seconds,
   690→      });
   691→    }
   692→  } catch {
   693→    /* ignore */
   694→  }
   695→
   696→  return [...byId.values()].sort((a, b) => b.seconds - a.seconds).slice(0, limit);
   697→}

   722→export function getLangPref(malId?: number): LangPref {

   728→    return { lang: "sub", server: "zoko" };

   807→export function timeAgo(iso: string | number): string {
   808→  const t = typeof iso === "number" ? iso : Date.parse(iso);
   809→  const diff = Date.now() - t;
   810→  const min = Math.floor(diff / 60_000);
   811→  if (min < 1) return "just now";
   812→  if (min < 60) return `${min}m ago`;
   813→  const hr = Math.floor(min / 60);
   814→  if (hr < 24) return `${hr}h ago`;
   815→  const day = Math.floor(hr / 24);
   816→  if (day < 30) return `${day}d ago`;
   817→  return new Date(t).toLocaleDateString();
   818→}
